const AFFILIATE_PORTAL_API_ENDPOINT =
  'https://panglima-lynk-webhook.ruang-belajar-mandiri01.workers.dev/portal';

const AFFILIATE_PORTAL_SESSION_KEY =
  'umkmnaikkelas_affiliate_portal_session_v1';


function affiliatePortalFormatRupiah(value) {
  const amount = Number(value || 0);

  return new Intl.NumberFormat(
    'id-ID',
    {
      style: 'currency',
      currency: 'IDR',
      maximumFractionDigits: 0,
    }
  ).format(
    Number.isFinite(amount)
      ? amount
      : 0
  );
}


function affiliatePortalFormatPercent(value) {
  const number = Number(value || 0);

  return new Intl.NumberFormat(
    'id-ID',
    {
      style: 'percent',
      minimumFractionDigits: 0,
      maximumFractionDigits: 1,
    }
  ).format(
    Number.isFinite(number)
      ? number
      : 0
  );
}


function affiliatePortalFormatDate(value) {
  if (!value) return '-';

  const date = new Date(value);

  if (Number.isNaN(date.getTime())) {
    return '-';
  }

  return new Intl.DateTimeFormat(
    'id-ID',
    {
      day: '2-digit',
      month: 'short',
      year: 'numeric',
      hour: '2-digit',
      minute: '2-digit',
    }
  ).format(date);
}


function affiliatePortalReadSession() {
  if (
    typeof window === 'undefined'
  ) {
    return null;
  }

  try {
    const raw =
      window.localStorage.getItem(
        AFFILIATE_PORTAL_SESSION_KEY
      );

    if (!raw) {
      return null;
    }

    const parsed =
      JSON.parse(raw);

    if (
      !parsed?.token ||
      !parsed?.expiresAt
    ) {
      window.localStorage.removeItem(
        AFFILIATE_PORTAL_SESSION_KEY
      );

      return null;
    }

    const expiresAt =
      new Date(
        parsed.expiresAt
      ).getTime();

    if (
      !expiresAt ||
      Date.now() >= expiresAt
    ) {
      window.localStorage.removeItem(
        AFFILIATE_PORTAL_SESSION_KEY
      );

      return null;
    }

    return parsed;

  } catch (error) {
    window.localStorage.removeItem(
      AFFILIATE_PORTAL_SESSION_KEY
    );

    return null;
  }
}


function affiliatePortalSaveSession(
  token,
  expiresAt,
  email
) {
  if (
    typeof window === 'undefined'
  ) {
    return;
  }

  try {
    window.localStorage.setItem(
      AFFILIATE_PORTAL_SESSION_KEY,
      JSON.stringify({
        token,
        expiresAt,
        email,
      })
    );
  } catch (error) {
    // Jika storage browser tidak tersedia,
    // login tetap berjalan untuk sesi halaman saat ini.
  }
}


function affiliatePortalClearSession() {
  if (
    typeof window === 'undefined'
  ) {
    return;
  }

  try {
    window.localStorage.removeItem(
      AFFILIATE_PORTAL_SESSION_KEY
    );
  } catch (error) {
    // no-op
  }
}


async function affiliatePortalApi(
  payload
) {
  const response =
    await fetch(
      AFFILIATE_PORTAL_API_ENDPOINT,
      {
        method: 'POST',

        headers: {
          'Content-Type':
            'application/json',
        },

        body:
          JSON.stringify({
            ...payload,

            userAgent:
              typeof navigator !==
              'undefined'
                ? navigator.userAgent
                : '',
          }),
      }
    );

  let data = null;

  try {
    data =
      await response.json();
  } catch (error) {
    data = null;
  }

  if (!response.ok) {
    const err =
      new Error(
        data?.message ||
        data?.error ||
        'Portal sedang tidak dapat diakses.'
      );

    err.portalData = data;
    throw err;
  }

  return data || {};
}


function AffiliatePortalMetricCard({
  label,
  value,
  helper,
}) {
  return (
    <div className="rounded-2xl border border-black/10 bg-white p-5 shadow-sm">
      <p className="text-xs font-bold uppercase tracking-[0.12em] text-neutral-500">
        {label}
      </p>

      <p className="mt-2 text-2xl font-black tracking-tight text-neutral-900 sm:text-3xl">
        {value}
      </p>

      {helper ? (
        <p className="mt-2 text-xs leading-relaxed text-neutral-500">
          {helper}
        </p>
      ) : null}
    </div>
  );
}


function AffiliatePortalPage() {
  const [stage, setStage] =
    useState('loading');

  const [email, setEmail] =
    useState('');

  const [otp, setOtp] =
    useState('');

  const [sessionToken, setSessionToken] =
    useState('');

  const [dashboard, setDashboard] =
    useState(null);

  const [loading, setLoading] =
    useState(false);

  const [message, setMessage] =
    useState('');

  const [error, setError] =
    useState('');

  const [copiedKey, setCopiedKey] =
    useState('');


  const showError =
    (text) => {
      setMessage('');
      setError(
        String(
          text ||
          'Terjadi kesalahan.'
        )
      );
    };


  const showMessage =
    (text) => {
      setError('');
      setMessage(
        String(
          text || ''
        )
      );
    };


  const loadDashboard =
    async (
      token,
      options = {}
    ) => {
      const {
        quiet = false,
      } = options;

      if (!token) {
        setStage('email');
        return;
      }

      if (!quiet) {
        setLoading(true);
      }

      try {
        const result =
          await affiliatePortalApi({
            action:
              'portal_dashboard',

            sessionToken:
              token,
          });

        if (
          !result?.ok ||
          !result?.authenticated
        ) {
          affiliatePortalClearSession();

          setSessionToken('');
          setDashboard(null);
          setStage('email');

          if (!quiet) {
            showError(
              'Sesi login berakhir. Silakan masuk kembali.'
            );
          }

          return;
        }

        setDashboard(result);
        setStage('dashboard');
        setError('');

      } catch (err) {
        if (
          err?.portalData?.error ===
            'SESSION_INVALID_OR_EXPIRED' ||
          err?.portalData?.error ===
            'SESSION_REQUIRED' ||
          err?.portalData?.error ===
            'AFFILIATE_NOT_ACTIVE'
        ) {
          affiliatePortalClearSession();

          setSessionToken('');
          setDashboard(null);
          setStage('email');

          showError(
            'Sesi login berakhir. Silakan masuk kembali.'
          );

          return;
        }

        showError(
          err?.message ||
          'Dashboard belum dapat dimuat.'
        );

      } finally {
        if (!quiet) {
          setLoading(false);
        }
      }
    };


  useEffect(() => {
    const stored =
      affiliatePortalReadSession();

    if (!stored) {
      setStage('email');
      return;
    }

    setEmail(
      stored.email || ''
    );

    setSessionToken(
      stored.token
    );

    loadDashboard(
      stored.token
    );
  }, []);


  const handleRequestOtp =
    async (event) => {
      event.preventDefault();

      const normalizedEmail =
        String(email || '')
          .trim()
          .toLowerCase();

      if (
        !/^[^\s@]+@[^\s@]+\.[^\s@]+$/
          .test(normalizedEmail)
      ) {
        showError(
          'Masukkan alamat email yang valid.'
        );

        return;
      }

      setLoading(true);
      setError('');
      setMessage('');

      try {
        const result =
          await affiliatePortalApi({
            action:
              'portal_request_otp',

            email:
              normalizedEmail,
          });

        if (!result?.ok) {
          showError(
            result?.message ||
            'Kode OTP belum dapat dikirim.'
          );

          return;
        }

        setEmail(
          normalizedEmail
        );

        setOtp('');
        setStage('otp');

        showMessage(
          'Jika email terdaftar sebagai affiliate aktif, kode OTP telah dikirim. Periksa Inbox atau Spam.'
        );

      } catch (err) {
        showError(
          err?.message ||
          'Kode OTP belum dapat dikirim.'
        );

      } finally {
        setLoading(false);
      }
    };


  const handleVerifyOtp =
    async (event) => {
      event.preventDefault();

      const cleanOtp =
        String(
          otp || ''
        )
          .replace(/\D/g, '')
          .slice(0, 6);

      if (
        !/^\d{6}$/.test(
          cleanOtp
        )
      ) {
        showError(
          'Masukkan kode OTP 6 digit dari email.'
        );

        return;
      }

      setLoading(true);
      setError('');
      setMessage('');

      try {
        const result =
          await affiliatePortalApi({
            action:
              'portal_verify_otp',

            email:
              String(email || '')
                .trim()
                .toLowerCase(),

            otp:
              cleanOtp,
          });

        if (
          !result?.ok ||
          !result?.authenticated ||
          !result?.sessionToken
        ) {
          showError(
            result?.message ||
            'Kode OTP salah atau sudah kedaluwarsa.'
          );

          return;
        }

        affiliatePortalSaveSession(
          result.sessionToken,
          result.expiresAt,
          email
        );

        setSessionToken(
          result.sessionToken
        );

        setOtp('');

        await loadDashboard(
          result.sessionToken
        );

      } catch (err) {
        showError(
          err?.message ||
          'Kode OTP salah atau sudah kedaluwarsa.'
        );

      } finally {
        setLoading(false);
      }
    };


  const handleResendOtp =
    async () => {
      if (loading) return;

      setLoading(true);
      setError('');
      setMessage('');

      try {
        const result =
          await affiliatePortalApi({
            action:
              'portal_request_otp',

            email:
              String(email || '')
                .trim()
                .toLowerCase(),

          showError(
            result?.message ||
            'Kode OTP belum dapat dikirim.'
          );

          return;
        }

        showMessage(
          'Kode OTP dikirim. Jika baru saja meminta OTP, gunakan kode yang masih aktif di email.'
        );

      } catch (err) {
        showError(
          err?.message ||
          'Kode OTP belum dapat dikirim.'
        );

      } finally {
        setLoading(false);
      }
    };


  const handleLogout =
    async () => {

        sessionToken;

      setDashboard(null);
      setSessionToken('');
      setOtp('');
      setMessage('');
      setError('');
      setStage('email');

      if (!token) return;

      try {
        await affiliatePortalApi({
          action:
            'portal_logout',

          sessionToken:
            token,
        });
      } catch (error) {
        // Logout lokal sudah selesai.
      }
    };


  const handleCopy =
    async (
      value,
      key
    ) => {
      if (!value) return;

      try {
        await navigator.clipboard.writeText(
          value
        );

        setCopiedKey(key);

        window.setTimeout(
          () => {
            setCopiedKey('');
          },
          1800
        );

        showError(
          'Tidak dapat menyalin otomatis. Tekan dan tahan link untuk menyalin.'

      }
    };

    dashboard?.affiliate ||
    {};

    dashboard?.metrics ||
    {};

    metrics?.monthlyBonus ||
    {};

    Array.isArray(
      dashboard?.transactions
    )
      ? dashboard.transactions
      : [];

    Number(
      monthlyBonus?.nextTarget ||
      0
    );

    Number(
      metrics?.currentMonthSales ||
      0
    );

      ? Math.min(
          100,
          Math.max(
            0,
            (
              currentMonthSales /
              bonusNextTarget
            ) * 100
          )
        )
      : 100;


  if (
    stage === 'loading'
  ) {
    return (
      <>
        <Helmet>
          <title>
            Portal Affiliate | UMKM Naik Kelas
          </title>

          <meta
            name="robots"
            content="noindex,nofollow"
          />
        </Helmet>

        <main className="flex min-h-screen items-center justify-center bg-[#faf8f3] px-5">
          <div className="text-center">
            <div className="mx-auto h-10 w-10 animate-spin rounded-full border-4 border-neutral-200 border-t-red-700" />

            <p className="mt-4 text-sm font-semibold text-neutral-600">
              Membuka Portal Affiliate…
            </p>
          </div>
        </main>
      </>
    );
  }


  if (
    stage !== 'dashboard'
  ) {
    return (
      <>
        <Helmet>
          <title>
            Login Portal Affiliate | UMKM Naik Kelas
          </title>

          <meta
            name="description"
            content="Login Portal Affiliate UMKM Naik Kelas."
          />

          <meta
            name="robots"
            content="noindex,nofollow"
          />
        </Helmet>

        <main className="min-h-screen bg-[#faf8f3] px-5 py-10 sm:py-16">
          <div className="mx-auto max-w-[460px]">

            <a
              href="/"
              className="mb-7 inline-flex items-center gap-2 text-sm font-bold text-neutral-600 hover:text-red-800"
            >
              ← Kembali ke UMKM Naik Kelas
            </a>

            <div className="overflow-hidden rounded-[28px] border border-black/10 bg-white shadow-xl shadow-black/5">

              <div className="bg-gradient-to-br from-[#760d10] via-[#9f1519] to-[#5d090b] px-7 py-8 text-white">

                  Pasukan Panglima UMKM

                  Portal Affiliate

                <p className="mt-3 max-w-sm text-sm leading-relaxed text-white/80">
                  Pantau klik, penjualan, komisi, bonus bulanan, dan link affiliate Anda.
                </p>
              </div>

              <div className="p-7">

                {stage === 'email' ? (
                  <form
                    onSubmit={
                      handleRequestOtp
                    }
                  >
                    <label
                      htmlFor="affiliate-portal-email"
                      className="text-sm font-bold text-neutral-800"
                    >
                      Email Affiliate
                    </label>

                    <input
                      id="affiliate-portal-email"
                      type="email"
                      autoComplete="email"
                      value={email}
                      onChange={
                        (event) =>
                          setEmail(
                            event.target.value
                          )
                      }
                      placeholder="user_71ec9927@example.invalid"
                      className="mt-2 min-h-[52px] w-full rounded-xl border border-black/15 bg-white px-4 text-base outline-none transition focus:border-red-700 focus:ring-4 focus:ring-red-100"
                      required
                    />

                    <button
                      type="submit"
                      disabled={loading}
                      className="mt-5 inline-flex min-h-[52px] w-full items-center justify-center rounded-xl bg-[#8c0d10] px-5 text-sm font-black text-white transition hover:bg-[#6f090c] disabled:cursor-not-allowed disabled:opacity-60"
                    >
                      {loading
                        ? 'Mengirim OTP…'
                        : 'Kirim Kode OTP'}
                    </button>

                    <p className="mt-4 text-center text-xs leading-relaxed text-neutral-500">
                      Hanya affiliate dengan status APPROVED dan email terverifikasi yang dapat masuk.

                      {email}

                      className="mt-6 block text-sm font-bold text-neutral-800"

                      inputMode="numeric"
                      autoComplete="one-time-code"
                      value={otp}
                      onChange={
                        (event) =>
                          setOtp(
                            event.target.value

                      }

                      maxLength={6}

                        Kirim Ulang OTP

                        type="button"
                        onClick={
                          () => {
                            setStage(
                              'email'
                            );

                            setOtp('');
                            setMessage('');
                            setError('');
                          }
                        }

                        Ganti Email

                {message ? (

                    loadDashboard(
                      sessionToken

                className="rounded-xl border border-black/10 bg-white px-4 py-2.5 text-xs font-black text-neutral-700 hover:bg-neutral-50 disabled:opacity-50"

                onClick={
                  handleLogout
                }
                className="rounded-xl bg-[#8c0d10] px-4 py-2.5 text-xs font-black text-white hover:bg-[#6f090c]"

                Keluar

            </div>
          </div>
        </header>


        <div className="mx-auto max-w-7xl px-5 py-8 sm:py-10">

          {error ? (
            <div className="mb-6 rounded-xl border border-red-200 bg-red-50 px-4 py-3 text-sm text-red-800">
              {error}
            </div>
          ) : null}


          <section className="overflow-hidden rounded-[26px] bg-gradient-to-br from-[#700b0e] via-[#951317] to-[#5c080a] p-6 text-white shadow-xl shadow-red-950/10 sm:p-8">

function AffiliateQueryGateway() {
  const location =
    useLocation();

  const params =
    new URLSearchParams(
      location.search
    );

  const portalMode =
    String(
      params.get('portal') ||
      ''
    )
      .trim()
      .toLowerCase();

  const affiliateSlug =
    normalizeAffiliateSlug(
      params.get(
        'affiliate'
      ) || ''
    );

  const isKit =
    params.get('kit') ===
    '1';

  const [
    partner,
    setPartner,
  ] =
    useState(
      () =>
        getAffiliatePartnerBySlug(
          affiliateSlug
        )
    );

  const [
    registryLoading,
    setRegistryLoading,
  ] =
    useState(
      Boolean(
        affiliateSlug
      ) &&
      !partner
    );

  useEffect(
    () => {
      if (
        !affiliateSlug
      ) {
        return undefined;
      }

      let cancelled =
        false;

      const syncPartner =
        async () => {
          if (
            !getAffiliatePartnerBySlug(
              affiliateSlug
            )
          ) {
            setRegistryLoading(
              true
            );
          }

          await loadAffiliateRegistry();

          if (
            cancelled
          ) {
            return;
          }

          setPartner(
            getAffiliatePartnerBySlug(
              affiliateSlug
            )
          );

          setRegistryLoading(
            false
          );
        };

      syncPartner();

      return () => {
        cancelled =
          true;
      };
    },
    [
      affiliateSlug,
    ]
  );

  /*
   * STEP 10B:
   * Portal dicek SETELAH seluruh hooks dipanggil
   * agar Rules of Hooks React tetap aman.
   */
  if (
    portalMode ===
    'affiliate'
  ) {
    return (
      <AffiliatePortalPage />
    );
  }

  if (
    !affiliateSlug
  ) {
    return (
      <HomePage />
    );
  }

  if (
    registryLoading
  ) {
    return (
      <main className="flex min-h-[60vh] items-center justify-center bg-white px-5">
        <div className="text-center">

          <div className="mx-auto h-9 w-9 animate-spin rounded-full border-4 border-neutral-200 border-t-red-600" />

          <p className="mt-4 text-sm font-semibold text-neutral-600">
            Memuat halaman Partner Panglima…
          </p>

        </div>
      </main>
    );
  }

  if (
    !isActiveAffiliatePartner(
      partner
    )
  ) {
    return (
      <NotFoundPage />
    );
  }

  if (
    isKit
  ) {
    return (
      <AffiliateMarketingKitPage
        initialPartner={
          partner
        }
        initialSlug={
          affiliateSlug
        }
      />
    );
  }

  return (
    <AffiliateSalesPage
      initialPartner={
        partner
      }
      initialSlug={
        affiliateSlug
      }
    />
  );
}
