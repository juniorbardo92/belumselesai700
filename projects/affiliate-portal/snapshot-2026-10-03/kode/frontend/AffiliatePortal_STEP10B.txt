const PORTAL_SESSIONS_SHEET = 'Affiliate Portal Sessions';
const PORTAL_REGISTRY_SHEET = 'Affiliate Registry';
const PORTAL_TRANSACTIONS_SHEET = 'Affiliate Transactions';
const PORTAL_COMMISSIONS_SHEET = 'Affiliate Commissions';

const PORTAL_OTP_TTL_MINUTES = 10;
const PORTAL_OTP_MAX_ATTEMPTS = 5;
const PORTAL_SESSION_DAYS = 7;

const PORTAL_SENDER_ALIAS = 'halo@umkmnaikkelas.id';


/* =========================================================
   MAIN PORTAL ROUTER
   ========================================================= */

function portalHandleAction_(action, payload, e) {
  action = String(action || '')
    .trim()
    .toLowerCase();

  payload = payload || {};

  if (action === 'portal_request_otp') {
    return portalRequestOtp_(payload, e);
  }

  if (action === 'portal_verify_otp') {
    return portalVerifyOtp_(payload, e);
  }

  if (action === 'portal_dashboard') {
    return portalGetDashboard_(payload, e);
  }

  if (action === 'portal_logout') {
    return portalLogout_(payload, e);
  }

  return {
    ok: false,
    error: 'UNKNOWN_PORTAL_ACTION'
  };
}


/* =========================================================
   REQUEST OTP
   ========================================================= */

function portalRequestOtp_(payload, e) {
  const email = portalNormalizeEmail_(
    payload.email
  );

  if (!portalIsValidEmail_(email)) {
    return {
      ok: false,
      error: 'INVALID_EMAIL',
      message: 'Masukkan email affiliate yang valid.'
    };
  }

  const affiliate =
    portalFindApprovedAffiliateByEmail_(
      email
    );

  /*
   * Security:
   * Jangan bocorkan apakah email terdaftar
   * atau tidak.
   */
  if (!affiliate) {
    Utilities.sleep(250);

    return {
      ok: true,
      message:
        'Jika email terdaftar sebagai affiliate aktif, kode OTP akan dikirim.'
    };
  }

  const lock =
    LockService.getScriptLock();

  lock.waitLock(15000);

  try {
    const recent =
      portalFindRecentOtpRequest_(
        email
      );

    /*
     * Rate limit sederhana:
     * OTP baru tidak dikirim jika
     * request sebelumnya < 60 detik.
     */
    if (
      recent &&
      recent.ageSeconds < 60
    ) {
      return {
        ok: true,
        message:
          'Kode OTP baru saja dikirim. Periksa email Anda.'
      };
    }

    portalEnsureSecuritySalt_();

    const otp = String(
      Math.floor(
        100000 +
        Math.random() * 900000
      )
    );

    const recordId =
      'OTP-' +
      Utilities.getUuid();

    const createdAt =
      new Date();

    const expiresAt =
      new Date(
        createdAt.getTime() +
        PORTAL_OTP_TTL_MINUTES *
        60 *
        1000
      );

    /*
     * OTP tidak disimpan mentah.
     * Yang disimpan hanya hash.
     */
    const otpHash =
      portalHash_(
        email +
        '|' +
        otp +
        '|' +
        recordId
      );

    const userAgent =
      portalGetUserAgent_(e);

    const sheet =
      portalGetSheet_(
        PORTAL_SESSIONS_SHEET
      );

    sheet.appendRow([
      recordId,          // A Record ID
      createdAt,         // B Created At
      email,             // C Email
      affiliate.code,    // D Affiliate Code
      affiliate.slug,    // E Affiliate Slug
      otpHash,           // F OTP Hash
      expiresAt,         // G OTP Expires At
      'PENDING',         // H OTP Status
      0,                 // I OTP Attempts
      '',                // J Session Token Hash
      '',                // K Session Expires At
      '',                // L Last Seen At
      'OTP_PENDING',     // M Status
      userAgent,         // N User Agent
      ''                 // O Notes
    ]);

    portalSendOtpEmail_(
      email,
      affiliate.name,
      otp,
      PORTAL_OTP_TTL_MINUTES
    );

    return {
      ok: true,
      message:
        'Jika email terdaftar sebagai affiliate aktif, kode OTP akan dikirim.',
      expiresInMinutes:
        PORTAL_OTP_TTL_MINUTES
    };

  } finally {
    lock.releaseLock();
  }
}


/* =========================================================
   VERIFY OTP
   ========================================================= */

function portalVerifyOtp_(payload, e) {
  const email =
    portalNormalizeEmail_(
      payload.email
    );

  const otp =
    String(
      payload.otp || ''
    )
      .replace(/\D/g, '')
      .trim();

  if (
    !portalIsValidEmail_(email) ||
    !/^\d{6}$/.test(otp)
  ) {
    return {
      ok: false,
      error:
        'INVALID_OTP_REQUEST',
      message:
        'Email atau kode OTP tidak valid.'
    };
  }

  const affiliate =
    portalFindApprovedAffiliateByEmail_(
      email
    );

  if (!affiliate) {
    return {
      ok: false,
      error:
        'OTP_INVALID_OR_EXPIRED',
      message:
        'Kode OTP salah atau sudah kedaluwarsa.'
    };
  }

  const lock =
    LockService.getScriptLock();

  lock.waitLock(15000);

  try {
    const record =
      portalFindLatestPendingOtp_(
        email
      );

    if (!record) {
      return {
        ok: false,
        error:
          'OTP_INVALID_OR_EXPIRED',
        message:
          'Kode OTP salah atau sudah kedaluwarsa.'
      };
    }

    const now =
      new Date();

    const expiresAt =
      portalAsDate_(
        record.values[6]
      );

    const attempts =
      Number(
        record.values[8] || 0
      );

    /*
     * OTP kedaluwarsa
     */
    if (
      !expiresAt ||
      now.getTime() >
      expiresAt.getTime()
    ) {
      portalUpdateSessionRow_(
        record.row,
        {
          otpStatus: 'EXPIRED',
          status: 'OTP_EXPIRED',
          notes:
            'OTP expired before successful verification.'
        }
      );

      return {
        ok: false,
        error:
          'OTP_INVALID_OR_EXPIRED',
        message:
          'Kode OTP salah atau sudah kedaluwarsa.'
      };
    }

    /*
     * Maksimal percobaan
     */
    if (
      attempts >=
      PORTAL_OTP_MAX_ATTEMPTS
    ) {
      portalUpdateSessionRow_(
        record.row,
        {
          otpStatus: 'LOCKED',
          status: 'OTP_LOCKED',
          notes:
            'Maximum OTP attempts reached.'
        }
      );

      return {
        ok: false,
        error: 'OTP_LOCKED',
        message:
          'Terlalu banyak percobaan. Minta kode OTP baru.'
      };
    }

    const expectedHash =
      portalHash_(
        email +
        '|' +
        otp +
        '|' +
        record.values[0]
      );

    const storedHash =
      String(
        record.values[5] || ''
      );

    /*
     * OTP salah
     */
    if (
      !portalSafeEqual_(
        expectedHash,
        storedHash
      )
    ) {
      const newAttempts =
        attempts + 1;

      portalUpdateSessionRow_(
        record.row,
        {
          otpAttempts:
            newAttempts,
          notes:
            'Invalid OTP attempt.'
        }
      );

      return {
        ok: false,
        error:
          'OTP_INVALID_OR_EXPIRED',
        message:
          'Kode OTP salah atau sudah kedaluwarsa.',
        attemptsRemaining:
          Math.max(
            0,
            PORTAL_OTP_MAX_ATTEMPTS -
            newAttempts
          )
      };
    }

    /*
     * OTP benar:
     * buat session token.
     */
    const sessionToken =
      Utilities
        .getUuid()
        .replace(/-/g, '') +
      Utilities
        .getUuid()
        .replace(/-/g, '');

    const sessionHash =
      portalHash_(
        sessionToken
      );

    const sessionExpiresAt =
      new Date(
        now.getTime() +
        PORTAL_SESSION_DAYS *
        24 *
        60 *
        60 *
        1000
      );

    portalUpdateSessionRow_(
      record.row,
      {
        otpStatus:
          'VERIFIED',
        otpAttempts:
          attempts,
        sessionTokenHash:
          sessionHash,
        sessionExpiresAt:
          sessionExpiresAt,
        lastSeenAt:
          now,
        status:
          'ACTIVE',
        notes:
          'Portal login successful.'
      }
    );

    /*
     * Token mentah hanya dikirim
     * sekali ke browser.
     * Di Sheet hanya hash.
     */
    return {
      ok: true,
      authenticated: true,

      sessionToken:
        sessionToken,

      expiresAt:
        sessionExpiresAt
          .toISOString(),

      affiliate: {
        name:
          affiliate.name,
        code:
          affiliate.code,
        slug:
          affiliate.slug
      }
    };

  } finally {
    lock.releaseLock();
  }
}


/* =========================================================
   DASHBOARD
   ========================================================= */

function portalGetDashboard_(payload, e) {
  const token =
    String(
      payload.sessionToken ||
      payload.session_token ||
      ''
    ).trim();

  if (!token) {
    return {
      ok: false,
      authenticated: false,
      error:
        'SESSION_REQUIRED'
    };
  }

  const session =
    portalFindActiveSession_(
      token
    );

  if (!session) {
    return {
      ok: false,
      authenticated: false,
      error:
        'SESSION_INVALID_OR_EXPIRED'
    };
  }

  const email =
    portalNormalizeEmail_(
      session.values[2]
    );

  const code =
    String(
      session.values[3] || ''
    ).trim();

  const slug =
    String(
      session.values[4] || ''
    ).trim();

  /*
   * Affiliate diverifikasi ulang
   * setiap dashboard dibuka.
   */
  const affiliate =
    portalFindApprovedAffiliateByEmail_(
      email
    );

  if (
    !affiliate ||
    affiliate.code !== code ||
    affiliate.slug !== slug
  ) {
    portalUpdateSessionRow_(
      session.row,
      {
        status:
          'REVOKED',
        notes:
          'Affiliate is no longer eligible.'
      }
    );

    return {
      ok: false,
      authenticated: false,
      error:
        'AFFILIATE_NOT_ACTIVE'
    };
  }

  const now =
    new Date();

  portalUpdateSessionRow_(
    session.row,
    {
      lastSeenAt:
        now
    }
  );

  const metrics =
    portalBuildAffiliateMetrics_(
      affiliate.code,
      affiliate.slug
    );

  return {
    ok: true,
    authenticated: true,

    generatedAt:
      now.toISOString(),

    affiliate: {
      name:
        affiliate.name,

      firstName:
        affiliate.firstName,

      email:
        affiliate.email,

      code:
        affiliate.code,

      slug:
        affiliate.slug,

      city:
        affiliate.city,

      status:
        affiliate.status,

      affiliatePage:
        affiliate.affiliatePage,

      checkoutUrl:
        affiliate.checkoutUrl,

      marketingKitUrl:
        affiliate.marketingKitUrl
    },

    metrics:
      metrics.summary,

    transactions:
      metrics.transactions
  };
}


/* =========================================================
   LOGOUT
   ========================================================= */

function portalLogout_(payload, e) {
  const token =
    String(
      payload.sessionToken ||
      payload.session_token ||
      ''
    ).trim();

  if (!token) {
    return {
      ok: true
    };
  }

  const session =
    portalFindActiveSession_(
      token
    );

  if (session) {
    portalUpdateSessionRow_(
      session.row,
      {
        status:
          'LOGGED_OUT',
        notes:
          'Affiliate logged out.'
      }
    );
  }

  return {
    ok: true
  };
}


/* =========================================================
   DASHBOARD METRICS
   ========================================================= */

function portalBuildAffiliateMetrics_(
  affiliateCode,
  affiliateSlug
) {
  const registry =
    portalFindAffiliateByCode_(
      affiliateCode
    );

  const pageViews =
    registry
      ? Number(
          registry.totalPageViews ||
          0
        )
      : 0;

  const checkoutClicks =
    registry
      ? Number(
          registry.totalCheckoutClicks ||
          0
        )
      : 0;

  const txSheet =
    portalGetSheet_(
      PORTAL_TRANSACTIONS_SHEET
    );

  const txLastRow =
    txSheet.getLastRow();

  let purchases = 0;
  let grossRevenue = 0;

  const transactions = [];

  if (txLastRow >= 2) {
    const rows =
      txSheet
        .getRange(
          2,
          1,
          txLastRow - 1,
          32
        )
        .getValues();

    rows.forEach(
      function(row) {
        const rowCode =
          String(
            row[11] || ''
          ).trim(); // L

        const rowSlug =
          String(
            row[12] || ''
          ).trim(); // M

        const eligible =
          String(
            row[15] || ''
          )
            .toUpperCase() ===
          'YES'; // P

        const paymentStatus =
          String(
            row[3] || ''
          )
            .toUpperCase(); // D

        if (
          rowCode !==
            affiliateCode &&
          rowSlug !==
            affiliateSlug
        ) {
          return;
        }

        if (
          paymentStatus !==
          'PAID'
        ) {
          return;
        }

        purchases += 1;

        grossRevenue +=
          Number(
            row[9] || 0
          ); // J

        transactions.push({
          receivedAt:
            portalDateIso_(
              row[0]
            ),

          transactionId:
            String(
              row[1] || ''
            ),

          product:
            String(
              row[5] || ''
            ),

          buyerName:
            String(
              row[6] || ''
            ),

          amount:
            Number(
              row[9] || 0
            ),

          commissionEligible:
            eligible,

          commissionAmount:
            Number(
              row[16] || 0
            ),

          commissionStatus:
            String(
              row[17] || ''
            ),

          attributionSource:
            String(
              row[14] || ''
            )
        });
      }
    );
  }

  transactions.sort(
    function(a, b) {
      return String(
        b.receivedAt
      ).localeCompare(
        String(
          a.receivedAt
        )
      );
    }
  );

  const commissionSummary =
    portalBuildCommissionSummary_(
      affiliateCode
    );

  const conversionRate =
    checkoutClicks > 0
      ? purchases /
        checkoutClicks
      : 0;

  const monthSales =
    portalCountCurrentMonthSales_(
      transactions
    );

  const monthlyBonus =
    portalCalculateMonthlyBonus_(
      monthSales
    );

  return {
    summary: {
      pageViews:
        pageViews,

      checkoutClicks:
        checkoutClicks,

      purchases:
        purchases,

      conversionRate:
        conversionRate,

      grossRevenue:
        grossRevenue,

      baseCommissionEarned:
        commissionSummary
          .totalEarned,

      commissionPending:
        commissionSummary
          .pending,

      commissionApproved:
        commissionSummary
          .approved,

      commissionPaid:
        commissionSummary
          .paid,

      currentMonthSales:
        monthSales,

      monthlyBonus:
        monthlyBonus
    },

    /*
     * Maksimal 50 transaksi
     * terbaru untuk dashboard.
     */
    transactions:
      transactions.slice(
        0,
        50
      )
  };
}


/* =========================================================
   COMMISSION SUMMARY
   ========================================================= */

function portalBuildCommissionSummary_(
  affiliateCode
) {
  const sheet =
    portalGetSheet_(
      PORTAL_COMMISSIONS_SHEET
    );

  const lastRow =
    sheet.getLastRow();

  let totalEarned = 0;
  let pending = 0;
  let approved = 0;
  let paid = 0;

  if (lastRow < 2) {
    return {
      totalEarned: 0,
      pending: 0,
      approved: 0,
      paid: 0
    };
  }

  const rows =
    sheet
      .getRange(
        2,
        1,
        lastRow - 1,
        16
      )
      .getValues();

  rows.forEach(
    function(row) {
      const code =
        String(
          row[3] || ''
        ).trim(); // D

      if (
        code !==
        affiliateCode
      ) {
        return;
      }

      const amount =
        Number(
          row[9] || 0
        ); // J

      const commissionStatus =
        String(
          row[10] || ''
        ).toUpperCase(); // K

      const payoutStatus =
        String(
          row[11] || ''
        ).toUpperCase(); // L

      totalEarned +=
        amount;

      if (
        commissionStatus ===
          'PENDING' ||
        payoutStatus ===
          'PENDING'
      ) {
        pending +=
          amount;
      }

      if (
        commissionStatus ===
          'APPROVED' &&
        payoutStatus !==
          'PAID'
      ) {
        approved +=
          amount;
      }

      if (
        payoutStatus ===
        'PAID'
      ) {
        paid +=
          amount;
      }
    }
  );

  return {
    totalEarned:
      totalEarned,

    pending:
      pending,

    approved:
      approved,

    paid:
      paid
  };
}


/* =========================================================
   CURRENT MONTH SALES
   ========================================================= */

function portalCountCurrentMonthSales_(
  transactions
) {
  const now =
    new Date();

  const year =
    now.getFullYear();

  const month =
    now.getMonth();

  return transactions
    .filter(
      function(tx) {
        if (
          !tx.receivedAt
        ) {
          return false;
        }

        const d =
          new Date(
            tx.receivedAt
          );

        return (
          !isNaN(
            d.getTime()
          ) &&
          d.getFullYear() ===
            year &&
          d.getMonth() ===
            month
        );
      }
    )
    .length;
}


/* =========================================================
   MONTHLY BONUS
   ========================================================= */

function portalCalculateMonthlyBonus_(
  sales
) {
  let earned = 0;
  let level =
    'Belum ada bonus';

  let nextTarget = 5;
  let nextBonus = 100000;

  if (sales >= 20) {
    earned = 600000;
    level =
      '20 Penjualan';

    nextTarget = null;
    nextBonus = null;

  } else if (
    sales >= 10
  ) {
    earned = 250000;
    level =
      '10 Penjualan';

    nextTarget = 20;
    nextBonus = 600000;

  } else if (
    sales >= 5
  ) {
    earned = 100000;
    level =
      '5 Penjualan';

    nextTarget = 10;
    nextBonus = 250000;
  }

  return {
    earned:
      earned,

    level:
      level,

    sales:
      sales,

    nextTarget:
      nextTarget,

    nextBonus:
      nextBonus,

    remainingToNext:
      nextTarget === null
        ? 0
        : Math.max(
            0,
            nextTarget -
            sales
          )
  };
}


/* =========================================================
   AFFILIATE LOOKUP BY EMAIL
   ========================================================= */

function portalFindApprovedAffiliateByEmail_(
  email
) {
  const sheet =
    portalGetSheet_(
      PORTAL_REGISTRY_SHEET
    );

  const lastRow =
    sheet.getLastRow();

  if (lastRow < 2) {
    return null;
  }

  const rows =
    sheet
      .getRange(
        2,
        1,
        lastRow - 1,
        39
      )
      .getValues();

  for (
    let i = 0;
    i < rows.length;
    i++
  ) {
    const r =
      rows[i];

    const rowEmail =
      portalNormalizeEmail_(
        r[4]
      ); // E

    const status =
      String(
        r[13] || ''
      )
        .trim()
        .toUpperCase(); // N

    const emailVerified =
      String(
        r[23] || ''
      )
        .trim()
        .toUpperCase(); // X

    if (
      rowEmail === email &&
      status ===
        'APPROVED' &&
      (
        emailVerified ===
          'YES' ||
        emailVerified ===
          'TRUE' ||
        emailVerified ===
          'VERIFIED'
      )
    ) {
      return {
        row:
          i + 2,

        name:
          String(
            r[1] || ''
          ),

        firstName:
          String(
            r[2] || ''
          ),

        email:
          rowEmail,

        city:
          String(
            r[6] || ''
          ),

        code:
          String(
            r[10] || ''
          ).trim(),

        slug:
          String(
            r[11] || ''
          ).trim(),

        affiliatePage:
          String(
            r[12] || ''
          ),

        status:
          status,

        checkoutUrl:
          String(
            r[14] || ''
          ),

        marketingKitUrl:
          String(
            r[28] || ''
          )
      };
    }
  }

  return null;
}

   AFFILIATE LOOKUP BY CODE
   ========================================================= */

function portalFindAffiliateByCode_(
  affiliateCode
) {
  const sheet =
    portalGetSheet_(
      PORTAL_REGISTRY_SHEET

/* =========================================================
   RECENT OTP LOOKUP
   ========================================================= */

function portalFindRecentOtpRequest_(
  email
) {
  const sheet =
    portalGetSheet_(
      PORTAL_SESSIONS_SHEET
    );

  const lastRow =
    sheet.getLastRow();

  if (lastRow < 2) {
    return null;
  }

  const rows =
    sheet
      .getRange(
        2,
        1,
        lastRow - 1,
        15
      )
      .getValues();

  for (
    let i =
      rows.length - 1;
    i >= 0;
    i--
  ) {
    const r =
      rows[i];

    if (
      portalNormalizeEmail_(
        r[2]
      ) !== email
    ) {
      continue;
    }

    const createdAt =
      portalAsDate_(
        r[1]
      );

    if (!createdAt) {
      return null;
    }

    return {
      row:
        i + 2,

      values:
        r,

      ageSeconds:
        (
          Date.now() -
          createdAt.getTime()
        ) /
        1000
    };
  }

  return null;
}

/* =========================================================
   ACTIVE SESSION LOOKUP
   ========================================================= */

function portalFindActiveSession_(
  rawToken
) {
  const sheet =
    portalGetSheet_(
      PORTAL_SESSIONS_SHEET
    );

  const lastRow =
    sheet.getLastRow();

  if (lastRow < 2) {
    return null;
  }

  const tokenHash =
    portalHash_(
      rawToken
    );

  const rows =
    sheet
      .getRange(
        2,
        1,
        lastRow - 1,
        15
      )
      .getValues();

  for (
    let i =
      rows.length - 1;
    i >= 0;
    i--
  ) {
    const r =
      rows[i];

    const storedHash =
      String(
        r[9] || ''
      );

    const expiresAt =
      portalAsDate_(
        r[10]
      );

    const status =
      String(
        r[12] || ''
      )
        .toUpperCase();

    if (
      storedHash &&
      portalSafeEqual_(
        tokenHash,
        storedHash
      ) &&
      status ===
        'ACTIVE'
    ) {
      /*
       * Session expired.
       */
      if (
        !expiresAt ||
        Date.now() >
          expiresAt.getTime()
      ) {
        portalUpdateSessionRow_(
          i + 2,
          {
            status:
              'EXPIRED',
            notes:
              'Session expired.'
          }
        );

        return null;
      }

      return {
        row:
          i + 2,

        values:
          r
      };
    }
  }

  return null;
}

/* =========================================================
   SECURITY SALT
   ========================================================= */

function portalEnsureSecuritySalt_() {
  const props =
    PropertiesService
      .getScriptProperties();

  let salt =
    props.getProperty(
      'PORTAL_SECURITY_SALT'
    );

  if (!salt) {
    salt =
      Utilities
        .getUuid()
        .replace(/-/g, '') +
      Utilities
        .getUuid()
        .replace(/-/g, '');

    props.setProperty(
      'PORTAL_SECURITY_SALT',
      salt
    );
  }

  return salt;
}


/* =========================================================
   HASH
   ========================================================= */

function portalHash_(
  value
) {
  const salt =
    portalEnsureSecuritySalt_();

  const bytes =
    Utilities.computeDigest(
      Utilities
        .DigestAlgorithm
        .SHA_256,

      String(value) +
        '|' +
        salt,

      Utilities
        .Charset
        .UTF_8
    );

  return bytes
    .map(
      function(b) {
        const n =
          b < 0
            ? b + 256
            : b;

        return (
          '0' +
          n.toString(16)
        ).slice(-2);
      }
    )
    .join('');
}


/* =========================================================
   CONSTANT-TIME STRING COMPARE
   ========================================================= */

function portalSafeEqual_(
  a,
  b
) {
  a =
    String(
      a || ''
    );

  b =
    String(
      b || ''
    );

  if (
    a.length !==
    b.length
  ) {
    return false;
  }

  let diff = 0;

  for (
    let i = 0;
    i < a.length;
    i++
  ) {
    diff |=
      a.charCodeAt(i) ^
      b.charCodeAt(i);
  }

  return diff === 0;
}

/* =========================================================
   DATE PARSER
   ========================================================= */

function portalAsDate_(
  value
) {
  if (!value) {
    return null;
  }

  if (
    Object.prototype
      .toString
      .call(value) ===
    '[object Date]'
  ) {
    return isNaN(
      value.getTime()
    )
      ? null
      : value;
  }

  const d =
    new Date(
      value
    );

  return isNaN(
    d.getTime()
  )
    ? null
    : d;
}


/* =========================================================
   DATE TO ISO
   ========================================================= */

function portalDateIso_(
  value
) {
  const d =
    portalAsDate_(
      value
    );

  return d
    ? d.toISOString()
    : '';
}


/* =========================================================
   HTML ESCAPE
   ========================================================= */

function portalEscapeHtml_(
  value
) {
  return String(
    value || ''
  )
    .replace(
      /&/g,
      '&amp;'
    )
    .replace(
      /</g,
      '&lt;'
    )
    .replace(
      />/g,
      '&gt;'
    )
    .replace(
      /"/g,
      '&quot;'
    )
    .replace(
      /'/g,
      '&#039;'
    );
}

/* =========================================================
   FULL PRODUCTION doPost()
   STEP 10B FIX — PORTAL + EVENT + LYNK + REGISTRATION
   ========================================================= */

function doPost(e) {
  try {
    /*
     * 1. BACA SEMUA JENIS PAYLOAD
     */
    const p =
      (e && e.parameter)
        ? e.parameter
        : {};

    const rawPostBody =
      e &&
      e.postData &&
      e.postData.contents
        ? e.postData.contents
        : '';

    const contentType =
      e &&
      e.postData &&
      e.postData.type
        ? String(e.postData.type)
        : '';

    let rawJson = {};
    let rawForm = {};

    /*
     * Coba JSON.
     */
    if (rawPostBody) {
      try {
        rawJson =
          JSON.parse(
            rawPostBody
          ) || {};
      } catch (jsonError) {
        rawJson = {};
      }
    }

    /*
     * Fallback parser application/x-www-form-urlencoded.
     */
    if (
      rawPostBody &&
      (
        contentType
          .toLowerCase()
          .indexOf(
            'application/x-www-form-urlencoded'
          ) >= 0 ||
        rawPostBody
          .indexOf('=') >= 0
      )
    ) {
      try {
        rawPostBody
          .split('&')
          .forEach(function(pair) {
            if (!pair) return;

            const parts =
              pair.split('=');

            const rawKey =
              parts.shift() || '';

            const rawValue =
              parts.join('=');

            const key =
              decodeURIComponent(
                rawKey.replace(
                  /\+/g,
                  ' '
                )
              );

            const value =
              decodeURIComponent(
                String(
                  rawValue || ''
                ).replace(
                  /\+/g,
                  ' '
                )
              );

            if (key) {
              rawForm[key] =
                value;
            }
          });
      } catch (formError) {
        rawForm = {};
      }
    }

    /*
     * Prioritas:
     * e.parameter > parsed form > JSON.
     */
    const payload =
      Object.assign(
        {},
        rawJson,
        rawForm,
        p
      );

    const action =
      clean_(
        payload.action
      ).toLowerCase();


    /*
     * 2. AFFILIATE PORTAL
     * WAJIB diproses sebelum form pendaftaran.
     */
    if (
      action ===
        'portal_request_otp' ||
      action ===
        'portal_verify_otp' ||
      action ===
        'portal_dashboard' ||
      action ===
        'portal_logout'
    ) {
      return json_(
        portalHandleAction_(
          action,
          payload,
          e
        )
      );
    }


    /*
     * 3. AFFILIATE EVENT TRACKING
     */
    if (
      action ===
      'affiliate_event'
    ) {
      return handleAffiliateEvent_(
        payload
      );
    }


    /*
     * 4. LYNK PRODUCTION WEBHOOK
     */
    const jsonWebhookEvent =
      clean_(
        rawJson.event ||
        rawJson.type ||
        rawJson.eventType ||
        rawJson.event_type ||
        (
          rawJson.message &&
          rawJson.message.action
        ) ||
        (
          rawJson.data &&
          rawJson.data.event
        ) ||
        (
          rawJson.data &&
          rawJson.data.message &&
          rawJson.data.message.action
        )
      ).toLowerCase();

    const formWebhookEvent =
      clean_(
        payload.event ||
        payload.type ||
        payload.eventType ||
        payload.event_type
      ).toLowerCase();

    const webhookEvent =
      jsonWebhookEvent ||
      formWebhookEvent;

    if (
      action ===
        'payment_webhook' ||
      webhookEvent ===
        'payment.received' ||
      webhookEvent ===
        'payment_received'
    ) {
      return handleStep9CProductionWebhook_(
        e,
        payload
      );
    }


    /*
     * 5. FORM PENDAFTARAN AFFILIATE
     */
    const name =
      clean_(
        payload.name
      );

    const whatsapp =
      normalizePhone_(
        payload.whatsapp
      );

    const email =
      clean_(
        payload.email
      ).toLowerCase();

    const social =
      clean_(
        payload.social
      );

    const city =
      clean_(
        payload.city
      );

    const profession =
      clean_(
        payload.profession
      );

    const reason =
      clean_(
        payload.reason
      );

    const agree =
      clean_(
        payload.agree
      ) === 'YA'
        ? 'YA'
        : 'TIDAK';

    const sourcePage =
      clean_(
        payload.sourcePage
      );

    const userAgent =
      clean_(
        payload.userAgent
      );


    if (
      !name ||
      !whatsapp ||
      !email ||
      !city ||
      !profession ||
      !reason ||
      agree !== 'YA'
    ) {
      return json_({
        ok: false,
        message:
          'Data wajib belum lengkap.'
      });
    }


    if (
      !isValidEmail_(
        email
      )
    ) {
      return json_({
        ok: false,
        message:
          'Alamat email tidak valid.'
      });
    }


    /*
     * 6. BUAT KODE REFERRAL
     */
    const firstName =
      getFirstName_(
        name
      );

    const referralCode =
      createReferralCode_(
        firstName,
        whatsapp
      );

    const ss =
      SpreadsheetApp.openById(
        SPREADSHEET_ID
      );

    const sheet =
      ss.getSheetByName(
        SHEET_NAME
      );

    if (!sheet) {
      throw new Error(
        'Sheet tidak ditemukan: ' +
        SHEET_NAME
      );
    }

    ensureStep6AHeaders_(
      sheet
    );

    const lastRow =
      sheet.getLastRow();

    const existing =
      lastRow > 1
        ? sheet
            .getRange(
              2,
              1,
              lastRow - 1,
              29
            )
            .getDisplayValues()
        : [];

    const codeExists =
      existing.some(
        function(r) {
          return (
            clean_(
              r[10]
            ).toUpperCase() ===
            referralCode
          );
        }
      );

    const phoneExists =
      existing.some(
        function(r) {
          return (
            normalizePhone_(
              r[3]
            ) ===
            whatsapp
          );
        }
      );

    const emailExists =
      existing.some(
        function(r) {
          return (
            clean_(
              r[4]
            ).toLowerCase() ===
            email
          );
        }
      );


    /*
     * 7. BUAT SLUG
     */
    const slugBase =
      slugify_(
        firstName
      );

    let slug =
      slugBase;

    const usedSlugs =
      new Set(
        existing
          .map(
            function(r) {
              return clean_(
                r[11]
              ).toLowerCase();
            }
          )
          .filter(
            Boolean
          )
      );

    if (
      usedSlugs.has(
        slug
      )
    ) {
      slug =
        slugBase +
        '-' +
        whatsapp.slice(
          -3
        );
    }

    let suffix = 2;

    while (
      usedSlugs.has(
        slug
      )
    ) {
      slug =
        slugBase +
        '-' +
        whatsapp.slice(
          -3
        ) +
        '-' +
        suffix;

      suffix++;
    }


    /*
     * 8. DUPLICATE CHECK
     */
    const duplicate =
      codeExists ||
      phoneExists ||
      emailExists;

    const duplicateNotes =
      [
        codeExists
          ? 'KODE'
          : '',
        phoneExists
          ? 'WHATSAPP'
          : '',
        emailExists
          ? 'EMAIL'
          : ''
      ]
        .filter(
          Boolean
        )
        .join(
          ', '
        );

    const status =
      duplicate
        ? 'REVIEW_DUPLICATE'
        : 'PENDING_EMAIL';

    const affiliatePage =
      SITE_URL +
      '/?affiliate=' +
      encodeURIComponent(
        slug
      );

    const verificationToken =
      duplicate
        ? ''
        : createVerificationToken_();


    /*
     * 9. SIMPAN KE AFFILIATE REGISTRY
     */
    sheet.appendRow([
      new Date(),                         // A
      name,                               // B
      firstName,                          // C
      whatsapp,                           // D
      email,                              // E
      social,                             // F
      city,                               // G
      profession,                         // H
      reason,                             // I
      agree,                              // J
      referralCode,                       // K
      slug,                               // L
      affiliatePage,                      // M
      status,                             // N
      '',                                 // O
      '',                                 // P
      duplicate
        ? 'Periksa duplikat: ' +
          duplicateNotes
        : '',                             // Q
      sourcePage,                         // R
      userAgent,                          // S
      duplicate
        ? 'DUPLICATE: ' +
          duplicateNotes
        : 'OK',                           // T
      '',                                 // U
      '',                                 // V
      '',                                 // W
      'NO',                               // X
      verificationToken,                  // Y
      '',                                 // Z
      '',                                 // AA
      '',                                 // AB
      SITE_URL +
        '/?affiliate=' +
        encodeURIComponent(
          slug
        ) +
        '&kit=1'                          // AC
    ]);

    const row =
      sheet.getLastRow();


    /*
     * 10. EMAIL VERIFICATION
     */
    if (!duplicate) {
      try {
        sendVerificationEmailForRow_(
          sheet,
          row
        );
      } catch (err) {
        sheet
          .getRange(
            row,
            14
          )
          .setValue(
            'EMAIL_ERROR'
          );

        sheet
          .getRange(
            row,
            28
          )
          .setValue(
            String(
              err &&
              err.message
                ? err.message
                : err
            )
          );
      }
    }


    return json_({
      ok: true,
      status:
        status,
      referralCode:
        referralCode,
      slug:
        slug,
      affiliatePage:
        affiliatePage,
      duplicate:
        duplicate,
      verificationRequired:
        !duplicate
    });


  } catch (err) {
    console.error(
      err
    );

    return json_({
      ok: false,
      message:
        String(
          err &&
          err.message
            ? err.message
            : err
        )
    });
  }
}
