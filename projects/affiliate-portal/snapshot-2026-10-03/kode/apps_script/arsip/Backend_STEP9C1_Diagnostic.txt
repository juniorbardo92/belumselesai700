function doGet(e) {
  try {
    const p = (e && e.parameter) || {};
    const action = clean_(p.action || 'health').toLowerCase();
    const callback = clean_(p.callback);

    if (action === 'verify') {
      return verifyAffiliateEmail_(clean_(p.token));
    }

    if (action === 'registry') {
      return jsonOrJsonp_(getApprovedAffiliateRegistry_(), callback);
    }

    if (action === 'webhook_health') {
      const suppliedKey = clean_(p.key);
      const expectedKey = getStep9CWebhookSecret_();

      if (!expectedKey || suppliedKey !== expectedKey) {
        return jsonOrJsonp_({
          ok: false,
          service: 'Panglima Lynk Webhook',
          error: 'UNAUTHORIZED',
        }, callback);
      }

      return jsonOrJsonp_({
        ok: true,
        service: 'Panglima Lynk Webhook',
        version: 'step9c1-diagnostic-fix',
        status: 'READY',
        event: 'payment.received',
      }, callback);
    }

    return jsonOrJsonp_({
      ok: true,
      service: 'Panglima Affiliate Registry',
      version: 'step9c1-diagnostic-fix',
    }, callback);
  } catch (err) {
    return jsonOrJsonp_({
      ok: false,
      message: String(err && err.message ? err.message : err),
    }, clean_((e && e.parameter && e.parameter.callback) || ''));
  }
}

function doPost(e) {
  try {
    const p = (e && e.parameter) || {};
    const action = clean_(p.action).toLowerCase();

    // STEP 8 — Affiliate page-view / checkout-click tracking.
    // Diproses sebelum validasi form pendaftaran.
    if (action === 'affiliate_event') {
      return handleAffiliateEvent_(p);
    }

    // STEP 9C FIX — Lynk production webhook via /exec polos.
    // Lynk mengirim JSON POST, sehingga event harus dibaca juga dari postData.contents.
    const rawPostBody =
      e && e.postData && e.postData.contents
        ? e.postData.contents
        : '';

    let rawPostJson = {};
    try {
      rawPostJson = rawPostBody ? JSON.parse(rawPostBody) : {};
    } catch (jsonError) {
      rawPostJson = {};
    }

    const jsonWebhookEvent = clean_(
      rawPostJson.event ||
      rawPostJson.type ||
      rawPostJson.eventType ||
      rawPostJson.event_type ||
      (rawPostJson.message && rawPostJson.message.action) ||
      (rawPostJson.data && rawPostJson.data.event) ||
      (rawPostJson.data &&
        rawPostJson.data.message &&
        rawPostJson.data.message.action)
    ).toLowerCase();

    const formWebhookEvent = clean_(
      p.event || p.type || p.eventType || p.event_type
    ).toLowerCase();

    const webhookEvent = jsonWebhookEvent || formWebhookEvent;

    if (
      action === 'payment_webhook' ||
      webhookEvent === 'payment.received' ||
      webhookEvent === 'payment_received'
    ) {
      return handleStep9CProductionWebhook_(e, p);
    }

    const name = clean_(p.name);
    const whatsapp = normalizePhone_(p.whatsapp);
    const email = clean_(p.email).toLowerCase();
    const social = clean_(p.social);
    const city = clean_(p.city);
    const profession = clean_(p.profession);
    const reason = clean_(p.reason);
    const agree = clean_(p.agree) === 'YA' ? 'YA' : 'TIDAK';
    const sourcePage = clean_(p.sourcePage);
    const userAgent = clean_(p.userAgent);

    if (!name || !whatsapp || !email || !city || !profession || !reason || agree !== 'YA') {
      return json_({ ok: false, message: 'Data wajib belum lengkap.' });
    }

    if (!isValidEmail_(email)) {
      return json_({ ok: false, message: 'Alamat email tidak valid.' });
    }

    const firstName = getFirstName_(name);
    const referralCode = createReferralCode_(firstName, whatsapp);

    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName(SHEET_NAME);
    if (!sheet) throw new Error('Sheet tidak ditemukan: ' + SHEET_NAME);

    ensureStep6AHeaders_(sheet);

    const lastRow = sheet.getLastRow();
    const existing = lastRow > 1
      ? sheet.getRange(2, 1, lastRow - 1, 29).getDisplayValues()
      : [];

    const codeExists = existing.some(r => clean_(r[10]).toUpperCase() === referralCode);
    const phoneExists = existing.some(r => normalizePhone_(r[3]) === whatsapp);
    const emailExists = existing.some(r => clean_(r[4]).toLowerCase() === email);

    let slugBase = slugify_(firstName);
    let slug = slugBase;
    const usedSlugs = new Set(existing.map(r => clean_(r[11]).toLowerCase()).filter(Boolean));
    if (usedSlugs.has(slug)) slug = `${slugBase}-${whatsapp.slice(-3)}`;
    let suffix = 2;
    while (usedSlugs.has(slug)) {
      slug = `${slugBase}-${whatsapp.slice(-3)}-${suffix}`;
      suffix++;
    }

    const duplicate = codeExists || phoneExists || emailExists;
    const duplicateNotes = [
      codeExists ? 'KODE' : '',
      phoneExists ? 'WHATSAPP' : '',
      emailExists ? 'EMAIL' : '',
    ].filter(Boolean).join(', ');

    const status = duplicate ? 'REVIEW_DUPLICATE' : 'PENDING_EMAIL';
    const affiliatePage = `${SITE_URL}/?affiliate=${encodeURIComponent(slug)}`;
    const verificationToken = duplicate ? '' : createVerificationToken_();

    sheet.appendRow([
      new Date(),                  // A Timestamp
      name,                        // B Nama Lengkap
      firstName,                   // C Nama Depan
      whatsapp,                    // D WhatsApp
      email,                       // E Email
      social,                      // F Social
      city,                        // G Kota
      profession,                  // H Profesi
      reason,                      // I Alasan
      agree,                       // J Setuju
      referralCode,                // K Kode
      slug,                        // L Slug
      affiliatePage,               // M Halaman
      status,                      // N Status
      '',                          // O Link Lynk
      '',                          // P Tanggal Approval
      duplicate ? `Periksa duplikat: ${duplicateNotes}` : '', // Q Catatan
      sourcePage,                  // R Source
      userAgent,                   // S UA
      duplicate ? `DUPLICATE: ${duplicateNotes}` : 'OK', // T Duplicate
      '',                          // U Approval Email Status
      '',                          // V Approval Email Sent At
      '',                          // W Approval Email Error
      duplicate ? 'NO' : 'NO',     // X Email Verified
      verificationToken,           // Y Token
      '',                          // Z Verified At
      '',                          // AA Verification Email Sent At
      '',                          // AB Verification Email Error
      `${SITE_URL}/?affiliate=${encodeURIComponent(slug)}&kit=1`, // AC Marketing Kit URL
    ]);

    const row = sheet.getLastRow();

    if (!duplicate) {
      try {
        sendVerificationEmailForRow_(sheet, row);
      } catch (err) {
        sheet.getRange(row, 14).setValue('EMAIL_ERROR');
        sheet.getRange(row, 28).setValue(String(err && err.message ? err.message : err));
      }
    }

    return json_({
      ok: true,
      status,
      referralCode,
      slug,
      affiliatePage,
      duplicate,
      verificationRequired: !duplicate,
    });
  } catch (err) {
    console.error(err);
    return json_({
      ok: false,
      message: String(err && err.message ? err.message : err),
    });
  }
}

function setupStep6C() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) throw new Error('Sheet tidak ditemukan: ' + SHEET_NAME);

  ensureStep6AHeaders_(sheet);

  // Migrate existing APPROVED affiliates so active pages do not disappear.
  const lastRow = sheet.getLastRow();
  if (lastRow > 1) {
    const rows = sheet.getRange(2, 1, lastRow - 1, 29).getDisplayValues();
    rows.forEach((r, i) => {
      const row = i + 2;
      const status = clean_(r[13]).toUpperCase();
      const verified = clean_(r[23]).toUpperCase();
      if (status === 'APPROVED' && verified !== 'YES') {
        sheet.getRange(row, 24).setValue('YES');
        const approvalDate = sheet.getRange(row, 16).getValue();
        sheet.getRange(row, 26).setValue(approvalDate || new Date());
      }
    });
  }

  // Remove old approval triggers to prevent duplicate welcome emails.
  ScriptApp.getProjectTriggers().forEach(trigger => {
    if (trigger.getHandlerFunction() === 'handleAffiliateApproval') {
      ScriptApp.deleteTrigger(trigger);
    }
  });

  ScriptApp.newTrigger('handleAffiliateApproval')
    .forSpreadsheet(SPREADSHEET_ID)
    .onEdit()
    .create();

  const quota = MailApp.getRemainingDailyQuota();
  Logger.log(`Step 6C aktif. Sisa kuota email hari ini: ${quota}`);
  return `Step 6C aktif. Sisa kuota email hari ini: ${quota}`;
}


/* =========================================================
   STEP 8 — AFFILIATE EVENT TRACKING
   Event types:
   PAGE_VIEW
   CHECKOUT_CLICK

   Data disimpan ke tab "Affiliate Events".
   Ringkasan otomatis disimpan di Affiliate Registry:
   AD = Total Page Views
   AE = Total Checkout Clicks
   AF = Last Affiliate Activity
   ========================================================= */

function handleAffiliateEvent_(p) {
  const eventType = clean_(p.eventType).toUpperCase();
  if (!['PAGE_VIEW', 'CHECKOUT_CLICK'].includes(eventType)) {
    return json_({ ok: false, message: 'Event affiliate tidak valid.' });
  }

  const referralCode = clean_(p.referralCode).toUpperCase();
  const slug = clean_(p.slug).toLowerCase();

  if (!referralCode && !slug) {
    return json_({ ok: false, message: 'Kode referral / slug kosong.' });
  }

  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const eventsSheet = ensureAffiliateEventsSheet_(ss);
  const registrySheet = ss.getSheetByName(SHEET_NAME);
  if (!registrySheet) throw new Error('Sheet tidak ditemukan: ' + SHEET_NAME);

  const now = new Date();
  const eventId = Utilities.getUuid();

  eventsSheet.appendRow([
    now,
    eventId,
    eventType,
    referralCode,
    slug,
    clean_(p.partnerName),
    clean_(p.visitorId),
    clean_(p.pageUrl),
    clean_(p.destinationUrl),
    clean_(p.referrer),
    clean_(p.utmSource),
    clean_(p.utmMedium),
    clean_(p.utmCampaign),
    clean_(p.utmContent),
    clean_(p.utmTerm),
    clean_(p.userAgent),
  ]);

  updateAffiliateActivitySummary_(
    registrySheet,
    referralCode,
    slug,
    eventType,
    now
  );

  return json_({
    ok: true,
    eventId,
    eventType,
  });
}

function updateAffiliateActivitySummary_(sheet, referralCode, slug, eventType, now) {
  ensureStep8RegistryColumns_(sheet);

  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return;

  const rows = sheet.getRange(2, 1, lastRow - 1, 32).getDisplayValues();
  const targetIndex = rows.findIndex(r => {
    const rowCode = clean_(r[10]).toUpperCase();
    const rowSlug = clean_(r[11]).toLowerCase();

    if (referralCode && rowCode === referralCode) return true;
    if (slug && rowSlug === slug) return true;
    return false;
  });

  if (targetIndex < 0) return;

  const row = targetIndex + 2;

  if (eventType === 'PAGE_VIEW') {
    const current = Number(sheet.getRange(row, 30).getValue()) || 0;
    sheet.getRange(row, 30).setValue(current + 1);
  }

  if (eventType === 'CHECKOUT_CLICK') {
    const current = Number(sheet.getRange(row, 31).getValue()) || 0;
    sheet.getRange(row, 31).setValue(current + 1);
  }

  sheet.getRange(row, 32).setValue(now);
}

function parseWebhookPayload_(e, p) {
  let json = {};
  const raw = (e && e.postData && e.postData.contents) || '';

  try {
    if (raw) json = JSON.parse(raw);
  } catch (err) {
    json = {};
  }

  return {
    raw: raw || JSON.stringify(p || {}),
    root: json && typeof json === 'object' ? json : {},
    form: p || {},
  };
}

function pickWebhookValue_(payload, candidates) {
  for (const key of candidates) {
    const parts = key.split('.');
    let v = payload.root;
    let ok = true;

    for (const part of parts) {
      if (v && Object.prototype.hasOwnProperty.call(v, part)) {
        v = v[part];
      } else {
        ok = false;
        break;
      }
    }

    if (ok && v !== undefined && v !== null && String(v).trim() !== '') {
      return v;
    }

    if (
      Object.prototype.hasOwnProperty.call(payload.form, key) &&
      clean_(payload.form[key])
    ) {
      return payload.form[key];
    }
  }

  return '';
}

function numberWebhookValue_(value) {
  if (typeof value === 'number') return value;
  const digits = String(value || '').replace(/[^\d.-]/g, '');
  return Number(digits) || 0;
}

function normalizePaymentStatus_(value, eventType) {
  const s = clean_(value).toLowerCase();
  const e = clean_(eventType).toLowerCase();

  if (
    e === 'payment.received' ||
    e === 'payment_received' ||
    ['paid', 'success', 'successful', 'completed', 'received'].includes(s)
  ) {
    return 'PAID';
  }

  if (['refunded', 'refund'].includes(s)) return 'REFUNDED';
  if (['cancelled', 'canceled', 'failed', 'expired'].includes(s)) return s.toUpperCase();

  return s ? s.toUpperCase() : 'UNKNOWN';
}

function findAffiliateForPayment_(registrySheet, affiliateCode, affiliateSlug, checkoutUrl) {
  const lastRow = registrySheet.getLastRow();
  if (lastRow < 2) return null;

  const rows = registrySheet.getRange(2, 1, lastRow - 1, Math.max(39, registrySheet.getLastColumn())).getDisplayValues();

  const code = clean_(affiliateCode).toUpperCase();
  const slug = clean_(affiliateSlug).toLowerCase();
  const checkout = clean_(checkoutUrl);

  for (let i = 0; i < rows.length; i++) {
    const r = rows[i];
    const rowCode = clean_(r[10]).toUpperCase();
    const rowSlug = clean_(r[11]).toLowerCase();
    const rowCheckout = clean_(r[14]);
    const rowStatus = clean_(r[13]).toUpperCase();
    const duplicateFlag = clean_(r[19]).toUpperCase();
    const emailVerified = clean_(r[23]).toUpperCase();

    if (rowStatus !== 'APPROVED' || duplicateFlag !== 'OK' || emailVerified !== 'YES') {
      continue;
    }

    let source = '';
    if (code && rowCode === code) source = 'AFFILIATE_CODE';
    else if (slug && rowSlug === slug) source = 'AFFILIATE_SLUG';
    else if (checkout && rowCheckout && rowCheckout === checkout) source = 'CHECKOUT_URL';

    if (source) {
      return {
        row: i + 2,
        code: rowCode,
        slug: rowSlug,
        name: clean_(r[1]),
        checkoutUrl: rowCheckout,
        source,
      };
    }
  }

  return null;
}

function pickWebhookObject_(payload, candidates) {
  for (const key of candidates) {
    const parts = key.split('.');
    let v = payload.root;
    let ok = true;

    for (const part of parts) {
      if (v && Object.prototype.hasOwnProperty.call(v, part)) {
        v = v[part];
      } else {
        ok = false;
        break;
      }
    }

    if (ok && v && typeof v === 'object') return v;
  }

  return null;
}

function pickWebhookArray_(payload, candidates) {
  for (const key of candidates) {
    const parts = key.split('.');
    let v = payload.root;
    let ok = true;

    for (const part of parts) {
      if (v && Object.prototype.hasOwnProperty.call(v, part)) {
        v = v[part];
      } else {
        ok = false;
        break;
      }
    }

    if (ok && Array.isArray(v)) return v;
  }

  return [];
}

function extractLynkItems_(payload) {
  const items = pickWebhookArray_(payload, [
    'items',
    'data.items',
    'order.items',
    'data.order.items',
  ]);

  return items.map(item => ({
    id: clean_(item && (item.id || item.productId || item.product_id)),
    name: clean_(item && (item.name || item.productName || item.product_name || item.title)),
    price: numberWebhookValue_(item && (item.price || item.amount || item.total)),
    qty: Number(item && (item.qty || item.quantity || 1)) || 1,
    raw: item || {},
  }));
}

function summarizeLynkItems_(items) {
  if (!items || !items.length) return '';
  return items
    .map(item => `${item.name || item.id || 'Item'} x${item.qty}`)
    .join(' | ');
}

        source: 'MANUAL_REVIEW',
      };
    }
  }

  return null;
}

function commissionExistsForTransaction_(commissionSheet, transactionId) {
  const id = clean_(transactionId);
  if (!id || commissionSheet.getLastRow() < 2) return false;

  const values = commissionSheet
    .getRange(2, 2, commissionSheet.getLastRow() - 1, 1)
    .getDisplayValues();

  return values.some(r => clean_(r[0]) === id);
}

function createCommissionForReviewedTransaction_(txSheet, txRow, registrySheet, commissionSheet, affiliate) {
  const transactionId = clean_(txSheet.getRange(txRow, 2).getDisplayValue());
  if (!transactionId) throw new Error('Transaction ID kosong.');

  if (commissionExistsForTransaction_(commissionSheet, transactionId)) {
    txSheet.getRange(txRow, 31).setValue('ALREADY_COMMISSIONED');
    txSheet.getRange(txRow, 32).setValue(new Date());
    return 'Transaksi sudah memiliki komisi.';
  }

  const paymentStatus = clean_(txSheet.getRange(txRow, 4).getDisplayValue()).toUpperCase();
  const productName = clean_(txSheet.getRange(txRow, 6).getDisplayValue());
  const amount = Number(txSheet.getRange(txRow, 10).getValue()) || 0;

  const productLooksPanglima =
    /panglima/i.test(productName) ||
    amount === PANGLIMA_PRODUCT_PRICE;

  if (paymentStatus !== 'PAID') {
    throw new Error('Transaksi belum berstatus PAID.');
  }

  if (!productLooksPanglima) {
    throw new Error('Transaksi tidak terdeteksi sebagai Paket Panglima.');
  }

  const now = new Date();
  const commissionId = Utilities.getUuid();

  txSheet.getRange(txRow, 12).setValue(affiliate.code);
  txSheet.getRange(txRow, 13).setValue(affiliate.slug);
  txSheet.getRange(txRow, 14).setValue('ATTRIBUTED');
  txSheet.getRange(txRow, 15).setValue('MANUAL_REVIEW');
  txSheet.getRange(txRow, 16).setValue('YES');
  txSheet.getRange(txRow, 17).setValue(PANGLIMA_COMMISSION_AMOUNT);
  txSheet.getRange(txRow, 18).setValue('PENDING');
  txSheet.getRange(txRow, 31).setValue('REVIEWED');
  txSheet.getRange(txRow, 32).setValue(now);

  commissionSheet.appendRow([
    commissionId,
    transactionId,
    now,
    affiliate.code,
    affiliate.slug,
    affiliate.name,
    productName || 'Paket Panglima UMKM',
    amount,
    PANGLIMA_COMMISSION_RATE,
    PANGLIMA_COMMISSION_AMOUNT,
    'PENDING',
    'UNPAID',
    '',
    '',
    '',
    'Atribusi disetujui melalui review manual Step 9B.',
  ]);

  updateRegistryPurchaseSummary_(
    registrySheet,
    affiliate.row,
    amount,
    PANGLIMA_COMMISSION_AMOUNT,
    now
  );

  return `Komisi ${affiliate.code} berhasil dibuat untuk transaksi ${transactionId}.`;
}

function handleStep9BManualReviewEdit(e) {
  if (!e || !e.range) return;

  const sheet = e.range.getSheet();
  if (sheet.getName() !== AFFILIATE_TRANSACTIONS_SHEET) return;
  if (e.range.getRow() < 2) return;
  if (e.range.getColumn() !== 30) return; // AD = Manual Affiliate Code

  const code = clean_(e.value).toUpperCase();
  if (!code) return;

  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const registry = ss.getSheetByName(SHEET_NAME);
  const commissionSheet = ss.getSheetByName(AFFILIATE_COMMISSIONS_SHEET);

  if (!registry || !commissionSheet) {
    throw new Error('Registry / Commission sheet tidak ditemukan.');
  }

  const affiliate = findAffiliateByCode_(registry, code);

  if (!affiliate) {
    sheet.getRange(e.range.getRow(), 31).setValue('INVALID_AFFILIATE');
    return;
  }

  try {
    const result = createCommissionForReviewedTransaction_(
      sheet,
      e.range.getRow(),
      registry,
      commissionSheet,
      affiliate
    );
    sheet.getRange(e.range.getRow(), 20).setValue(result);
  } catch (err) {
    sheet.getRange(e.range.getRow(), 31).setValue('ERROR');
    sheet.getRange(e.range.getRow(), 20).setValue(String(err && err.message ? err.message : err));
  }
}


function handlePaymentWebhook_(e, p) {
  const payload = parseWebhookPayload_(e, p);

  const eventType = clean_(pickWebhookValue_(payload, [
    'event','type','eventType','event_type','name'
  ])) || 'payment.received';

  const refId = clean_(pickWebhookValue_(payload, [
    'refId','ref_id','data.refId','data.ref_id',
    'transaction.refId','transaction.ref_id'
  ]));

  const transactionId = clean_(pickWebhookValue_(payload, [
    'transaction_id','transactionId','payment_id','paymentId',
    'refId','ref_id','id',
    'data.transaction_id','data.transactionId','data.payment_id','data.paymentId',
    'data.refId','data.ref_id','data.id'
  ])) || refId;

  const statusRaw = pickWebhookValue_(payload, [
    'status','payment_status','paymentStatus',
    'data.status','data.payment_status','data.paymentStatus'
  ]);
  const paymentStatus = normalizePaymentStatus_(statusRaw, eventType);

  const createdAt = clean_(pickWebhookValue_(payload, [
    'createdAt','created_at','data.createdAt','data.created_at'
  ]));

  const messageAction = clean_(pickWebhookValue_(payload, [
    'message.action','data.message.action','action'
  ]));

  const messageCode = clean_(pickWebhookValue_(payload, [
    'message.code','data.message.code','code'
  ]));

  const customerObj = pickWebhookObject_(payload, [
    'customer','data.customer','buyer','data.buyer'
  ]) || {};

  const buyerName = clean_(
    customerObj.name ||
    customerObj.fullName ||
    customerObj.full_name ||
    pickWebhookValue_(payload, [
      'buyer_name','buyerName','customer_name','customerName',
      'data.buyer_name','data.customer_name'
    ])
  );

  const buyerEmail = clean_(
    customerObj.email ||
    pickWebhookValue_(payload, [
      'buyer_email','buyerEmail','customer_email','customerEmail',
      'data.buyer_email','data.customer_email'
    ])
  );

  const buyerPhone = clean_(
    customerObj.phone ||
    customerObj.whatsapp ||
    pickWebhookValue_(payload, [
      'buyer_phone','buyerPhone','customer_phone','customerPhone',
      'data.buyer_phone','data.customer_phone'
    ])
  );

  const totalsObj = pickWebhookObject_(payload, [
    'totals','data.totals','order.totals','data.order.totals'
  ]) || {};

  const grandTotal = numberWebhookValue_(
    totalsObj.grandTotal ||
    totalsObj.grand_total ||
    pickWebhookValue_(payload, [
      'grandTotal','grand_total','amount','total','gross_amount','grossAmount',
      'data.grandTotal','data.grand_total','data.amount','data.total'
    ])
  );

  const totalPrice = numberWebhookValue_(
    totalsObj.totalPrice ||
    totalsObj.total_price ||
    pickWebhookValue_(payload, ['totalPrice','total_price','data.totalPrice','data.total_price'])
  );

  const totalAddon = numberWebhookValue_(
    totalsObj.totalAddon ||
    totalsObj.total_addon ||
    pickWebhookValue_(payload, ['totalAddon','total_addon','data.totalAddon','data.total_addon'])
  );

  const discount = numberWebhookValue_(
    totalsObj.discount ||
    totalsObj.totalDiscount ||
    totalsObj.total_discount ||
    pickWebhookValue_(payload, ['discount','data.discount'])
  );

  const shipping = numberWebhookValue_(
    totalsObj.shipping ||
    totalsObj.shippingFee ||
    totalsObj.shipping_fee ||
    pickWebhookValue_(payload, ['shipping','shippingFee','shipping_fee','data.shipping'])
  );

  const amount = grandTotal;
  const currency = clean_(pickWebhookValue_(payload, [
    'currency','data.currency','totals.currency','data.totals.currency'
  ])) || 'IDR';

  const items = extractLynkItems_(payload);
  const firstItem = items[0] || {};
  const productId = clean_(
    firstItem.id ||
    pickWebhookValue_(payload, [
      'product_id','productId','data.product_id','data.productId','product.id'
    ])
  );
  const productName = clean_(
    firstItem.name ||
    pickWebhookValue_(payload, [
      'product_name','productName','data.product_name','data.productName','product.name'
    ])
  );

  const affiliateCode = clean_(pickWebhookValue_(payload, [
    'affiliate_code','affiliateCode','referral_code','referralCode',
    'metadata.affiliate_code','metadata.referral_code',
    'data.affiliate_code','data.referral_code'
  ])).toUpperCase();

  const affiliateSlug = clean_(pickWebhookValue_(payload, [
    'affiliate_slug','affiliateSlug','metadata.affiliate_slug','data.affiliate_slug'
  ])).toLowerCase();

  const checkoutUrl = clean_(pickWebhookValue_(payload, [
    'checkout_url','checkoutUrl','product_url','productUrl','url',
    'data.checkout_url','data.product_url'
  ]));

  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const registry = ss.getSheetByName(SHEET_NAME);
  if (!registry) throw new Error('Sheet tidak ditemukan: ' + SHEET_NAME);

  const sheets = ensureStep9Sheets_(ss);
  ensureStep9RegistryColumns_(registry);
  ensureStep9BTransactionColumns_(sheets.tx);

  logRawLynkWebhook_(ss, {
    receivedAt: new Date(),
    eventType,
    refId,
    messageAction,
    messageCode,
    customerLabel: [buyerName, buyerEmail, buyerPhone].filter(Boolean).join(' | '),
    grandTotal,
    itemsSummary: summarizeLynkItems_(items),
    parseResult: transactionId ? 'PARSED' : 'NO_TRANSACTION_ID',
    rawPayload: payload.raw,
  });

  if (transactionAlreadyExists_(sheets.tx, transactionId)) {
    return json_({
      ok: true,
      duplicate: true,
      transactionId,
      message: 'Transaksi sudah pernah diproses.',
    });
  }

  const affiliate = findAffiliateForPayment_(
    registry,
    affiliateCode,
    affiliateSlug,
    checkoutUrl
  );

  const attributionStatus = affiliate ? 'ATTRIBUTED' : 'REVIEW';
  const attributionSource = affiliate ? affiliate.source : 'UNMATCHED';

  const productLooksPanglima =
    /panglima/i.test(productName) ||
    amount === PANGLIMA_PRODUCT_PRICE;

  const eligible =
    Boolean(affiliate) &&
    paymentStatus === 'PAID' &&
    productLooksPanglima;

  const commissionAmount = eligible ? PANGLIMA_COMMISSION_AMOUNT : 0;
  const commissionStatus = eligible ? 'PENDING' : 'NOT_ELIGIBLE';
  const now = new Date();

  sheets.tx.appendRow([
    now,
    transactionId || Utilities.getUuid(),
    eventType,
    paymentStatus,
    productId,
    productName,
    buyerName,
    buyerEmail,
    buyerPhone,
    amount,
    currency,
    affiliate ? affiliate.code : affiliateCode,
    affiliate ? affiliate.slug : affiliateSlug,
    attributionStatus,
    attributionSource,
    eligible ? 'YES' : 'NO',
    commissionAmount,
    commissionStatus,
    payload.raw,
    eligible ? '' : 'Jika transaksi ini milik affiliate, isi kolom Manual Affiliate Code.',
    now,
    createdAt,
    messageAction,
    messageCode,
    totalPrice,
    totalAddon,
    discount,
    shipping,
    JSON.stringify(items),
    '',
    affiliate ? 'AUTO_ATTRIBUTED' : 'NEEDS_REVIEW',
    '',
  ]);

  if (eligible) {
    const commissionId = Utilities.getUuid();

    sheets.comm.appendRow([
      commissionId,
      transactionId || '',
      now,
      affiliate.code,
      affiliate.slug,
      affiliate.name,
      productName || 'Paket Panglima UMKM',
      amount,
      PANGLIMA_COMMISSION_RATE,
      PANGLIMA_COMMISSION_AMOUNT,
      'PENDING',
      'UNPAID',
      '',
      '',
      '',
      '',
    ]);

    updateRegistryPurchaseSummary_(
      registry,
      affiliate.row,
      amount,
      PANGLIMA_COMMISSION_AMOUNT,
      now
    );
  }

  return json_({
    ok: true,
    transactionId,
    paymentStatus,
    attributionStatus,
    attributionSource,
    commissionEligible: eligible,
    commissionAmount,
  });
}

function updateRegistryPurchaseSummary_(sheet, row, amount, commissionAmount, now) {
  ensureStep9RegistryColumns_(sheet);

  const totalPurchasesCell = sheet.getRange(row, 33);
  const grossRevenueCell = sheet.getRange(row, 34);
  const earnedCell = sheet.getRange(row, 35);
  const pendingCell = sheet.getRange(row, 36);
  const lastPurchaseCell = sheet.getRange(row, 39);

  totalPurchasesCell.setValue((Number(totalPurchasesCell.getValue()) || 0) + 1);
  grossRevenueCell.setValue((Number(grossRevenueCell.getValue()) || 0) + (Number(amount) || 0));
  earnedCell.setValue((Number(earnedCell.getValue()) || 0) + commissionAmount);
  pendingCell.setValue((Number(pendingCell.getValue()) || 0) + commissionAmount);
  lastPurchaseCell.setValue(now);
}

function handleStep9CProductionWebhook_(e, p) {
  const payload = parseWebhookPayload_(e, p);

  const eventType = clean_(pickWebhookValue_(payload, [
    'event',
    'type',
    'eventType',
    'event_type',
    'message.action',
    'data.event',
    'data.message.action',
  ])).toLowerCase();

  // Endpoint /exec juga dipakai oleh pendaftaran affiliate.
  // Jadi hanya payload transaksi Lynk yang valid yang diteruskan ke ledger.
  if (
    eventType !== 'payment.received' &&
    eventType !== 'payment_received'
  ) {
    return json_({
      ok: true,
      ignored: true,
      reason: 'NOT_PAYMENT_RECEIVED',
      eventType,
    });
  }

  const refId = clean_(pickWebhookValue_(payload, [
    'refId',
    'ref_id',
    'data.refId',
    'data.ref_id',
    'transaction.refId',
    'transaction.ref_id',
  ]));

  const items = extractLynkItems_(payload);

  const totalsObj = pickWebhookObject_(payload, [
    'totals',
    'data.totals',
    'order.totals',
    'data.order.totals',
  ]) || {};

  const grandTotal = numberWebhookValue_(
    totalsObj.grandTotal ||
    totalsObj.grand_total ||
    pickWebhookValue_(payload, [
      'grandTotal',
      'grand_total',
      'amount',
      'total',
      'data.grandTotal',
      'data.grand_total',
      'data.amount',
      'data.total',
    ])
  );

  // Guardrail minimum supaya request acak ke /exec tidak membuat komisi.
  // Payload tanpa refId / item / nominal tetap dicatat sebagai tidak valid
  // dan tidak masuk commission ledger.
  if (!refId || !items.length || grandTotal <= 0) {

      rawPayload: payload.raw,
    });

    return json_({
      ok: false,
      ignored: true,
      error: 'INVALID_LYNK_PAYMENT_PAYLOAD',
    });
  }

  return handlePaymentWebhook_(e, p);
}

function deleteRowsByTestId_(sheet, idColumn) {
  if (!sheet || sheet.getLastRow() < 2) return 0;

  let deleted = 0;

  for (let row = sheet.getLastRow(); row >= 2; row--) {
    const id = clean_(sheet.getRange(row, idColumn).getDisplayValue());

    if (isStep9TestTransactionId_(id)) {
      sheet.deleteRow(row);

    }
  }

  return deleted;
}

function rebuildAffiliateFinancialSummary_() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const registry = ss.getSheetByName(SHEET_NAME);
  const commissionSheet = ss.getSheetByName(AFFILIATE_COMMISSIONS_SHEET);

  if (!registry || !commissionSheet) {
    throw new Error('Registry / Affiliate Commissions tidak ditemukan.');
  }

  ensureStep9RegistryColumns_(registry);

  const lastRegistryRow = registry.getLastRow();

  if (lastRegistryRow < 2) {
    return 'Tidak ada affiliate untuk dihitung.';
  }

  // Reset kolom AG:AM / 33:39 untuk seluruh affiliate.
  registry
    .getRange(2, 33, lastRegistryRow - 1, 7)
    .clearContent();

  const registryRows = registry
    .getRange(2, 1, lastRegistryRow - 1, Math.max(39, registry.getLastColumn()))
    .getDisplayValues();

  const registryByCode = {};

  registryRows.forEach((r, index) => {
    const code = clean_(r[10]).toUpperCase();

    if (code) {
      registryByCode[code] = {
        row: index + 2,
        purchases: 0,
        revenue: 0,
        earned: 0,
        pending: 0,
        approved: 0,
        paid: 0,
        lastPurchase: null,
      };
    }
  });

  if (commissionSheet.getLastRow() >= 2) {
    const rows = commissionSheet
      .getRange(
        2,
        1,
        commissionSheet.getLastRow() - 1,
        Math.max(16, commissionSheet.getLastColumn())
      )
      .getValues();

    rows.forEach(r => {
      const transactionId = clean_(r[1]);

      if (isStep9TestTransactionId_(transactionId)) return;

      const affiliateCode = clean_(r[3]).toUpperCase();
      const summary = registryByCode[affiliateCode];

      if (!summary) return;

      const paymentDate = r[2] instanceof Date ? r[2] : new Date(r[2]);
      const saleAmount = Number(r[7]) || 0;
      const commissionAmount = Number(r[9]) || 0;
      const commissionStatus = clean_(r[10]).toUpperCase();
      const payoutStatus = clean_(r[11]).toUpperCase();

      summary.purchases += 1;
      summary.revenue += saleAmount;
      summary.earned += commissionAmount;

      if (payoutStatus === 'PAID' || commissionStatus === 'PAID') {
        summary.paid += commissionAmount;
      } else if (commissionStatus === 'APPROVED') {
        summary.approved += commissionAmount;
      } else {
        summary.pending += commissionAmount;
      }

      if (
        paymentDate instanceof Date &&
        !isNaN(paymentDate.getTime()) &&
        (!summary.lastPurchase || paymentDate > summary.lastPurchase)
      ) {
        summary.lastPurchase = paymentDate;
      }
    });
  }

  Object.keys(registryByCode).forEach(code => {
    const s = registryByCode[code];

    registry.getRange(s.row, 33, 1, 7).setValues([[
      s.purchases,
      s.revenue,
      s.earned,
      s.pending,
      s.approved,
      s.paid,
      s.lastPurchase || '',
    ]]);
  });

  return 'Financial summary affiliate berhasil dihitung ulang.';
}

  const tx = ss.getSheetByName(AFFILIATE_TRANSACTIONS_SHEET);
  const comm = ss.getSheetByName(AFFILIATE_COMMISSIONS_SHEET);

function verifyAffiliateEmail_(token) {
  if (!token || token.length < 20) {
    return verificationPage_(false, 'Tautan verifikasi tidak valid.', 'Silakan gunakan tautan terbaru dari email pendaftaran Anda.');
  }

  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) throw new Error('Sheet tidak ditemukan.');

  const lastRow = sheet.getLastRow();
  if (lastRow < 2) {
    return verificationPage_(false, 'Data pendaftaran tidak ditemukan.', 'Silakan daftar kembali.');
  }

  const tokens = sheet.getRange(2, 25, lastRow - 1, 1).getDisplayValues();
  let foundRow = -1;
  for (let i = 0; i < tokens.length; i++) {
    if (clean_(tokens[i][0]) === token) {
      foundRow = i + 2;
      break;
    }
  }

  if (foundRow === -1) {
    return verificationPage_(false, 'Tautan verifikasi tidak ditemukan.', 'Tautan mungkin salah atau data pendaftaran sudah berubah.');
  }

  const data = getAffiliateRowData_(sheet, foundRow);

  if (data.emailVerified !== 'YES') {
    sheet.getRange(foundRow, 24).setValue('YES');
    sheet.getRange(foundRow, 26).setValue(new Date());

    if (data.status === 'PENDING_EMAIL' || data.status === 'EMAIL_ERROR') {
      sheet.getRange(foundRow, 14).setValue('PENDING');
    }
  }

  return verificationPage_(
    true,
    'Email berhasil diverifikasi.',
    `Terima kasih, ${data.firstName || data.name}. Pendaftaran Anda sekarang masuk ke proses review admin.`
  );
}

/* =========================================================
   APPROVAL TRIGGER
   ========================================================= */
function handleAffiliateApproval(e) {
  try {
    if (!e || !e.range) return;

    const sheet = e.range.getSheet();
    if (sheet.getName() !== SHEET_NAME) return;
    if (e.range.getRow() < 2) return;
    if (e.range.getColumn() !== 14) return; // N Status only

    const row = e.range.getRow();
    const status = clean_(sheet.getRange(row, 14).getDisplayValue()).toUpperCase();
    if (status !== 'APPROVED') return;

    const data = getAffiliateRowData_(sheet, row);
    const validCheckout = /^https:\/\//i.test(data.checkoutUrl) && data.checkoutUrl.indexOf('lynk.id') !== -1;
    const duplicateOkay = data.duplicateFlag === 'OK';
    const verifiedOkay = data.emailVerified === 'YES';
    const validEmail = isValidEmail_(data.email);

    if (!verifiedOkay || !validCheckout || !duplicateOkay || !validEmail) {
      let nextStatus = 'PENDING';
      let reason = '';

      if (!verifiedOkay) {
        nextStatus = 'PENDING_EMAIL';
        reason = 'Email affiliate belum diverifikasi.';
      } else if (!validCheckout) {
        nextStatus = 'PENDING';
        reason = 'Link Lynk Affiliate belum valid.';
      } else if (!duplicateOkay) {
        nextStatus = 'REVIEW_DUPLICATE';
        reason = 'Duplicate Flag belum OK.';
      } else {
        nextStatus = 'PENDING';
        reason = 'Alamat email tidak valid.';
      }

      sheet.getRange(row, 14).setValue(nextStatus);
      sheet.getRange(row, 23).setValue(reason);
      return;
    }

    const approvalCell = sheet.getRange(row, 16); // P
    if (!approvalCell.getValue()) approvalCell.setValue(new Date());

    if (data.approvalEmailStatus === 'SENT') return;
    sendApprovalEmailForRow_(sheet, row);
  } catch (err) {
    console.error(err);
    try {
      if (e && e.range) {
        e.range.getSheet().getRange(e.range.getRow(), 23).setValue(String(err && err.message ? err.message : err));
      }
    } catch (_) {}
  }
}

   PUBLIC APPROVED REGISTRY
   ========================================================= */
function getApprovedAffiliateRegistry_() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) throw new Error('Sheet tidak ditemukan: ' + SHEET_NAME);

  ensureStep6AHeaders_(sheet);

  const lastRow = sheet.getLastRow();
  const rows = lastRow > 1
    ? sheet.getRange(2, 1, lastRow - 1, 29).getDisplayValues()
    : [];

  const partners = {};

  rows.forEach((r) => {
    const name = clean_(r[1]);
    const whatsapp = normalizePhone_(r[3]);
    const referralCode = clean_(r[10]).toUpperCase();
    const slug = slugify_(r[11]);
    const status = clean_(r[13]).toUpperCase();
    const checkoutUrl = clean_(r[14]);
    const duplicateFlag = clean_(r[19]).toUpperCase();
    const emailVerified = clean_(r[23]).toUpperCase();

    const validCheckout = /^https:\/\//i.test(checkoutUrl) && checkoutUrl.indexOf('lynk.id') !== -1;

    if (!name || !whatsapp || !referralCode || !slug) return;
    if (status !== 'APPROVED') return;
    if (duplicateFlag !== 'OK') return;
    if (emailVerified !== 'YES') return;
    if (!validCheckout) return;

    const affiliatePage = clean_(r[12]) || `${SITE_URL}/?affiliate=${encodeURIComponent(slug)}`;
    const kitUrl = clean_(r[28]) || `${SITE_URL}/?affiliate=${encodeURIComponent(slug)}&kit=1`;

    partners[slug] = {
      name,
      title: 'Partner Panglima UMKM',
      whatsapp,
      code: referralCode,
      affiliatePage,
      kitUrl,
      checkoutUrl,
      status: 'APPROVED',
      active: true,
    };
  });

  return {
    ok: true,
    generatedAt: new Date().toISOString(),
    count: Object.keys(partners).length,
    partners,
  };
}

function createVerificationToken_() {
  return (Utilities.getUuid() + Utilities.getUuid()).replace(/-/g, '');
}

function createReferralCode_(firstName, whatsapp) {
  const letters = removeDiacritics_(firstName)
    .replace(/[^a-zA-Z]/g, '')
    .toUpperCase()
    .slice(0, 3)
    .padEnd(3, 'X');
  const digits = normalizePhone_(whatsapp);
  const last3 = digits.slice(-3).padStart(3, '0');
  return letters + last3;
}

function getFirstName_(name) {
  return clean_(name).split(/\s+/)[0] || '';
}

function normalizePhone_(value) {
  let digits = String(value || '').replace(/\D/g, '');
  if (digits.startsWith('0')) digits = '62' + digits.slice(1);
  return digits;
}

function slugify_(value) {
  return removeDiacritics_(String(value || ''))
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-+|-+$/g, '') || 'partner';
}

function removeDiacritics_(value) {
  return String(value || '').normalize('NFD').replace(/[\u0300-\u036f]/g, '');
}

function isValidEmail_(email) {
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(clean_(email));
}

function clean_(value) {
  return String(value || '').trim();
}

function escapeHtml_(value) {
  return String(value || '')
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#039;');
}

function json_(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function jsonOrJsonp_(data, callback) {
  const safeCallback = String(callback || '').replace(/[^a-zA-Z0-9_.$]/g, '');
  if (safeCallback) {
    return ContentService
      .createTextOutput(`${safeCallback}(${JSON.stringify(data)});`)
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return json_(data);
}
