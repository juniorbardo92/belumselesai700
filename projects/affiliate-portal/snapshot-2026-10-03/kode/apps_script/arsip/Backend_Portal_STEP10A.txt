const PORTAL_SESSIONS_SHEET = 'Affiliate Portal Sessions';
const PORTAL_REGISTRY_SHEET = 'Affiliate Registry';
const PORTAL_TRANSACTIONS_SHEET = 'Affiliate Transactions';
const PORTAL_COMMISSIONS_SHEET = 'Affiliate Commissions';

const PORTAL_OTP_TTL_MINUTES = 10;
const PORTAL_OTP_MAX_ATTEMPTS = 5;
const PORTAL_SESSION_DAYS = 7;
const PORTAL_SENDER_ALIAS = 'halo@umkmnaikkelas.id';

function portalHandleAction_(action, payload, e) {
  action = String(action || '').trim().toLowerCase();
  payload = payload || {};

  if (action === 'portal_request_otp') {
    return portalRequestOtp_(payload, e);
  }

  if (action === 'portal_verify_otp') {
    return portalVerifyOtp_(payload, e);
  }

  if (action === 'portal_dashboard') {
    return portalGetDashboard_(payload, e);
  }

  if (action === 'portal_logout') {
    return portalLogout_(payload, e);
  }

  return {
    ok: false,
    error: 'UNKNOWN_PORTAL_ACTION'
  };
}

function portalRequestOtp_(payload, e) {
  const email = portalNormalizeEmail_(payload.email);

  if (!portalIsValidEmail_(email)) {
    return {
      ok: false,
      error: 'INVALID_EMAIL',
      message: 'Masukkan email affiliate yang valid.'
    };
  }

  const affiliate = portalFindApprovedAffiliateByEmail_(email);

  /*
   * Jangan bocorkan apakah email terdaftar atau tidak.
   * Response sengaja sama untuk email yang tidak eligible.
   */
  if (!affiliate) {
    Utilities.sleep(250);
    return {
      ok: true,
      message: 'Jika email terdaftar sebagai affiliate aktif, kode OTP akan dikirim.'
    };
  }

  const lock = LockService.getScriptLock();
  lock.waitLock(15000);

  try {
    const recent = portalFindRecentOtpRequest_(email);

    if (recent && recent.ageSeconds < 60) {
      return {
        ok: true,
        message: 'Kode OTP baru saja dikirim. Periksa email Anda.'
      };
    }

    portalEnsureSecuritySalt_();

    const otp = String(Math.floor(100000 + Math.random() * 900000));
    const recordId = 'OTP-' + Utilities.getUuid();
    const createdAt = new Date();
    const expiresAt = new Date(
      createdAt.getTime() + PORTAL_OTP_TTL_MINUTES * 60 * 1000
    );

    const otpHash = portalHash_(
      email + '|' + otp + '|' + recordId
    );

    const userAgent = portalGetUserAgent_(e);

    const sheet = portalGetSheet_(PORTAL_SESSIONS_SHEET);

    sheet.appendRow([
      recordId,                 // A Record ID
      createdAt,                // B Created At
      email,                    // C Email
      affiliate.code,           // D Affiliate Code
      affiliate.slug,           // E Affiliate Slug
      otpHash,                  // F OTP Hash
      expiresAt,                // G OTP Expires At
      'PENDING',                // H OTP Status
      0,                        // I OTP Attempts
      '',                       // J Session Token Hash
      '',                       // K Session Expires At
      '',                       // L Last Seen At
      'OTP_PENDING',            // M Status
      userAgent,                // N User Agent
      ''                        // O Notes
    ]);

    portalSendOtpEmail_(
      email,
      affiliate.name,
      otp,
      PORTAL_OTP_TTL_MINUTES
    );

    return {
      ok: true,
      message: 'Jika email terdaftar sebagai affiliate aktif, kode OTP akan dikirim.',
      expiresInMinutes: PORTAL_OTP_TTL_MINUTES
    };

  } finally {
    lock.releaseLock();
  }
}

function portalVerifyOtp_(payload, e) {
  const email = portalNormalizeEmail_(payload.email);
  const otp = String(payload.otp || '').replace(/\D/g, '').trim();

  if (!portalIsValidEmail_(email) || !/^\d{6}$/.test(otp)) {
    return {
      ok: false,
      error: 'INVALID_OTP_REQUEST',
      message: 'Email atau kode OTP tidak valid.'
    };
  }

  const affiliate = portalFindApprovedAffiliateByEmail_(email);

  if (!affiliate) {
    return {
      ok: false,
      error: 'OTP_INVALID_OR_EXPIRED',
      message: 'Kode OTP salah atau sudah kedaluwarsa.'
    };
  }

  const lock = LockService.getScriptLock();
  lock.waitLock(15000);

  try {
    const record = portalFindLatestPendingOtp_(email);

    if (!record) {
      return {
        ok: false,
        error: 'OTP_INVALID_OR_EXPIRED',
        message: 'Kode OTP salah atau sudah kedaluwarsa.'
      };
    }

    const now = new Date();
    const expiresAt = portalAsDate_(record.values[6]);
    const attempts = Number(record.values[8] || 0);

    if (!expiresAt || now.getTime() > expiresAt.getTime()) {
      portalUpdateSessionRow_(record.row, {
        otpStatus: 'EXPIRED',
        status: 'OTP_EXPIRED',
        notes: 'OTP expired before successful verification.'
      });

      return {
        ok: false,
        error: 'OTP_INVALID_OR_EXPIRED',
        message: 'Kode OTP salah atau sudah kedaluwarsa.'
      };
    }

    if (attempts >= PORTAL_OTP_MAX_ATTEMPTS) {
      portalUpdateSessionRow_(record.row, {
        otpStatus: 'LOCKED',
        status: 'OTP_LOCKED',
        notes: 'Maximum OTP attempts reached.'
      });

      return {
        ok: false,
        error: 'OTP_LOCKED',
        message: 'Terlalu banyak percobaan. Minta kode OTP baru.'
      };
    }

    const expectedHash = portalHash_(
      email + '|' + otp + '|' + record.values[0]
    );

    const storedHash = String(record.values[5] || '');

    if (!portalSafeEqual_(expectedHash, storedHash)) {
      const newAttempts = attempts + 1;

      portalUpdateSessionRow_(record.row, {
        otpAttempts: newAttempts,
        notes: 'Invalid OTP attempt.'
      });

      return {
        ok: false,
        error: 'OTP_INVALID_OR_EXPIRED',
        message: 'Kode OTP salah atau sudah kedaluwarsa.',
        attemptsRemaining: Math.max(
          0,
          PORTAL_OTP_MAX_ATTEMPTS - newAttempts
        )
      };
    }

    const sessionToken =
      Utilities.getUuid().replace(/-/g, '') +
      Utilities.getUuid().replace(/-/g, '');

    const sessionHash = portalHash_(sessionToken);
    const sessionExpiresAt = new Date(
      now.getTime() + PORTAL_SESSION_DAYS * 24 * 60 * 60 * 1000
    );

    portalUpdateSessionRow_(record.row, {
      otpStatus: 'VERIFIED',
      otpAttempts: attempts,
      sessionTokenHash: sessionHash,
      sessionExpiresAt: sessionExpiresAt,
      lastSeenAt: now,
      status: 'ACTIVE',
      notes: 'Portal login successful.'
    });

    return {
      ok: true,
      authenticated: true,
      sessionToken: sessionToken,
      expiresAt: sessionExpiresAt.toISOString(),
      affiliate: {
        name: affiliate.name,
        code: affiliate.code,
        slug: affiliate.slug
      }
    };

  } finally {
    lock.releaseLock();
  }
}

function portalGetDashboard_(payload, e) {
  const token = String(
    payload.sessionToken ||
    payload.session_token ||
    ''
  ).trim();

  if (!token) {
    return {
      ok: false,
      authenticated: false,
      error: 'SESSION_REQUIRED'
    };
  }

  const session = portalFindActiveSession_(token);

  if (!session) {
    return {
      ok: false,
      authenticated: false,
      error: 'SESSION_INVALID_OR_EXPIRED'
    };
  }

  const email = portalNormalizeEmail_(session.values[2]);
  const code = String(session.values[3] || '').trim();
  const slug = String(session.values[4] || '').trim();

  const affiliate = portalFindApprovedAffiliateByEmail_(email);

  if (
    !affiliate ||
    affiliate.code !== code ||
    affiliate.slug !== slug
  ) {
    portalUpdateSessionRow_(session.row, {
      status: 'REVOKED',
      notes: 'Affiliate is no longer eligible.'
    });

    return {
      ok: false,
      authenticated: false,
      error: 'AFFILIATE_NOT_ACTIVE'
    };
  }

  const now = new Date();
  portalUpdateSessionRow_(session.row, {
    lastSeenAt: now
  });

  const metrics = portalBuildAffiliateMetrics_(
    affiliate.code,
    affiliate.slug
  );

  return {
    ok: true,
    authenticated: true,
    generatedAt: now.toISOString(),
    affiliate: {
      name: affiliate.name,
      firstName: affiliate.firstName,
      email: affiliate.email,
      code: affiliate.code,
      slug: affiliate.slug,
      city: affiliate.city,
      status: affiliate.status,
      affiliatePage: affiliate.affiliatePage,
      checkoutUrl: affiliate.checkoutUrl,
      marketingKitUrl: affiliate.marketingKitUrl
    },
    metrics: metrics.summary,
    transactions: metrics.transactions
  };
}

function portalLogout_(payload, e) {
  const token = String(
    payload.sessionToken ||
    payload.session_token ||
    ''
  ).trim();

  if (!token) {
    return { ok: true };
  }

  const session = portalFindActiveSession_(token);

  if (session) {
    portalUpdateSessionRow_(session.row, {
      status: 'LOGGED_OUT',
      notes: 'Affiliate logged out.'
    });
  }

  return { ok: true };
}

function portalBuildAffiliateMetrics_(affiliateCode, affiliateSlug) {
  const registry = portalFindAffiliateByCode_(affiliateCode);

  const pageViews = registry
    ? Number(registry.totalPageViews || 0)
    : 0;

  const checkoutClicks = registry
    ? Number(registry.totalCheckoutClicks || 0)
    : 0;

  const txSheet = portalGetSheet_(PORTAL_TRANSACTIONS_SHEET);
  const txLastRow = txSheet.getLastRow();

  let purchases = 0;
  let grossRevenue = 0;
  let pendingCommission = 0;
  const transactions = [];

  if (txLastRow >= 2) {
    const rows = txSheet
      .getRange(2, 1, txLastRow - 1, 32)
      .getValues();

    rows.forEach(function(row) {
      const rowCode = String(row[11] || '').trim(); // L
      const rowSlug = String(row[12] || '').trim(); // M
      const eligible = String(row[15] || '').toUpperCase() === 'YES'; // P
      const paymentStatus = String(row[3] || '').toUpperCase(); // D

      if (
        rowCode !== affiliateCode &&
        rowSlug !== affiliateSlug
      ) {
        return;
      }

      if (paymentStatus !== 'PAID') {
        return;
      }

      purchases += 1;
      grossRevenue += Number(row[9] || 0); // J

      if (eligible) {
        pendingCommission += Number(row[16] || 0); // Q
      }

      transactions.push({
        receivedAt: portalDateIso_(row[0]),
        transactionId: String(row[1] || ''),
        product: String(row[5] || ''),
        buyerName: String(row[6] || ''),
        amount: Number(row[9] || 0),
        commissionEligible: eligible,
        commissionAmount: Number(row[16] || 0),
        commissionStatus: String(row[17] || ''),
        attributionSource: String(row[14] || '')
      });
    });
  }

  transactions.sort(function(a, b) {
    return String(b.receivedAt).localeCompare(String(a.receivedAt));
  });

  const commissionSummary =
    portalBuildCommissionSummary_(affiliateCode);

  const conversionRate =
    checkoutClicks > 0
      ? purchases / checkoutClicks
      : 0;

  const monthSales = portalCountCurrentMonthSales_(
    transactions
  );

  const monthlyBonus = portalCalculateMonthlyBonus_(monthSales);

  return {
    summary: {
      pageViews: pageViews,
      checkoutClicks: checkoutClicks,
      purchases: purchases,
      conversionRate: conversionRate,
      grossRevenue: grossRevenue,
      baseCommissionEarned:
        commissionSummary.totalEarned,
      commissionPending:
        commissionSummary.pending,
      commissionApproved:
        commissionSummary.approved,
      commissionPaid:
        commissionSummary.paid,
      currentMonthSales: monthSales,
      monthlyBonus: monthlyBonus
    },
    transactions: transactions.slice(0, 50)
  };
}

function portalBuildCommissionSummary_(affiliateCode) {
  const sheet = portalGetSheet_(PORTAL_COMMISSIONS_SHEET);
  const lastRow = sheet.getLastRow();

  let totalEarned = 0;
  let pending = 0;
  let approved = 0;
  let paid = 0;

  if (lastRow < 2) {
    return {
      totalEarned: 0,
      pending: 0,
      approved: 0,
      paid: 0
    };
  }

  const rows = sheet
    .getRange(2, 1, lastRow - 1, 16)
    .getValues();

  rows.forEach(function(row) {
    const code = String(row[3] || '').trim(); // D
    if (code !== affiliateCode) return;

    const amount = Number(row[9] || 0); // J
    const commissionStatus =
      String(row[10] || '').toUpperCase(); // K
    const payoutStatus =
      String(row[11] || '').toUpperCase(); // L

    totalEarned += amount;

    if (
      commissionStatus === 'PENDING' ||
      payoutStatus === 'PENDING'
    ) {
      pending += amount;
    }

    if (
      commissionStatus === 'APPROVED' &&
      payoutStatus !== 'PAID'
    ) {
      approved += amount;
    }

    if (payoutStatus === 'PAID') {
      paid += amount;
    }
  });

  return {
    totalEarned: totalEarned,
    pending: pending,
    approved: approved,
    paid: paid
  };
}

function portalCountCurrentMonthSales_(transactions) {
  const now = new Date();
  const year = now.getFullYear();
  const month = now.getMonth();

  return transactions.filter(function(tx) {
    if (!tx.receivedAt) return false;
    const d = new Date(tx.receivedAt);
    return (
      !isNaN(d.getTime()) &&
      d.getFullYear() === year &&
      d.getMonth() === month
    );
  }).length;
}

function portalCalculateMonthlyBonus_(sales) {
  let earned = 0;
  let level = 'Belum ada bonus';
  let nextTarget = 5;
  let nextBonus = 100000;

  if (sales >= 20) {
    earned = 600000;
    level = '20 Penjualan';
    nextTarget = null;
    nextBonus = null;
  } else if (sales >= 10) {
    earned = 250000;
    level = '10 Penjualan';
    nextTarget = 20;
    nextBonus = 600000;
  } else if (sales >= 5) {
    earned = 100000;
    level = '5 Penjualan';
    nextTarget = 10;
    nextBonus = 250000;
  }

  return {
    earned: earned,
    level: level,
    sales: sales,
    nextTarget: nextTarget,
    nextBonus: nextBonus,
    remainingToNext:
      nextTarget === null
        ? 0
        : Math.max(0, nextTarget - sales)
  };
}

function portalFindApprovedAffiliateByEmail_(email) {
  const sheet = portalGetSheet_(PORTAL_REGISTRY_SHEET);
  const lastRow = sheet.getLastRow();

  if (lastRow < 2) return null;

  const rows = sheet
    .getRange(2, 1, lastRow - 1, 39)
    .getValues();

  for (let i = 0; i < rows.length; i++) {
    const r = rows[i];

    const rowEmail =
      portalNormalizeEmail_(r[4]); // E Email
    const status =
      String(r[13] || '').trim().toUpperCase(); // N
    const emailVerified =
      String(r[23] || '').trim().toUpperCase(); // X

    if (
      rowEmail === email &&
      status === 'APPROVED' &&
      (
        emailVerified === 'YES' ||
        emailVerified === 'TRUE' ||
        emailVerified === 'VERIFIED'
      )
    ) {
      return {
        row: i + 2,
        name: String(r[1] || ''),
        firstName: String(r[2] || ''),
        email: rowEmail,
        city: String(r[6] || ''),
        code: String(r[10] || '').trim(),
        slug: String(r[11] || '').trim(),
        affiliatePage: String(r[12] || ''),
        status: status,
        checkoutUrl: String(r[14] || ''),
        marketingKitUrl: String(r[28] || '')
      };
    }
  }

  return null;
}

function portalFindAffiliateByCode_(affiliateCode) {
  const sheet = portalGetSheet_(PORTAL_REGISTRY_SHEET);
  const lastRow = sheet.getLastRow();

  if (lastRow < 2) return null;

  const rows = sheet
    .getRange(2, 1, lastRow - 1, 39)
    .getValues();

  for (let i = 0; i < rows.length; i++) {
    const r = rows[i];

    if (
      String(r[10] || '').trim() === affiliateCode
    ) {
      return {
        row: i + 2,
        code: affiliateCode,
        totalPageViews: Number(r[29] || 0),       // AD
        totalCheckoutClicks: Number(r[30] || 0),  // AE
        lastAffiliateActivity: r[31] || ''        // AF
      };
    }
  }

  return null;
}

function portalFindRecentOtpRequest_(email) {
  const sheet = portalGetSheet_(PORTAL_SESSIONS_SHEET);
  const lastRow = sheet.getLastRow();

  if (lastRow < 2) return null;

  const rows = sheet
    .getRange(2, 1, lastRow - 1, 15)
    .getValues();

  for (let i = rows.length - 1; i >= 0; i--) {
    const r = rows[i];

    if (
      portalNormalizeEmail_(r[2]) !== email
    ) {
      continue;
    }

    const createdAt = portalAsDate_(r[1]);
    if (!createdAt) return null;

    return {
      row: i + 2,
      values: r,
      ageSeconds:
        (Date.now() - createdAt.getTime()) / 1000
    };
  }

  return null;
}

function portalFindLatestPendingOtp_(email) {
  const sheet = portalGetSheet_(PORTAL_SESSIONS_SHEET);
  const lastRow = sheet.getLastRow();

  if (lastRow < 2) return null;

  const rows = sheet
    .getRange(2, 1, lastRow - 1, 15)
    .getValues();

  for (let i = rows.length - 1; i >= 0; i--) {
    const r = rows[i];

    if (
      portalNormalizeEmail_(r[2]) === email &&
      String(r[7] || '').toUpperCase() === 'PENDING'
    ) {
      return {
        row: i + 2,
        values: r
      };
    }
  }

  return null;
}

function portalFindActiveSession_(rawToken) {
  const sheet = portalGetSheet_(PORTAL_SESSIONS_SHEET);
  const lastRow = sheet.getLastRow();

  if (lastRow < 2) return null;

  const tokenHash = portalHash_(rawToken);

  const rows = sheet
    .getRange(2, 1, lastRow - 1, 15)
    .getValues();

  for (let i = rows.length - 1; i >= 0; i--) {
    const r = rows[i];

    const storedHash = String(r[9] || '');
    const expiresAt = portalAsDate_(r[10]);
    const status = String(r[12] || '').toUpperCase();

    if (
      storedHash &&
      portalSafeEqual_(tokenHash, storedHash) &&
      status === 'ACTIVE'
    ) {
      if (
        !expiresAt ||
        Date.now() > expiresAt.getTime()
      ) {
        portalUpdateSessionRow_(i + 2, {
          status: 'EXPIRED',
          notes: 'Session expired.'
        });

        return null;
      }

      return {
        row: i + 2,
        values: r
      };
    }
  }

  return null;
}

function portalUpdateSessionRow_(row, patch) {
  const sheet = portalGetSheet_(PORTAL_SESSIONS_SHEET);

  if (patch.otpStatus !== undefined) {
    sheet.getRange(row, 8).setValue(patch.otpStatus); // H
  }

  if (patch.otpAttempts !== undefined) {
    sheet.getRange(row, 9).setValue(patch.otpAttempts); // I
  }

  if (patch.sessionTokenHash !== undefined) {
    sheet.getRange(row, 10).setValue(
      patch.sessionTokenHash
    ); // J
  }

  if (patch.sessionExpiresAt !== undefined) {
    sheet.getRange(row, 11).setValue(
      patch.sessionExpiresAt
    ); // K
  }

  if (patch.lastSeenAt !== undefined) {
    sheet.getRange(row, 12).setValue(
      patch.lastSeenAt
    ); // L
  }

  if (patch.status !== undefined) {
    sheet.getRange(row, 13).setValue(
      patch.status
    ); // M
  }

  if (patch.notes !== undefined) {
    sheet.getRange(row, 15).setValue(
      patch.notes
    ); // O
  }
}

function portalEnsureSecuritySalt_() {
  const props =
    PropertiesService.getScriptProperties();

  let salt = props.getProperty(
    'PORTAL_SECURITY_SALT'
  );

  if (!salt) {
    salt =
      Utilities.getUuid().replace(/-/g, '') +
      Utilities.getUuid().replace(/-/g, '');

    props.setProperty(
      'PORTAL_SECURITY_SALT',
      salt
    );
  }

  return salt;
}

function portalHash_(value) {
  const salt = portalEnsureSecuritySalt_();

  const bytes = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256,
    String(value) + '|' + salt,
    Utilities.Charset.UTF_8
  );

  return bytes.map(function(b) {
    const n = b < 0 ? b + 256 : b;
    return ('0' + n.toString(16)).slice(-2);
  }).join('');
}

function portalSafeEqual_(a, b) {
  a = String(a || '');
  b = String(b || '');

  if (a.length !== b.length) return false;

  let diff = 0;

  for (let i = 0; i < a.length; i++) {
    diff |= a.charCodeAt(i) ^ b.charCodeAt(i);
  }

  return diff === 0;
}

        '</p>' +
      '</div>' +
    '</div>';

  const options = {
    htmlBody: htmlBody,
    name: 'UMKM Naik Kelas',
    replyTo: PORTAL_SENDER_ALIAS
  };

  try {
    const aliases = GmailApp.getAliases();

    if (
      aliases &&
      aliases.indexOf(PORTAL_SENDER_ALIAS) >= 0
    ) {
      options.from = PORTAL_SENDER_ALIAS;
    }
  } catch (err) {
    // Jika Gmail alias tidak tersedia,
    // email tetap dikirim dari akun eksekutor.
  }

  GmailApp.sendEmail(
    email,
    subject,
    plainBody,
    options
  );
}

function portalGetSheet_(name) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(name);

  if (!sheet) {
    throw new Error(
      'Sheet tidak ditemukan: ' + name
    );
  }

  return sheet;
}

function portalNormalizeEmail_(value) {
  return String(value || '')
    .trim()
    .toLowerCase();
}

function portalIsValidEmail_(email) {
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(
    String(email || '')
  );
}

function portalGetUserAgent_(e) {
  try {
    return String(
      e &&
      e.parameter &&
      e.parameter.userAgent
        ? e.parameter.userAgent
        : ''
    );
  } catch (err) {
    return '';
  }
}

function portalAsDate_(value) {
  if (!value) return null;

  if (
    Object.prototype.toString.call(value) ===
    '[object Date]'
  ) {
    return isNaN(value.getTime())
      ? null
      : value;
  }

  const d = new Date(value);
  return isNaN(d.getTime()) ? null : d;
}

function portalDateIso_(value) {
  const d = portalAsDate_(value);
  return d ? d.toISOString() : '';
}

function portalEscapeHtml_(value) {
  return String(value || '')
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#039;');
}
