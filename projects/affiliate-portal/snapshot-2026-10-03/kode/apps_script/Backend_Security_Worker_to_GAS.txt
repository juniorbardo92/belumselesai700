/* =========================================================
   MAIN PORTAL ROUTER
   ========================================================= */

function portalHandleAction_(action, payload, e) {
  action = String(action || '')
    .trim()
    .toLowerCase();

  payload = payload || {};

  if (action === 'portal_request_otp') {
    return portalRequestOtp_(payload, e);
  }

  if (action === 'portal_verify_otp') {
    return portalVerifyOtp_(payload, e);
  }

  if (action === 'portal_dashboard') {
    return portalGetDashboard_(payload, e);
  }

  if (action === 'portal_logout') {
    return portalLogout_(payload, e);
  }

  return {
    ok: false,
    error: 'UNKNOWN_PORTAL_ACTION'
  };
}


/* =========================================================
   REQUEST OTP
   ========================================================= */

function portalRequestOtp_(payload, e) {
  const email = portalNormalizeEmail_(
    payload.email
  );

  if (!portalIsValidEmail_(email)) {
    return {
      ok: false,
      error: 'INVALID_EMAIL',
      message: 'Masukkan email affiliate yang valid.'
    };
  }

  const affiliate =
    portalFindApprovedAffiliateByEmail_(
      email
    );

  /*
   * Security:
   * Jangan bocorkan apakah email terdaftar
   * atau tidak.
   */
  if (!affiliate) {
    Utilities.sleep(250);

    return {
      ok: true,
      message:
        'Jika email terdaftar sebagai affiliate aktif, kode OTP akan dikirim.'
    };
  }

  const lock =
    LockService.getScriptLock();

  lock.waitLock(15000);

  try {
    const recent =
      portalFindRecentOtpRequest_(
        email
      );

    /*
     * Rate limit sederhana:
     * OTP baru tidak dikirim jika
     * request sebelumnya < 60 detik.
     */
    if (
      recent &&
      recent.ageSeconds < 60
    ) {
      return {
        ok: true,
        message:
          'Kode OTP baru saja dikirim. Periksa email Anda.'
      };
    }

    portalEnsureSecuritySalt_();

    const otp = String(
      Math.floor(
        100000 +
        Math.random() * 900000
      )
    );

    const recordId =
      'OTP-' +
      Utilities.getUuid();

    const createdAt =
      new Date();

    const expiresAt =
      new Date(
        createdAt.getTime() +
        PORTAL_OTP_TTL_MINUTES *
        60 *
        1000
      );

    /*
     * OTP tidak disimpan mentah.
     * Yang disimpan hanya hash.
     */
    const otpHash =
      portalHash_(
        email +
        '|' +
        otp +
        '|' +
        recordId
      );

    const userAgent =
      portalGetUserAgent_(e);

    const sheet =
      portalGetSheet_(
        PORTAL_SESSIONS_SHEET
      );

    sheet.appendRow([
      recordId,          // A Record ID
      createdAt,         // B Created At
      email,             // C Email
      affiliate.code,    // D Affiliate Code
      affiliate.slug,    // E Affiliate Slug
      otpHash,           // F OTP Hash
      expiresAt,         // G OTP Expires At
      'PENDING',         // H OTP Status
      0,                 // I OTP Attempts
      '',                // J Session Token Hash
      '',                // K Session Expires At
      '',                // L Last Seen At
      'OTP_PENDING',     // M Status
      userAgent,         // N User Agent
      ''                 // O Notes
    ]);

    portalSendOtpEmail_(
      email,
      affiliate.name,
      otp,
      PORTAL_OTP_TTL_MINUTES
    );

    return {
      ok: true,
      message:
        'Jika email terdaftar sebagai affiliate aktif, kode OTP akan dikirim.',
      expiresInMinutes:
        PORTAL_OTP_TTL_MINUTES
    };

  } finally {
    lock.releaseLock();
  }
}


/* =========================================================
   VERIFY OTP
   ========================================================= */

function portalVerifyOtp_(payload, e) {
  const email =
    portalNormalizeEmail_(
      payload.email
    );

  const otp =
    String(
      payload.otp || ''
    )
      .replace(/\D/g, '')
      .trim();

  if (
    !portalIsValidEmail_(email) ||
    !/^\d{6}$/.test(otp)
  ) {
    return {
      ok: false,
      error:
        'INVALID_OTP_REQUEST',
      message:
        'Email atau kode OTP tidak valid.'
    };
  }

  const affiliate =
    portalFindApprovedAffiliateByEmail_(
      email
    );

  if (!affiliate) {
    return {
      ok: false,
      error:
        'OTP_INVALID_OR_EXPIRED',
      message:
        'Kode OTP salah atau sudah kedaluwarsa.'
    };
  }

  const lock =
    LockService.getScriptLock();

  lock.waitLock(15000);

  try {
    const record =
      portalFindLatestPendingOtp_(
        email
      );

    if (!record) {
      return {
        ok: false,
        error:
          'OTP_INVALID_OR_EXPIRED',
        message:
          'Kode OTP salah atau sudah kedaluwarsa.'
      };
    }

    const now =
      new Date();

    const expiresAt =
      portalAsDate_(
        record.values[6]
      );

    const attempts =
      Number(
        record.values[8] || 0
      );

    /*
     * OTP kedaluwarsa
     */
    if (
      !expiresAt ||
      now.getTime() >
      expiresAt.getTime()
    ) {
      portalUpdateSessionRow_(
        record.row,
        {
          otpStatus: 'EXPIRED',
          status: 'OTP_EXPIRED',
          notes:
            'OTP expired before successful verification.'
        }
      );

      return {
        ok: false,
        error:
          'OTP_INVALID_OR_EXPIRED',
        message:
          'Kode OTP salah atau sudah kedaluwarsa.'
      };
    }

    /*
     * Maksimal percobaan
     */
    if (
      attempts >=
      PORTAL_OTP_MAX_ATTEMPTS
    ) {
      portalUpdateSessionRow_(
        record.row,
        {
          otpStatus: 'LOCKED',
          status: 'OTP_LOCKED',
          notes:
            'Maximum OTP attempts reached.'
        }
      );

      return {
        ok: false,
        error: 'OTP_LOCKED',
        message:
          'Terlalu banyak percobaan. Minta kode OTP baru.'
      };
    }

    const expectedHash =
      portalHash_(
        email +
        '|' +
        otp +
        '|' +
        record.values[0]
      );

    const storedHash =
      String(
        record.values[5] || ''
      );

    /*
     * OTP salah
     */
    if (
      !portalSafeEqual_(
        expectedHash,
        storedHash
      )
    ) {
      const newAttempts =
        attempts + 1;

      portalUpdateSessionRow_(
        record.row,
        {
          otpAttempts:
            newAttempts,
          notes:
            'Invalid OTP attempt.'
        }
      );

      return {
        ok: false,
        error:
          'OTP_INVALID_OR_EXPIRED',
        message:
          'Kode OTP salah atau sudah kedaluwarsa.',
        attemptsRemaining:
          Math.max(
            0,
            PORTAL_OTP_MAX_ATTEMPTS -
            newAttempts
          )
      };
    }

    /*
     * OTP benar:
     * buat session token.
     */
    const sessionToken =
      Utilities
        .getUuid()
        .replace(/-/g, '') +
      Utilities
        .getUuid()
        .replace(/-/g, '');

    const sessionHash =
      portalHash_(
        sessionToken
      );

    const sessionExpiresAt =
      new Date(
        now.getTime() +
        PORTAL_SESSION_DAYS *
        24 *
        60 *
        60 *
        1000
      );

    portalUpdateSessionRow_(
      record.row,
      {
        otpStatus:
          'VERIFIED',
        otpAttempts:
          attempts,
        sessionTokenHash:
          sessionHash,
        sessionExpiresAt:
          sessionExpiresAt,
        lastSeenAt:
          now,
        status:
          'ACTIVE',
        notes:
          'Portal login successful.'
      }
    );

    /*
     * Token mentah hanya dikirim
     * sekali ke browser.
     * Di Sheet hanya hash.
     */
    return {
      ok: true,
      authenticated: true,

      sessionToken:
        sessionToken,

      expiresAt:
        sessionExpiresAt
          .toISOString(),

      affiliate: {
        name:
          affiliate.name,
        code:
          affiliate.code,
        slug:
          affiliate.slug
      }
    };

  } finally {
    lock.releaseLock();
  }
}


/* =========================================================
   DASHBOARD
   ========================================================= */

function portalGetDashboard_(payload, e) {
  const token =
    String(
      payload.sessionToken ||
      payload.session_token ||
      ''
    ).trim();

  if (!token) {
    return {
      ok: false,
      authenticated: false,
      error:
        'SESSION_REQUIRED'
    };
  }

  const session =
    portalFindActiveSession_(
      token
    );

  if (!session) {
    return {
      ok: false,
      authenticated: false,
      error:
        'SESSION_INVALID_OR_EXPIRED'
    };
  }

  const email =
    portalNormalizeEmail_(
      session.values[2]
    );

  const code =
    String(
      session.values[3] || ''
    ).trim();

  const slug =
    String(
      session.values[4] || ''
    ).trim();

  /*
   * Affiliate diverifikasi ulang
   * setiap dashboard dibuka.
   */
  const affiliate =
    portalFindApprovedAffiliateByEmail_(
      email
    );

  if (
    !affiliate ||
    affiliate.code !== code ||
    affiliate.slug !== slug
  ) {
    portalUpdateSessionRow_(
      session.row,
      {
        status:
          'REVOKED',
        notes:
          'Affiliate is no longer eligible.'
      }
    );

    return {
      ok: false,
      authenticated: false,
      error:
        'AFFILIATE_NOT_ACTIVE'
    };
  }

  const now =
    new Date();

  portalUpdateSessionRow_(
    session.row,
    {
      lastSeenAt:
        now
    }
  );

  const metrics =
    portalBuildAffiliateMetrics_(
      affiliate.code,
      affiliate.slug
    );

  return {
    ok: true,
    authenticated: true,

    generatedAt:
      now.toISOString(),

    affiliate: {
      name:
        affiliate.name,

      firstName:
        affiliate.firstName,

      email:
        affiliate.email,

      code:
        affiliate.code,

      slug:
        affiliate.slug,

      city:
        affiliate.city,

      status:
        affiliate.status,

      affiliatePage:
        affiliate.affiliatePage,

      checkoutUrl:
        affiliate.checkoutUrl,

      marketingKitUrl:
        affiliate.marketingKitUrl
    },

    metrics:
      metrics.summary,

    transactions:
      metrics.transactions
  };
}


/* =========================================================
   LOGOUT
   ========================================================= */

function portalLogout_(payload, e) {
  const token =
    String(
      payload.sessionToken ||
      payload.session_token ||
      ''
    ).trim();

  if (!token) {
    return {
      ok: true
    };
  }

  const session =
    portalFindActiveSession_(
      token
    );

  if (session) {
    portalUpdateSessionRow_(
      session.row,
      {
        status:
          'LOGGED_OUT',
        notes:
          'Affiliate logged out.'
      }
    );
  }

  return {
    ok: true
  };
}


/* =========================================================
   DASHBOARD METRICS
   ========================================================= */

function portalBuildAffiliateMetrics_(
  affiliateCode,
  affiliateSlug
) {
  const registry =
    portalFindAffiliateByCode_(
      affiliateCode
    );

  const pageViews =
    registry
      ? Number(
          registry.totalPageViews ||
          0
        )
      : 0;

  const checkoutClicks =
    registry
      ? Number(
          registry.totalCheckoutClicks ||
          0
        )
      : 0;

  const txSheet =
    portalGetSheet_(
      PORTAL_TRANSACTIONS_SHEET
    );

  const txLastRow =
    txSheet.getLastRow();

  let purchases = 0;
  let grossRevenue = 0;

  const transactions = [];

  if (txLastRow >= 2) {
    const rows =
      txSheet
        .getRange(
          2,
          1,
          txLastRow - 1,
          32
        )
        .getValues();

    rows.forEach(
      function(row) {
        const rowCode =
          String(
            row[11] || ''
          ).trim(); // L

        const rowSlug =
          String(
            row[12] || ''
          ).trim(); // M

        const eligible =
          String(
            row[15] || ''
          )
            .toUpperCase() ===
          'YES'; // P

        const paymentStatus =
          String(
            row[3] || ''
          )
            .toUpperCase(); // D

        if (
          rowCode !==
            affiliateCode &&
          rowSlug !==
            affiliateSlug
        ) {
          return;
        }

        if (
          paymentStatus !==
          'PAID'
        ) {
          return;
        }

        purchases += 1;

        grossRevenue +=
          Number(
            row[9] || 0
          ); // J

        transactions.push({
          receivedAt:
            portalDateIso_(
              row[0]
            ),

          transactionId:
            String(
              row[1] || ''
            ),

          product:
            String(
              row[5] || ''
            ),

          buyerName:
            String(
              row[6] || ''
            ),

          amount:
            Number(
              row[9] || 0
            ),

          commissionEligible:
            eligible,

          commissionAmount:
            Number(
              row[16] || 0
            ),

          commissionStatus:
            String(
              row[17] || ''
            ),

          attributionSource:
            String(
              row[14] || ''
            )
        });
      }
    );
  }

  transactions.sort(
    function(a, b) {
      return String(
        b.receivedAt
      ).localeCompare(
        String(
          a.receivedAt
        )
      );
    }
  );

  const commissionSummary =
    portalBuildCommissionSummary_(
      affiliateCode
    );

  const conversionRate =
    checkoutClicks > 0
      ? purchases /
        checkoutClicks
      : 0;

  const monthSales =
    portalCountCurrentMonthSales_(
      transactions
    );

  const monthlyBonus =
    portalCalculateMonthlyBonus_(
      monthSales
    );

  return {
    summary: {
      pageViews:
        pageViews,

      checkoutClicks:
        checkoutClicks,

      purchases:
        purchases,

      conversionRate:
        conversionRate,

      grossRevenue:
        grossRevenue,

      baseCommissionEarned:
        commissionSummary
          .totalEarned,

      commissionPending:
        commissionSummary
          .pending,

      commissionApproved:
        commissionSummary
          .approved,

      commissionPaid:
        commissionSummary
          .paid,

      currentMonthSales:
        monthSales,

      monthlyBonus:
        monthlyBonus
    },

    /*
     * Maksimal 50 transaksi
     * terbaru untuk dashboard.
     */
    transactions:
      transactions.slice(
        0,
        50
      )
  };
}


/* =========================================================
   COMMISSION SUMMARY
   ========================================================= */

function portalBuildCommissionSummary_(
  affiliateCode
) {
  const sheet =
    portalGetSheet_(
      PORTAL_COMMISSIONS_SHEET
    );

  const lastRow =
    sheet.getLastRow();

  let totalEarned = 0;
  let pending = 0;
  let approved = 0;
  let paid = 0;

  if (lastRow < 2) {
    return {
      totalEarned: 0,
      pending: 0,
      approved: 0,
      paid: 0
    };
  }

  const rows =
    sheet
      .getRange(
        2,
        1,
        lastRow - 1,
        16
      )
      .getValues();

  rows.forEach(
    function(row) {
      const code =
        String(
          row[3] || ''
        ).trim(); // D

      if (
        code !==
        affiliateCode
      ) {
        return;
      }

      const amount =
        Number(
          row[9] || 0
        ); // J

      const commissionStatus =
        String(
          row[10] || ''
        ).toUpperCase(); // K

      const payoutStatus =
        String(
          row[11] || ''
        ).toUpperCase(); // L

      totalEarned +=
        amount;

      if (
        commissionStatus ===
          'PENDING' ||
        payoutStatus ===
          'PENDING'
      ) {
        pending +=
          amount;
      }

      if (
        commissionStatus ===
          'APPROVED' &&
        payoutStatus !==
          'PAID'
      ) {
        approved +=
          amount;
      }

      if (
        payoutStatus ===
        'PAID'
      ) {
        paid +=
          amount;
      }
    }
  );

  return {
    totalEarned:
      totalEarned,

    pending:
      pending,

    approved:
      approved,

    paid:
      paid
  };
}


/* =========================================================
   CURRENT MONTH SALES
   ========================================================= */

function portalCountCurrentMonthSales_(
  transactions
) {
  const now =
    new Date();

  const year =
    now.getFullYear();

  const month =
    now.getMonth();

  return transactions
    .filter(
      function(tx) {
        if (
          !tx.receivedAt
        ) {
          return false;
        }

        const d =
          new Date(
            tx.receivedAt
          );

        return (
          !isNaN(
            d.getTime()
          ) &&
          d.getFullYear() ===
            year &&
          d.getMonth() ===
            month
        );
      }
    )
    .length;
}


/* =========================================================
   MONTHLY BONUS
   ========================================================= */

function portalCalculateMonthlyBonus_(
  sales
) {
  let earned = 0;
  let level =
    'Belum ada bonus';

  let nextTarget = 5;
  let nextBonus = 100000;

  if (sales >= 20) {
    earned = 600000;
    level =
      '20 Penjualan';

    nextTarget = null;
    nextBonus = null;

  } else if (
    sales >= 10
  ) {
    earned = 250000;
    level =
      '10 Penjualan';

    nextTarget = 20;
    nextBonus = 600000;

  } else if (
    sales >= 5
  ) {
    earned = 100000;
    level =
      '5 Penjualan';

    nextTarget = 10;
    nextBonus = 250000;
  }

  return {
    earned:
      earned,

    level:
      level,

    sales:
      sales,

    nextTarget:
      nextTarget,

    nextBonus:
      nextBonus,

    remainingToNext:
      nextTarget === null
        ? 0
        : Math.max(
            0,
            nextTarget -
            sales
          )
  };
}


/* =========================================================
   AFFILIATE LOOKUP BY EMAIL
   ========================================================= */

function portalFindApprovedAffiliateByEmail_(
  email
) {
  const sheet =
    portalGetSheet_(
      PORTAL_REGISTRY_SHEET
    );

  const lastRow =
    sheet.getLastRow();

  if (lastRow < 2) {
    return null;
  }

  const rows =
    sheet
      .getRange(
        2,
        1,
        lastRow - 1,
        39
      )
      .getValues();

  for (
    let i = 0;
    i < rows.length;
    i++
  ) {
    const r =
      rows[i];

    const rowEmail =
      portalNormalizeEmail_(
        r[4]
      ); // E

    const status =
      String(
        r[13] || ''
      )
        .trim()
        .toUpperCase(); // N

    const emailVerified =
      String(
        r[23] || ''
      )
        .trim()
        .toUpperCase(); // X

    if (
      rowEmail === email &&
      status ===
        'APPROVED' &&
      (
        emailVerified ===
          'YES' ||
        emailVerified ===
          'TRUE' ||
        emailVerified ===
          'VERIFIED'
      )
    ) {
      return {
        row:
          i + 2,

        name:
          String(
            r[1] || ''
          ),

        firstName:
          String(
            r[2] || ''
          ),

        email:
          rowEmail,

        city:
          String(
            r[6] || ''
          ),

        code:
          String(
            r[10] || ''
          ).trim(),

        slug:
          String(
            r[11] || ''
          ).trim(),

        affiliatePage:
          String(
            r[12] || ''
          ),

        status:
          status,

        checkoutUrl:
          String(
            r[14] || ''
          ),

        marketingKitUrl:
          String(
            r[28] || ''
          )
      };
    }
  }

  return null;
}

/* =========================================================
   ACTIVE SESSION LOOKUP
   ========================================================= */

function portalFindActiveSession_(
  rawToken
) {
  const sheet =
    portalGetSheet_(
      PORTAL_SESSIONS_SHEET
    );

  const lastRow =
    sheet.getLastRow();

  if (lastRow < 2) {
    return null;
  }

  const tokenHash =
    portalHash_(
      rawToken
    );

  const rows =
    sheet
      .getRange(
        2,
        1,
        lastRow - 1,
        15
      )
      .getValues();

  for (
    let i =
      rows.length - 1;
    i >= 0;
    i--
  ) {
    const r =
      rows[i];

    const storedHash =
      String(
        r[9] || ''
      );

    const expiresAt =
      portalAsDate_(
        r[10]
      );

    const status =
      String(
        r[12] || ''
      )
        .toUpperCase();

    if (
      storedHash &&
      portalSafeEqual_(
        tokenHash,
        storedHash
      ) &&
      status ===
        'ACTIVE'
    ) {
      /*
       * Session expired.
       */
      if (
        !expiresAt ||
        Date.now() >
          expiresAt.getTime()
      ) {
        portalUpdateSessionRow_(
          i + 2,
          {
            status:
              'EXPIRED',
            notes:
              'Session expired.'
          }
        );

        return null;
      }

      return {
        row:
          i + 2,

        values:
          r
      };
    }
  }

  return null;
}

/* =========================================================
   SECURITY SALT
   ========================================================= */

function portalEnsureSecuritySalt_() {
  const props =
    PropertiesService
      .getScriptProperties();

  let salt =
    props.getProperty(
      'PORTAL_SECURITY_SALT'
    );

  if (!salt) {
    salt =
      Utilities
        .getUuid()
        .replace(/-/g, '') +
      Utilities
        .getUuid()
        .replace(/-/g, '');

    props.setProperty(
      'PORTAL_SECURITY_SALT',
      salt
    );
  }

  return salt;
}


/* =========================================================
   HASH
   ========================================================= */

function portalHash_(
  value
) {
  const salt =
    portalEnsureSecuritySalt_();

  const bytes =
    Utilities.computeDigest(
      Utilities
        .DigestAlgorithm
        .SHA_256,

      String(value) +
        '|' +
        salt,

      Utilities
        .Charset
        .UTF_8
    );

  return bytes
    .map(
      function(b) {
        const n =
          b < 0
            ? b + 256
            : b;

        return (
          '0' +
          n.toString(16)
        ).slice(-2);
      }
    )
    .join('');
}


/* =========================================================
   CONSTANT-TIME STRING COMPARE
   ========================================================= */

function portalSafeEqual_(
  a,
  b
) {
  a =
    String(
      a || ''
    );

  b =
    String(
      b || ''
    );

  if (
    a.length !==
    b.length
  ) {
    return false;
  }

  let diff = 0;

  for (
    let i = 0;
    i < a.length;
    i++
  ) {
    diff |=
      a.charCodeAt(i) ^
      b.charCodeAt(i);
  }

  return diff === 0;
}

/* =========================================================
   EMAIL VALIDATION
   ========================================================= */

function portalIsValidEmail_(
  email
) {

    .test(
      String(
        email || ''
      )
    );
}


/* =========================================================
   USER AGENT
   ========================================================= */

function portalGetUserAgent_(
  e
) {
  try {
    return String(
      e &&
      e.parameter &&
      e.parameter.userAgent
        ? e.parameter.userAgent
        : ''
    );

  } catch (err) {
    return '';
  }
}


/* =========================================================
   DATE PARSER
   ========================================================= */

function portalAsDate_(
  value
) {
  if (!value) {
    return null;
  }

  if (
    Object.prototype
      .toString
      .call(value) ===
    '[object Date]'
  ) {
    return isNaN(
      value.getTime()
    )
      ? null
      : value;
  }

  const d =
    new Date(
      value
    );

  return isNaN(
    d.getTime()
  )
    ? null
    : d;
}


/* =========================================================
   DATE TO ISO
   ========================================================= */

function portalDateIso_(
  value
) {
  const d =
    portalAsDate_(
      value
    );

  return d
    ? d.toISOString()
    : '';
}


/* =========================================================
   HTML ESCAPE

function portalEscapeHtml_(
  value
) {

    value || ''
  )
    .replace(
      /&/g,
      '&amp;'
    )
    .replace(
      /</g,
      '&lt;'
    )
    .replace(
      />/g,
      '&gt;'
    )
    .replace(
      /"/g,
      '&quot;'
    )
    .replace(
      /'/g,
      '&#039;'
    );
}

/* =========================================================
   ADMIN REQUEST OTP
   ========================================================= */

function affiliateAdminRequestOtp_(payload, e) {
  const email =
    portalNormalizeEmail_(
      payload.email
    );

  /*
   * Admin portal bersifat private.
   * Email yang bukan admin resmi langsung ditolak.
   * Tidak pernah mengungkap email admin yang valid.
   */
  if (
    !portalIsValidEmail_(email) ||
    !affiliateAdminIsAllowedEmail_(email)
  ) {
    Utilities.sleep(250);

    return {
      ok: false,
      error: 'ADMIN_ACCESS_DENIED',
      message:
        'Akses ditolak. Email ini tidak memiliki izin sebagai administrator.'
    };
  }

  const lock =
    LockService.getScriptLock();

  lock.waitLock(15000);

  try {
    const recent =
      affiliateAdminFindRecentOtp_(
        email
      );

    if (
      recent &&
      recent.ageSeconds < 60
    ) {
      return {
        ok: true,
        message:
          'Kode OTP sudah dikirim ke email administrator. Periksa inbox atau folder spam.'
      };
    }

    portalEnsureSecuritySalt_();

    const otp =
      String(
        Math.floor(
          100000 +
          Math.random() * 900000
        )
      );

    const recordId =
      'ADMIN-OTP-' +
      Utilities.getUuid();

    const createdAt =
      new Date();

    const expiresAt =
      new Date(
        createdAt.getTime() +
        AFFILIATE_ADMIN_OTP_TTL_MINUTES *
        60 *
        1000
      );

    const otpHash =
      portalHash_(
        'ADMIN|' +
        email +
        '|' +
        otp +
        '|' +
        recordId
      );

    const sheet =
      portalGetSheet_(
        AFFILIATE_ADMIN_SESSIONS_SHEET
      );

    const userAgent =
      portalGetUserAgent_(e);

    sheet.appendRow([
      recordId,          // A Record ID
      createdAt,         // B Created At
      email,             // C Admin Email
      otpHash,           // D OTP Hash
      expiresAt,         // E OTP Expires At
      'PENDING',         // F OTP Status
      0,                 // G OTP Attempts
      '',                // H Session Token Hash
      '',                // I Session Expires At
      '',                // J Last Seen At
      'OTP_PENDING',     // K Status
      userAgent,         // L User Agent
      ''                 // M Notes
    ]);

    affiliateAdminSendOtpEmail_(
      email,
      otp
    );

    return {
      ok: true,
      message:
        'Kode OTP sudah dikirim ke email administrator.',
      expiresInMinutes:
        AFFILIATE_ADMIN_OTP_TTL_MINUTES
    };

  } finally {
    lock.releaseLock();
  }
}


/* =========================================================
   ADMIN VERIFY OTP
   ========================================================= */

function affiliateAdminVerifyOtp_(payload, e) {
  const email =
    portalNormalizeEmail_(
      payload.email
    );

  const otp =
    String(
      payload.otp || ''
    )
      .replace(/\D/g, '')
      .trim();

  if (
    !portalIsValidEmail_(email) ||
    !affiliateAdminIsAllowedEmail_(email) ||
    !/^\d{6}$/.test(otp)
  ) {
    return {
      ok: false,
      error: 'ADMIN_OTP_INVALID_OR_EXPIRED',
      message:
        'Kode OTP salah atau sudah kedaluwarsa.'
    };
  }

  const lock =
    LockService.getScriptLock();

  lock.waitLock(15000);

  try {
    const record =
      affiliateAdminFindLatestPendingOtp_(
        email
      );

    if (!record) {
      return {
        ok: false,
        error: 'ADMIN_OTP_INVALID_OR_EXPIRED',
        message:
          'Kode OTP salah atau sudah kedaluwarsa.'
      };
    }

    const now =
      new Date();

    const expiresAt =
      portalAsDate_(
        record.values[4]
      );

    const attempts =
      Number(
        record.values[6] || 0
      );

    if (
      !expiresAt ||
      now.getTime() >
      expiresAt.getTime()
    ) {
      affiliateAdminUpdateSessionRow_(
        record.row,
        {
          otpStatus: 'EXPIRED',
          status: 'OTP_EXPIRED',
          notes:
            'Admin OTP expired.'
        }
      );

      return {
        ok: false,
        error: 'ADMIN_OTP_INVALID_OR_EXPIRED',
        message:
          'Kode OTP salah atau sudah kedaluwarsa.'
      };
    }

    if (
      attempts >=
      AFFILIATE_ADMIN_OTP_MAX_ATTEMPTS
    ) {
      affiliateAdminUpdateSessionRow_(
        record.row,
        {
          otpStatus: 'LOCKED',
          status: 'OTP_LOCKED',
          notes:
            'Maximum admin OTP attempts reached.'
        }
      );

      return {
        ok: false,
        error: 'ADMIN_OTP_LOCKED',
        message:
          'Terlalu banyak percobaan. Minta kode OTP baru.'
      };
    }

    const expectedHash =
      portalHash_(
        'ADMIN|' +
        email +
        '|' +
        otp +
        '|' +
        record.values[0]
      );

    const storedHash =
      String(
        record.values[3] || ''
      );

    if (
      !portalSafeEqual_(
        expectedHash,
        storedHash
      )
    ) {
      const newAttempts =
        attempts + 1;

      affiliateAdminUpdateSessionRow_(
        record.row,
        {
          otpAttempts: newAttempts,
          notes:
            'Invalid admin OTP attempt.'
        }
      );

      return {
        ok: false,
        error: 'ADMIN_OTP_INVALID_OR_EXPIRED',
        message:
          'Kode OTP salah atau sudah kedaluwarsa.',
        attemptsRemaining:
          Math.max(
            0,
            AFFILIATE_ADMIN_OTP_MAX_ATTEMPTS -
            newAttempts
          )
      };
    }

    const sessionToken =
      Utilities
        .getUuid()
        .replace(/-/g, '') +
      Utilities
        .getUuid()
        .replace(/-/g, '');

    const sessionHash =
      portalHash_(
        'ADMIN_SESSION|' +
        sessionToken
      );

    const sessionExpiresAt =
      new Date(
        now.getTime() +
        AFFILIATE_ADMIN_SESSION_DAYS *
        24 *
        60 *
        60 *
        1000
      );

    affiliateAdminUpdateSessionRow_(
      record.row,
      {
        otpStatus: 'VERIFIED',
        sessionTokenHash: sessionHash,
        sessionExpiresAt: sessionExpiresAt,
        lastSeenAt: now,
        status: 'ACTIVE',
        notes:
          'Admin portal login successful.'
      }
    );

    return {
      ok: true,
      authenticated: true,
      sessionToken: sessionToken,
      expiresAt:
        sessionExpiresAt.toISOString(),
      admin: {
        email: email
      }
    };

  } finally {
    lock.releaseLock();
  }
}


/* =========================================================
   ADMIN DASHBOARD
   ========================================================= */

function affiliateAdminDashboard_(payload, e) {
  const token =
    String(
      payload.sessionToken ||
      payload.session_token ||
      ''
    ).trim();

  if (!token) {
    return {
      ok: false,
      authenticated: false,
      error: 'ADMIN_SESSION_REQUIRED'
    };
  }

  const session =
    affiliateAdminFindActiveSession_(
      token
    );

  if (!session) {
    return {
      ok: false,
      authenticated: false,
      error:
        'ADMIN_SESSION_INVALID_OR_EXPIRED'
    };
  }

  const email =
    portalNormalizeEmail_(
      session.values[2]
    );

  if (
    !affiliateAdminIsAllowedEmail_(
      email
    )
  ) {
    affiliateAdminUpdateSessionRow_(
      session.row,
      {
        status: 'REVOKED',
        notes:
          'Admin email no longer allowed.'
      }
    );

    return {
      ok: false,
      authenticated: false,
      error: 'ADMIN_ACCESS_REVOKED'
    };
  }

  const now =
    new Date();

  affiliateAdminUpdateSessionRow_(
    session.row,
    {
      lastSeenAt: now
    }
  );

/* =========================================================
   ADMIN RESEND AFFILIATE EMAIL
   VERIFICATION / ACTIVATION
   ========================================================= */

function affiliateAdminResendAffiliateEmail_(payload, e) {
  const token =
    String(
      payload.sessionToken ||
      payload.session_token ||
      ''
    ).trim();

  if (!token) {
    return {
      ok: false,
      authenticated: false,
      error: 'ADMIN_SESSION_REQUIRED',
      message:
        'Sesi admin diperlukan.'
    };
  }

  const session =
    affiliateAdminFindActiveSession_(
      token
    );

  if (!session) {
    return {
      ok: false,
      authenticated: false,
      error:
        'ADMIN_SESSION_INVALID_OR_EXPIRED',
      message:
        'Sesi admin tidak valid atau sudah berakhir.'
    };
  }

  const adminEmail =
    portalNormalizeEmail_(
      session.values[2]
    );

  if (
    !affiliateAdminIsAllowedEmail_(
      adminEmail
    )
  ) {
    return {
      ok: false,
      authenticated: false,
      error: 'ADMIN_ACCESS_REVOKED',
      message:
        'Akses admin tidak diizinkan.'
    };
  }

  const referralCode =
    String(
      payload.referralCode ||
      payload.affiliateCode ||
      ''
    )
      .trim()
      .toUpperCase();

  const emailType =
    String(
      payload.emailType ||
      payload.type ||
      ''
    )
      .trim()
      .toUpperCase();

  if (!referralCode) {
    return {
      ok: false,
      error:
        'AFFILIATE_CODE_REQUIRED',
      message:
        'Kode affiliate wajib diisi.'
    };
  }

  if (
    emailType !==
      'VERIFICATION' &&
    emailType !==
      'ACTIVATION' &&
    emailType !==
      'PORTAL'
  ) {
    return {
      ok: false,
      error:
        'INVALID_EMAIL_TYPE',
      message:
        'Jenis email resend tidak valid.'
    };
  }

  const lock =
    LockService.getScriptLock();

  lock.waitLock(15000);

  try {
    const registry =
      portalGetSheet_(
        PORTAL_REGISTRY_SHEET
      );

    const lastRow =
      registry.getLastRow();

    if (lastRow < 2) {
      return {
        ok: false,
        error:
          'AFFILIATE_NOT_FOUND',
        message:
          'Affiliate tidak ditemukan.'
      };
    }

    const rows =
      registry
        .getRange(
          2,
          1,
          lastRow - 1,
          29
        )
        .getValues();

    let targetRow = 0;

    for (
      let i = 0;
      i < rows.length;
      i++
    ) {
      const code =
        String(
          rows[i][10] || ''
        )
          .trim()
          .toUpperCase();

      if (
        code ===
        referralCode
      ) {
        targetRow =
          i + 2;
        break;
      }
    }

    if (!targetRow) {
      return {
        ok: false,
        error:
          'AFFILIATE_NOT_FOUND',
        message:
          'Affiliate tidak ditemukan.'
      };
    }

    const data =
      getAffiliateRowData_(
        registry,
        targetRow
      );

    if (
      !isValidEmail_(
        data.email
      )
    ) {
      return {
        ok: false,
        error:
          'INVALID_AFFILIATE_EMAIL',
        message:
          'Email affiliate tidak valid.'
      };
    }

    const now =
      new Date();

    let actionName = '';
    let messageText = '';

    if (
      emailType ===
      'VERIFICATION'
    ) {
      if (
        data.emailVerified ===
        'YES'
      ) {
        return {
          ok: false,
          error:
            'EMAIL_ALREADY_VERIFIED',
          message:
            'Email affiliate sudah terverifikasi. Gunakan Resend Aktivasi bila diperlukan.'
        };
      }

      sendVerificationEmailForRow_(
        registry,
        targetRow
      );

      actionName =
        'VERIFICATION_EMAIL_RESENT';

      messageText =
        'Email verifikasi berhasil dikirim ulang ke ' +
        data.email +
        '.';
    }

    if (
      emailType ===
      'ACTIVATION'
    ) {
      if (
        data.status !==
        'APPROVED'
      ) {
        return {
          ok: false,
          error:
            'AFFILIATE_NOT_APPROVED',
          message:
            'Email aktivasi hanya dapat dikirim untuk affiliate berstatus APPROVED.'
        };
      }

      if (
        data.emailVerified !==
        'YES'
      ) {
        return {
          ok: false,
          error:
            'EMAIL_NOT_VERIFIED',
          message:
            'Email affiliate belum terverifikasi.'
        };
      }

      sendApprovalEmailForRow_(
        registry,
        targetRow,
        true
      );

      actionName =
        'ACTIVATION_EMAIL_RESENT';

      messageText =
        'Email aktivasi berhasil dikirim ulang ke ' +
        data.email +
        '.';
    }

    if (
      emailType ===
      'PORTAL'
    ) {
      if (
        data.status !==
        'APPROVED'
      ) {
        return {
          ok: false,
          error:
            'AFFILIATE_NOT_APPROVED',
          message:
            'Link Portal Affiliate hanya dapat dikirim untuk affiliate berstatus APPROVED.'
        };
      }

      if (
        data.emailVerified !==
        'YES'
      ) {
        return {
          ok: false,
          error:
            'EMAIL_NOT_VERIFIED',
          message:
            'Email affiliate belum terverifikasi.'
        };
      }

      affiliateAdminSendPortalAccessEmail_(
        data
      );

      actionName =
        'PORTAL_ACCESS_EMAIL_RESENT';

      messageText =
        'Link Portal Affiliate berhasil dikirim ulang ke ' +
        data.email +
        '.';
    }

    const actionsSheet =
      portalGetSheet_(
        AFFILIATE_ADMIN_ACTIONS_SHEET
      );

    actionsSheet.appendRow([
      now,
      'ACT-' +
        Utilities.getUuid(),
      adminEmail,
      actionName,
      referralCode,
      data.name,
      '',
      '',
      'Resend ' +
        emailType +
        ' email to ' +
        data.email,
      portalGetUserAgent_(
        e
      )
    ]);

    affiliateAdminUpdateSessionRow_(
      session.row,
      {
        lastSeenAt:
          now
      }
    );

    return {
      ok: true,
      authenticated: true,
      emailType:
        emailType,
      referralCode:
        referralCode,
      email:
        data.email,
      message:
        messageText
    };

  } finally {
    lock.releaseLock();
  }
}

/* =========================================================
   ADMIN UPDATE COMMISSION STATUS
   PENDING -> APPROVED -> PAID
   ========================================================= */

function affiliateAdminUpdateCommissionStatus_(payload, e) {
  const token =
    String(
      payload.sessionToken ||
      payload.session_token ||
      ''
    ).trim();

  if (!token) {
    return {
      ok: false,
      authenticated: false,
      error: 'ADMIN_SESSION_REQUIRED',
      message:
        'Sesi admin diperlukan.'
    };
  }

  const session =
    affiliateAdminFindActiveSession_(
      token
    );

  if (!session) {
    return {
      ok: false,
      authenticated: false,
      error:
        'ADMIN_SESSION_INVALID_OR_EXPIRED',
      message:
        'Sesi admin tidak valid atau sudah berakhir.'
    };
  }

  const adminEmail =
    portalNormalizeEmail_(
      session.values[2]
    );

  if (
    !affiliateAdminIsAllowedEmail_(
      adminEmail
    )
  ) {
    return {
      ok: false,
      authenticated: false,
      error: 'ADMIN_ACCESS_REVOKED',
      message:
        'Akses admin tidak diizinkan.'
    };
  }

  const commissionId =
    String(
      payload.commissionId ||
      payload.commission_id ||
      ''
    ).trim();

  const newStatus =
    String(
      payload.newStatus ||
      payload.status ||
      ''
    )
      .trim()
      .toUpperCase();

  const payoutId =
    String(
      payload.payoutId ||
      payload.payout_id ||
      ''
    )
      .trim()
      .slice(
        0,
        200
      );

  if (!commissionId) {
    return {
      ok: false,
      error:
        'COMMISSION_ID_REQUIRED',
      message:
        'Commission ID wajib diisi.'
    };
  }

  if (
    newStatus !==
      'APPROVED' &&
    newStatus !==
      'PAID'
  ) {
    return {
      ok: false,
      error:
        'INVALID_COMMISSION_STATUS',
      message:
        'Status komisi tidak valid.'
    };
  }

  if (
    newStatus ===
      'PAID' &&
    !payoutId
  ) {
    return {
      ok: false,
      error:
        'PAYOUT_ID_REQUIRED',
      message:
        'Payout ID wajib diisi sebelum komisi ditandai dibayar.'
    };
  }

  const lock =
    LockService.getScriptLock();

  lock.waitLock(15000);

  try {
    const sheet =
      portalGetSheet_(
        AFFILIATE_COMMISSIONS_SHEET
      );

    const lastRow =
      sheet.getLastRow();

    if (lastRow < 2) {
      return {
        ok: false,
        error:
          'COMMISSION_NOT_FOUND',
        message:
          'Data komisi tidak ditemukan.'
      };
    }

    const rows =
      sheet
        .getRange(
          2,
          1,
          lastRow - 1,
          16
        )
        .getValues();

    let targetRow = 0;
    let values = null;

    for (
      let i = 0;
      i < rows.length;
      i++
    ) {
      if (
        String(
          rows[i][0] || ''
        ).trim() ===
        commissionId
      ) {
        targetRow =
          i + 2;
        values =
          rows[i];
        break;
      }
    }

    if (
      !targetRow ||
      !values
    ) {
      return {
        ok: false,
        error:
          'COMMISSION_NOT_FOUND',
        message:
          'Data komisi tidak ditemukan.'
      };
    }

    const affiliateCode =
      String(
        values[3] || ''
      ).trim();

    const affiliateName =
      String(
        values[5] || ''
      ).trim();

    const commissionStatus =
      String(
        values[10] || ''
      )
        .trim()
        .toUpperCase();

    const payoutStatus =
      String(
        values[11] || ''
      )
        .trim()
        .toUpperCase();

    const now =
      new Date();

    let oldState =
      commissionStatus +
      '/' +
      payoutStatus;

    let newState = '';

    if (
      newStatus ===
      'APPROVED'
    ) {
      if (
        payoutStatus ===
        'PAID'
      ) {
        return {
          ok: false,
          error:
            'COMMISSION_ALREADY_PAID',
          message:
            'Komisi sudah dibayar dan tidak dapat diubah kembali.'
        };
      }

      if (
        commissionStatus ===
        'APPROVED'
      ) {
        return {
          ok: true,
          authenticated: true,
          unchanged: true,
          message:
            'Komisi sudah APPROVED.'
        };
      }

      if (
        commissionStatus !==
        'PENDING'
      ) {
        return {
          ok: false,
          error:
            'INVALID_COMMISSION_TRANSITION',
          message:
            'Hanya komisi PENDING yang dapat di-approve.'
        };
      }

      sheet
        .getRange(
          targetRow,
          11
        )
        .setValue(
          'APPROVED'
        );

      if (!payoutStatus) {
        sheet
          .getRange(
            targetRow,
            12
          )
          .setValue(
            'UNPAID'
          );
      }

      sheet
        .getRange(
          targetRow,
          13
        )
        .setValue(
          now
        );

      newState =
        'APPROVED/' +
        (
          payoutStatus ||
          'UNPAID'
        );
    }

    if (
      newStatus ===
      'PAID'
    ) {
      if (
        payoutStatus ===
        'PAID'
      ) {
        return {
          ok: true,
          authenticated: true,
          unchanged: true,
          message:
            'Komisi sudah berstatus PAID.'
        };
      }

      if (
        commissionStatus !==
        'APPROVED'
      ) {
        return {
          ok: false,
          error:
            'COMMISSION_NOT_APPROVED',
          message:
            'Komisi harus APPROVED sebelum dapat ditandai PAID.'
        };
      }

      sheet
        .getRange(
          targetRow,
          12
        )
        .setValue(
          'PAID'
        );

      sheet
        .getRange(
          targetRow,
          14
        )
        .setValue(
          now
        );

      sheet
        .getRange(
          targetRow,
          15
        )
        .setValue(
          payoutId
        );

      newState =
        'APPROVED/PAID';
    }

    const actionsSheet =
      portalGetSheet_(
        AFFILIATE_ADMIN_ACTIONS_SHEET
      );

    actionsSheet.appendRow([
      now,
      'ACT-' +
        Utilities.getUuid(),
      adminEmail,
      newStatus ===
        'PAID'
        ? 'COMMISSION_PAID'
        : 'COMMISSION_APPROVED',
      affiliateCode,
      affiliateName,
      oldState,
      newState,
      newStatus ===
        'PAID'
        ? 'Payout ID: ' +
          payoutId
        : '',
      portalGetUserAgent_(
        e
      )
    ]);

    /*
     * Sinkronkan ringkasan Registry
     * langsung setelah perubahan payout.
     */
    rebuildAffiliateFinancialSummary_();

    affiliateAdminUpdateSessionRow_(
      session.row,
      {
        lastSeenAt:
          now
      }
    );

    return {
      ok: true,
      authenticated: true,
      message:
        newStatus ===
          'PAID'
          ? 'Komisi berhasil ditandai PAID.'
          : 'Komisi berhasil di-approve.',
      commission: {
        id:
          commissionId,
        affiliateCode:
          affiliateCode,
        affiliateName:
          affiliateName,
        status:
          newStatus ===
            'PAID'
            ? 'APPROVED'
            : 'APPROVED',
        payoutStatus:
          newStatus ===
            'PAID'
            ? 'PAID'
            : (
                payoutStatus ||
                'UNPAID'
              ),
        payoutId:
          newStatus ===
            'PAID'
            ? payoutId
            : String(
                values[14] || ''
              )
      }
    };

  } finally {
    lock.releaseLock();
  }
}

          'EMAIL_NOT_VERIFIED',
        message:
          'Affiliate belum dapat di-approve karena email belum terverifikasi.'
      };
    }

    if (
      oldStatus ===
      newStatus
    ) {
      return {
        ok: true,
        authenticated: true,
        unchanged: true,
        message:
          'Status affiliate sudah ' +
          newStatus +
          '.',
        affiliate: {
          code:
            referralCode,
          name:
            affiliateName,
          status:
            newStatus
        }
      };
    }

    const now =
      new Date();

    /*
     * N = Status
     */
    registry
      .getRange(
        targetRow,
        14
      )
      .setValue(
        newStatus
      );

    /*
     * P = Tanggal Approval
     */
    if (
      newStatus ===
      'APPROVED'
    ) {
      registry
        .getRange(
          targetRow,
          16
        )
        .setValue(
          now
        );
    }

    /*
     * Q = Catatan Admin
     * Tambahkan catatan tanpa menghapus
     * histori catatan sebelumnya.
     */
    const existingNotes =
      String(
        targetValues[16] || ''
      ).trim();

    const actionNote =
      '[' +
      Utilities.formatDate(
        now,
        Session.getScriptTimeZone() ||
          'Asia/Jakarta',
        'yyyy-MM-dd HH:mm:ss'
      ) +
      '] ' +
      adminEmail +
      ': ' +
      oldStatus +
      ' → ' +
      newStatus +
      (
        adminNotes
          ? ' — ' +
            adminNotes
          : ''
      );

    const mergedNotes =
      existingNotes
        ? existingNotes +
          '\n' +
          actionNote
        : actionNote;

    registry
      .getRange(
        targetRow,
        17
      )
      .setValue(
        mergedNotes
      );

    /*
     * Jika REJECTED / SUSPENDED,
     * revoke seluruh session affiliate
     * yang masih ACTIVE.
     */
    if (
      newStatus ===
        'REJECTED' ||
      newStatus ===
        'SUSPENDED'
    ) {
      affiliateAdminRevokeAffiliateSessions_(
        referralCode,
        newStatus
      );
    }

    /*
     * Audit log wajib.
     */
    const actionsSheet =
      portalGetSheet_(
        AFFILIATE_ADMIN_ACTIONS_SHEET
      );

    actionsSheet.appendRow([
      now,
      'ACT-' +
        Utilities.getUuid(),
      adminEmail,
      'STATUS_CHANGE',
      referralCode,
      affiliateName,
      oldStatus,
      newStatus,
      adminNotes,
      portalGetUserAgent_(
        e
      )
    ]);

    /*
     * Jika project lama sudah memiliki
     * helper email approval,
     * pakai tanpa membuat dependency wajib.
     */
    let approvalEmailSent = false;
    let approvalEmailError = '';

    if (
      newStatus ===
        'APPROVED' &&
      typeof sendApprovalEmailForRow_ ===
        'function'
    ) {
      try {
        sendApprovalEmailForRow_(
          registry,
          targetRow
        );

        approvalEmailSent =
          true;

      } catch (mailErr) {
        approvalEmailError =
          String(
            mailErr &&
            mailErr.message
              ? mailErr.message
              : mailErr
          );
      }
    }

    affiliateAdminUpdateSessionRow_(
      session.row,
      {
        lastSeenAt:
          now
      }
    );

    return {
      ok: true,
      authenticated: true,
      message:
        affiliateName +
        ' berhasil diubah menjadi ' +
        newStatus +
        '.',
      affiliate: {
        code:
          referralCode,
        name:
          affiliateName,
        email:
          affiliateEmail,
        oldStatus:
          oldStatus,
        status:
          newStatus
      },
      approvalEmailSent:
        approvalEmailSent,
      approvalEmailError:
        approvalEmailError
    };

  } finally {
    lock.releaseLock();
  }
}


/* =========================================================
   REVOKE AFFILIATE SESSIONS
   ========================================================= */

function affiliateAdminRevokeAffiliateSessions_(
  referralCode,
  reasonStatus
) {
  const sheet =
    portalGetSheet_(
      PORTAL_SESSIONS_SHEET
    );

  const lastRow =
    sheet.getLastRow();

  if (lastRow < 2) {
    return 0;
  }

  const rows =
    sheet
      .getRange(
        2,
        1,
        lastRow - 1,
        15
      )
      .getValues();

  let revoked = 0;

  rows.forEach(
    function(row, index) {
      const code =
        String(
          row[3] || ''
        )
          .trim()
          .toUpperCase();

      const status =
        String(
          row[12] || ''
        )
          .trim()
          .toUpperCase();

      if (
        code ===
          referralCode &&
        status ===
          'ACTIVE'
      ) {
        const sheetRow =
          index + 2;

        sheet
          .getRange(
            sheetRow,
            13
          )
          .setValue(
            'REVOKED'
          );

        sheet
          .getRange(
            sheetRow,
            15
          )
          .setValue(
            'Session revoked by admin because affiliate status changed to ' +
            reasonStatus +
            '.'
          );

        revoked++;
      }
    }
  );

  return revoked;
}

/* =========================================================
   PANG LIMA LYNK AFFILIATE LINK VALIDATION
   ========================================================= */

function parsePanglimaAffiliateLynkUrl_(value) {
  const raw =
    clean_(
      value
    );

  if (!raw) {
    return {
      valid: false,
      error:
        'LYNK_AFFILIATE_URL_REQUIRED',
      message:
        'Link Affiliate Lynk.id Paket Panglima UMKM wajib diisi.'
    };
  }

  /*
   * Format produksi yang digunakan:
   * https://lynk.id/{username-affiliate}/9rov84wv3grz
   *
   * Query/hash akan dibuang saat URL disimpan.
   */
  const match =
    raw.match(
      /^https:\/\/(?:www\.)?lynk\.id\/([A-Za-z0-9._-]{2,80})\/([A-Za-z0-9_-]{6,80})\/?(?:[?#].*)?$/i
    );

  if (!match) {
    return {
      valid: false,
      error:
        'LYNK_AFFILIATE_URL_FORMAT_INVALID',
      message:
        'Link Affiliate Lynk.id belum valid. Gunakan link lengkap dengan format https://lynk.id/usernameanda/9rov84wv3grz.'
    };
  }

  const username =
    clean_(
      match[1]
    );

  const productId =
    clean_(
      match[2]
    );

  if (
    productId !==
    PANGLIMA_LYNK_PRODUCT_ID
  ) {
    return {
      valid: false,
      error:
        'LYNK_PRODUCT_ID_INVALID',
      message:
        'Link tersebut bukan link affiliate untuk Paket Panglima UMKM.'
    };
  }

  if (
    username.toLowerCase() ===
    PANGLIMA_LYNK_MERCHANT_USERNAME.toLowerCase()
  ) {
    return {
      valid: false,
      error:
        'LYNK_MERCHANT_LINK_NOT_ALLOWED',
      message:
        'Link merchant utama tidak dapat digunakan. Masukkan link affiliate Lynk.id pribadi Anda untuk Paket Panglima UMKM.'
    };
  }

  return {
    valid: true,
    username:
      username,
    productId:
      productId,
    canonicalUrl:
      'https://lynk.id/' +
      username +
      '/' +
      PANGLIMA_LYNK_PRODUCT_ID
  };
}


function doPost(e) {
  try {
    /*
     * 1. BACA SEMUA JENIS PAYLOAD
     */
    const p =
      (e && e.parameter)
        ? e.parameter
        : {};

    const rawPostBody =
      e &&
      e.postData &&
      e.postData.contents
        ? e.postData.contents
        : '';

    const contentType =
      e &&
      e.postData &&
      e.postData.type
        ? String(e.postData.type)
        : '';

    let rawJson = {};
    let rawForm = {};

    /*
     * Coba JSON.
     */
    if (rawPostBody) {
      try {
        rawJson =
          JSON.parse(
            rawPostBody
          ) || {};
      } catch (jsonError) {
        rawJson = {};
      }
    }

    /*
     * Fallback parser application/x-www-form-urlencoded.
     */
    if (
      rawPostBody &&
      (
        contentType
          .toLowerCase()
          .indexOf(
            'application/x-www-form-urlencoded'
          ) >= 0 ||
        rawPostBody
          .indexOf('=') >= 0
      )
    ) {
      try {
        rawPostBody
          .split('&')
          .forEach(function(pair) {
            if (!pair) return;

            const parts =
              pair.split('=');

            const rawKey =
              parts.shift() || '';

            const rawValue =
              parts.join('=');

            const key =
              decodeURIComponent(
                rawKey.replace(
                  /\+/g,
                  ' '
                )
              );

            const value =
              decodeURIComponent(
                String(
                  rawValue || ''
                ).replace(
                  /\+/g,
                  ' '
                )
              );

            if (key) {
              rawForm[key] =
                value;
            }
          });
      } catch (formError) {
        rawForm = {};
      }
    }

    /*
     * Prioritas:
     * e.parameter > parsed form > JSON.
     */
    const payload =
      Object.assign(
        {},
        rawJson,
        rawForm,
        p
      );

    const action =
      clean_(
        payload.action
      ).toLowerCase();


    /*
     * 2. AFFILIATE PORTAL
     * WAJIB diproses sebelum form pendaftaran.
     */
    if (
      action ===
        'portal_request_otp' ||
      action ===
        'portal_verify_otp' ||
      action ===
        'portal_dashboard' ||
      action ===
        'portal_logout'
    ) {
      return json_(
        portalHandleAction_(
          action,
          payload,
          e
        )
      );
    }

    /*
     * STEP 10C — ADMIN AFFILIATE PORTAL
     */
    if (
      action ===
        'admin_request_otp' ||
      action ===
        'admin_verify_otp' ||
      action ===
        'admin_dashboard' ||
      action ===
        'admin_update_affiliate_status' ||
      action ===
        'admin_resend_affiliate_email' ||
      action ===
        'admin_logout'
    ) {
      return json_(
        affiliateAdminHandleAction_(
          action,
          payload,
          e
        )
      );
    }


    /*
     * VALIDASI STRUKTUR LINK LYNK AFFILIATE
     * Live-check dilakukan oleh Cloudflare Worker.
     */
    if (
      action ===
      'validate_lynk_affiliate'
    ) {
      const validation =
        parsePanglimaAffiliateLynkUrl_(
          payload.lynkAffiliateUrl ||
          payload.checkoutUrl ||
          payload.lynkUrl ||
          ''
        );

      return json_(
        validation
      );
    }


    /*
     * 3. AFFILIATE EVENT TRACKING
     */
    if (
      action ===
      'affiliate_event'
    ) {
      return handleAffiliateEvent_(
        payload
      );
    }


    /*
     * 4. LYNK PRODUCTION WEBHOOK
     */
    const jsonWebhookEvent =
      clean_(
        rawJson.event ||
        rawJson.type ||
        rawJson.eventType ||
        rawJson.event_type ||
        (
          rawJson.message &&
          rawJson.message.action
        ) ||
        (
          rawJson.data &&
          rawJson.data.event
        ) ||
        (
          rawJson.data &&
          rawJson.data.message &&
          rawJson.data.message.action
        )
      ).toLowerCase();

    const formWebhookEvent =
      clean_(
        payload.event ||
        payload.type ||
        payload.eventType ||
        payload.event_type
      ).toLowerCase();

    const webhookEvent =
      jsonWebhookEvent ||
      formWebhookEvent;

    if (
      action ===
        'payment_webhook' ||
      webhookEvent ===
        'payment.received' ||
      webhookEvent ===
        'payment_received'
    ) {
      return handleStep9CProductionWebhook_(
        e,
        payload
      );
    }


    /*
     * 5. FORM PENDAFTARAN AFFILIATE
     */
    const name =
      clean_(
        payload.name
      );

    const whatsapp =
      normalizePhone_(
        payload.whatsapp
      );

    const email =
      clean_(
        payload.email
      ).toLowerCase();

    const social =
      clean_(
        payload.social
      );

    const city =
      clean_(
        payload.city
      );

    const profession =
      clean_(
        payload.profession
      );

    const reason =
      clean_(
        payload.reason
      );

    const agree =
      clean_(
        payload.agree
      ) === 'YA'
        ? 'YA'
        : 'TIDAK';

    const sourcePage =
      clean_(
        payload.sourcePage
      );

    const userAgent =
      clean_(
        payload.userAgent
      );

    const lynkAffiliateUrlRaw =
      clean_(
        payload.lynkAffiliateUrl ||
        payload.checkoutUrl ||
        payload.lynkUrl
      );

    const lynkValidationStatus =
      clean_(
        payload.lynkValidationStatus
      ).toUpperCase();

    const lynkValidation =
      parsePanglimaAffiliateLynkUrl_(
        lynkAffiliateUrlRaw
      );

    const lynkAffiliateUrl =
      lynkValidation.valid
        ? lynkValidation.canonicalUrl
        : '';


    if (
      !name ||
      !whatsapp ||
      !email ||
      !city ||
      !profession ||
      !reason ||
      !lynkAffiliateUrlRaw ||
      agree !== 'YA'
    ) {
      return json_({
        ok: false,
        message:
          'Data wajib belum lengkap.'
      });
    }


    if (
      !lynkValidation.valid
    ) {
      return json_({
        ok: false,
        error:
          lynkValidation.error ||
          'LYNK_AFFILIATE_URL_INVALID',
        message:
          lynkValidation.message ||
          'Link Affiliate Lynk.id Paket Panglima UMKM belum valid.'
      });
    }


    if (
      lynkValidationStatus !==
      'VALID'
    ) {
      return json_({
        ok: false,
        error:
          'LYNK_LIVE_VALIDATION_REQUIRED',
        message:
          'Link affiliate Lynk.id harus divalidasi terlebih dahulu sebelum pendaftaran dapat dikirim.'
      });
    }


    if (
      !isValidEmail_(
        email
      )
    ) {
      return json_({
        ok: false,
        message:
          'Alamat email tidak valid.'
      });
    }


    /*
     * 6. BUAT KODE REFERRAL
     */
    const firstName =
      getFirstName_(
        name
      );

    const referralCode =
      createReferralCode_(
        firstName,
        whatsapp
      );

    const ss =
      SpreadsheetApp.openById(
        SPREADSHEET_ID
      );

    const sheet =
      ss.getSheetByName(
        SHEET_NAME
      );

    if (!sheet) {
      throw new Error(
        'Sheet tidak ditemukan: ' +
        SHEET_NAME
      );
    }

    ensureStep6AHeaders_(
      sheet
    );

    const lastRow =
      sheet.getLastRow();

    const existing =
      lastRow > 1
        ? sheet
            .getRange(
              2,
              1,
              lastRow - 1,
              29
            )
            .getDisplayValues()
        : [];

    const codeExists =
      existing.some(
        function(r) {
          return (
            clean_(
              r[10]
            ).toUpperCase() ===
            referralCode
          );
        }
      );

    const phoneExists =
      existing.some(
        function(r) {
          return (
            normalizePhone_(
              r[3]
            ) ===
            whatsapp
          );
        }
      );

    const emailExists =
      existing.some(
        function(r) {
          return (
            clean_(
              r[4]
            ).toLowerCase() ===
            email
          );
        }
      );


    /*
     * 7. BUAT SLUG
     */
    const slugBase =
      slugify_(
        firstName
      );

    let slug =
      slugBase;

    const usedSlugs =
      new Set(
        existing
          .map(
            function(r) {
              return clean_(
                r[11]
              ).toLowerCase();
            }
          )
          .filter(
            Boolean
          )
      );

    if (
      usedSlugs.has(
        slug
      )
    ) {
      slug =
        slugBase +
        '-' +
        whatsapp.slice(
          -3
        );
    }

    let suffix = 2;

    while (
      usedSlugs.has(
        slug
      )
    ) {
      slug =
        slugBase +
        '-' +
        whatsapp.slice(
          -3
        ) +
        '-' +
        suffix;

      suffix++;
    }


    /*
     * 8. DUPLICATE CHECK
     */
    const duplicate =
      codeExists ||
      phoneExists ||
      emailExists;

    const duplicateNotes =
      [
        codeExists
          ? 'KODE'
          : '',
        phoneExists
          ? 'WHATSAPP'
          : '',
        emailExists
          ? 'EMAIL'
          : ''
      ]
        .filter(
          Boolean
        )
        .join(
          ', '
        );

    const status =
      duplicate
        ? 'REVIEW_DUPLICATE'
        : 'PENDING_EMAIL';

    const affiliatePage =
      SITE_URL +
      '/?affiliate=' +
      encodeURIComponent(
        slug
      );

    const verificationToken =
      duplicate
        ? ''
        : createVerificationToken_();


    /*
     * 9. SIMPAN KE AFFILIATE REGISTRY
     */
    sheet.appendRow([
      new Date(),                         // A
      name,                               // B
      firstName,                          // C
      whatsapp,                           // D
      email,                              // E
      social,                             // F
      city,                               // G
      profession,                         // H
      reason,                             // I
      agree,                              // J
      referralCode,                       // K
      slug,                               // L
      affiliatePage,                      // M
      status,                             // N
      lynkAffiliateUrl,                    // O Link Lynk Affiliate
      '',                                 // P
      duplicate
        ? 'Periksa duplikat: ' +
          duplicateNotes
        : '',                             // Q
      sourcePage,                         // R
      userAgent,                          // S
      duplicate
        ? 'DUPLICATE: ' +
          duplicateNotes
        : 'OK',                           // T
      '',                                 // U
      '',                                 // V
      '',                                 // W
      'NO',                               // X
      verificationToken,                  // Y
      '',                                 // Z
      '',                                 // AA
      '',                                 // AB
      SITE_URL +
        '/?affiliate=' +
        encodeURIComponent(
          slug
        ) +
        '&kit=1'                          // AC
    ]);

    const row =
      sheet.getLastRow();


    /*
     * 10. EMAIL VERIFICATION
     */
    if (!duplicate) {
      try {
        sendVerificationEmailForRow_(
          sheet,
          row
        );
      } catch (err) {
        sheet
          .getRange(
            row,
            14
          )
          .setValue(
            'EMAIL_ERROR'
          );

        sheet
          .getRange(
            row,
            28
          )
          .setValue(
            String(
              err &&
              err.message
                ? err.message
                : err
            )
          );
      }
    }


    return json_({
      ok: true,
      status:
        status,
      referralCode:
        referralCode,
      slug:
        slug,
      affiliatePage:
        affiliatePage,
      lynkAffiliateUrl:
        lynkAffiliateUrl,
      duplicate:
        duplicate,
      verificationRequired:
        !duplicate
    });


  } catch (err) {
    console.error(
      err
    );

    return json_({
      ok: false,
      message:
        String(
          err &&
          err.message
            ? err.message
            : err
        )
    });
  }
}

  return String(value || '').trim();
}

  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

   Sumber basis: Step 9C1 Diagnostic Fix.

   doPost(e) TIDAK diambil dari versi lama.
   Routing doPost(e) tetap menggunakan versi Portal/Admin
   terbaru yang sudah berhasil dipakai.
   ========================================================= */

function doGet(e) {
  try {
    const p = (e && e.parameter) || {};
    const action = clean_(p.action || 'health').toLowerCase();
    const callback = clean_(p.callback);

    if (action === 'verify_json') {
      return json_(
        verifyAffiliateEmailData_(
          clean_(p.token)
        )
      );
    }

    if (action === 'verify') {
      return verifyAffiliateEmail_(clean_(p.token));
    }

    if (action === 'registry') {
      return jsonOrJsonp_(getApprovedAffiliateRegistry_(), callback);
    }

    if (action === 'webhook_health') {
      const suppliedKey = clean_(p.key);
      const expectedKey = getStep9CWebhookSecret_();

      if (!expectedKey || suppliedKey !== expectedKey) {
        return jsonOrJsonp_({
          ok: false,
          service: 'Panglima Lynk Webhook',
          error: 'UNAUTHORIZED',
        }, callback);
      }

      return jsonOrJsonp_({
        ok: true,
        service: 'Panglima Lynk Webhook',
        version: 'step9c1-diagnostic-fix',
        status: 'READY',
        event: 'payment.received',
      }, callback);
    }

    return jsonOrJsonp_({
      ok: true,
      service: 'Panglima Affiliate Registry',
      version: 'step9c1-diagnostic-fix',
    }, callback);
  } catch (err) {
    return jsonOrJsonp_({
      ok: false,
      message: String(err && err.message ? err.message : err),
    }, clean_((e && e.parameter && e.parameter.callback) || ''));
  }
}

   Event types:
   PAGE_VIEW
   CHECKOUT_CLICK

   Data disimpan ke tab "Affiliate Events".
   Ringkasan otomatis disimpan di Affiliate Registry:
   AD = Total Page Views
   AE = Total Checkout Clicks
   AF = Last Affiliate Activity
   ========================================================= */


function handleAffiliateEvent_(p) {
  const eventType = clean_(p.eventType).toUpperCase();
  if (!['PAGE_VIEW', 'CHECKOUT_CLICK'].includes(eventType)) {
    return json_({ ok: false, message: 'Event affiliate tidak valid.' });
  }

  const referralCode = clean_(p.referralCode).toUpperCase();
  const slug = clean_(p.slug).toLowerCase();

  if (!referralCode && !slug) {
    return json_({ ok: false, message: 'Kode referral / slug kosong.' });
  }

  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const eventsSheet = ensureAffiliateEventsSheet_(ss);
  const registrySheet = ss.getSheetByName(SHEET_NAME);
  if (!registrySheet) throw new Error('Sheet tidak ditemukan: ' + SHEET_NAME);

  const now = new Date();
  const eventId = Utilities.getUuid();

  eventsSheet.appendRow([
    now,
    eventId,
    eventType,
    referralCode,
    slug,
    clean_(p.partnerName),
    clean_(p.visitorId),
    clean_(p.pageUrl),
    clean_(p.destinationUrl),
    clean_(p.referrer),
    clean_(p.utmSource),
    clean_(p.utmMedium),
    clean_(p.utmCampaign),
    clean_(p.utmContent),
    clean_(p.utmTerm),
    clean_(p.userAgent),
  ]);

  updateAffiliateActivitySummary_(
    registrySheet,
    referralCode,
    slug,
    eventType,
    now
  );

  return json_({
    ok: true,
    eventId,
    eventType,
  });
}

function updateAffiliateActivitySummary_(sheet, referralCode, slug, eventType, now) {
  ensureStep8RegistryColumns_(sheet);

  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return;

  const rows = sheet.getRange(2, 1, lastRow - 1, 32).getDisplayValues();
  const targetIndex = rows.findIndex(r => {
    const rowCode = clean_(r[10]).toUpperCase();
    const rowSlug = clean_(r[11]).toLowerCase();

    if (referralCode && rowCode === referralCode) return true;
    if (slug && rowSlug === slug) return true;
    return false;
  });

  if (targetIndex < 0) return;

  const row = targetIndex + 2;

  if (eventType === 'PAGE_VIEW') {
    const current = Number(sheet.getRange(row, 30).getValue()) || 0;
    sheet.getRange(row, 30).setValue(current + 1);
  }

  if (eventType === 'CHECKOUT_CLICK') {
    const current = Number(sheet.getRange(row, 31).getValue()) || 0;
    sheet.getRange(row, 31).setValue(current + 1);
  }

  sheet.getRange(row, 32).setValue(now);
}

function parseWebhookPayload_(e, p) {
  let json = {};
  const raw = (e && e.postData && e.postData.contents) || '';

  try {
    if (raw) json = JSON.parse(raw);
  } catch (err) {
    json = {};
  }

  return {
    raw: raw || JSON.stringify(p || {}),
    root: json && typeof json === 'object' ? json : {},
    form: p || {},
  };
}


function pickWebhookValue_(payload, candidates) {
  for (const key of candidates) {
    const parts = key.split('.');
    let v = payload.root;
    let ok = true;

    for (const part of parts) {
      if (v && Object.prototype.hasOwnProperty.call(v, part)) {
        v = v[part];
      } else {
        ok = false;
        break;
      }
    }

    if (ok && v !== undefined && v !== null && String(v).trim() !== '') {
      return v;
    }

    if (
      Object.prototype.hasOwnProperty.call(payload.form, key) &&
      clean_(payload.form[key])
    ) {
      return payload.form[key];
    }
  }

  return '';
}


function numberWebhookValue_(value) {
  if (typeof value === 'number') return value;
  const digits = String(value || '').replace(/[^\d.-]/g, '');
  return Number(digits) || 0;
}


function normalizePaymentStatus_(value, eventType) {
  const s = clean_(value).toLowerCase();
  const e = clean_(eventType).toLowerCase();

  if (
    e === 'payment.received' ||
    e === 'payment_received' ||
    ['paid', 'success', 'successful', 'completed', 'received'].includes(s)
  ) {
    return 'PAID';
  }

  if (['refunded', 'refund'].includes(s)) return 'REFUNDED';
  if (['cancelled', 'canceled', 'failed', 'expired'].includes(s)) return s.toUpperCase();

  return s ? s.toUpperCase() : 'UNKNOWN';
}


function findAffiliateForPayment_(registrySheet, affiliateCode, affiliateSlug, checkoutUrl) {
  const lastRow = registrySheet.getLastRow();
  if (lastRow < 2) return null;

  const rows = registrySheet.getRange(2, 1, lastRow - 1, Math.max(39, registrySheet.getLastColumn())).getDisplayValues();

  const code = clean_(affiliateCode).toUpperCase();
  const slug = clean_(affiliateSlug).toLowerCase();
  const checkout = clean_(checkoutUrl);

  for (let i = 0; i < rows.length; i++) {
    const r = rows[i];
    const rowCode = clean_(r[10]).toUpperCase();
    const rowSlug = clean_(r[11]).toLowerCase();
    const rowCheckout = clean_(r[14]);
    const rowStatus = clean_(r[13]).toUpperCase();
    const duplicateFlag = clean_(r[19]).toUpperCase();
    const emailVerified = clean_(r[23]).toUpperCase();

    if (rowStatus !== 'APPROVED' || duplicateFlag !== 'OK' || emailVerified !== 'YES') {
      continue;
    }

    let source = '';
    if (code && rowCode === code) source = 'AFFILIATE_CODE';
    else if (slug && rowSlug === slug) source = 'AFFILIATE_SLUG';
    else if (checkout && rowCheckout && rowCheckout === checkout) source = 'CHECKOUT_URL';

    if (source) {
      return {
        row: i + 2,
        code: rowCode,
        slug: rowSlug,
        name: clean_(r[1]),
        checkoutUrl: rowCheckout,
        source,
      };
    }
  }

  return null;
}

function pickWebhookObject_(payload, candidates) {
  for (const key of candidates) {
    const parts = key.split('.');
    let v = payload.root;
    let ok = true;

    for (const part of parts) {
      if (v && Object.prototype.hasOwnProperty.call(v, part)) {
        v = v[part];
      } else {
        ok = false;
        break;
      }
    }

    if (ok && v && typeof v === 'object') return v;
  }

  return null;
}


function pickWebhookArray_(payload, candidates) {
  for (const key of candidates) {
    const parts = key.split('.');
    let v = payload.root;
    let ok = true;

    for (const part of parts) {
      if (v && Object.prototype.hasOwnProperty.call(v, part)) {
        v = v[part];
      } else {
        ok = false;
        break;
      }
    }

    if (ok && Array.isArray(v)) return v;
  }

  return [];
}


function extractLynkItems_(payload) {
  const items = pickWebhookArray_(payload, [
    'items',
    'data.items',
    'data.message_data.items',
    'order.items',
    'data.order.items',
  ]);

  return items.map(item => ({
    id: clean_(item && (item.id || item.productId || item.product_id)),
    name: clean_(item && (item.name || item.productName || item.product_name || item.title)),
    price: numberWebhookValue_(item && (item.price || item.amount || item.total)),
    qty: Number(item && (item.qty || item.quantity || 1)) || 1,
    raw: item || {},
  }));
}


function summarizeLynkItems_(items) {
  if (!items || !items.length) return '';
  return items
    .map(item => `${item.name || item.id || 'Item'} x${item.qty}`)
    .join(' | ');
}

function createCommissionForReviewedTransaction_(txSheet, txRow, registrySheet, commissionSheet, affiliate) {
  const transactionId = clean_(txSheet.getRange(txRow, 2).getDisplayValue());
  if (!transactionId) throw new Error('Transaction ID kosong.');

  if (commissionExistsForTransaction_(commissionSheet, transactionId)) {
    txSheet.getRange(txRow, 31).setValue('ALREADY_COMMISSIONED');
    txSheet.getRange(txRow, 32).setValue(new Date());
    return 'Transaksi sudah memiliki komisi.';
  }

  const paymentStatus = clean_(txSheet.getRange(txRow, 4).getDisplayValue()).toUpperCase();
  const productName = clean_(txSheet.getRange(txRow, 6).getDisplayValue());
  const amount = Number(txSheet.getRange(txRow, 10).getValue()) || 0;

  const productLooksPanglima =
    /panglima/i.test(productName) ||
    amount === PANGLIMA_PRODUCT_PRICE;

  if (paymentStatus !== 'PAID') {
    throw new Error('Transaksi belum berstatus PAID.');
  }

  if (!productLooksPanglima) {
    throw new Error('Transaksi tidak terdeteksi sebagai Paket Panglima.');
  }

  const now = new Date();
  const commissionId = Utilities.getUuid();

  txSheet.getRange(txRow, 12).setValue(affiliate.code);
  txSheet.getRange(txRow, 13).setValue(affiliate.slug);
  txSheet.getRange(txRow, 14).setValue('ATTRIBUTED');
  txSheet.getRange(txRow, 15).setValue('MANUAL_REVIEW');
  txSheet.getRange(txRow, 16).setValue('YES');
  txSheet.getRange(txRow, 17).setValue(PANGLIMA_COMMISSION_AMOUNT);
  txSheet.getRange(txRow, 18).setValue('PENDING');
  txSheet.getRange(txRow, 31).setValue('REVIEWED');
  txSheet.getRange(txRow, 32).setValue(now);

  commissionSheet.appendRow([
    commissionId,
    transactionId,
    now,
    affiliate.code,
    affiliate.slug,
    affiliate.name,
    productName || 'Paket Panglima UMKM',
    amount,
    PANGLIMA_COMMISSION_RATE,
    PANGLIMA_COMMISSION_AMOUNT,
    'PENDING',
    'UNPAID',
    '',
    '',
    '',
    'Atribusi disetujui melalui review manual Step 9B.',
  ]);

  updateRegistryPurchaseSummary_(
    registrySheet,
    affiliate.row,
    amount,
    PANGLIMA_COMMISSION_AMOUNT,
    now
  );

  return `Komisi ${affiliate.code} berhasil dibuat untuk transaksi ${transactionId}.`;
}

function handleStep9BManualReviewEdit(e) {
  if (!e || !e.range) return;

  const sheet = e.range.getSheet();
  if (sheet.getName() !== AFFILIATE_TRANSACTIONS_SHEET) return;
  if (e.range.getRow() < 2) return;
  if (e.range.getColumn() !== 30) return; // AD = Manual Affiliate Code

  const code = clean_(e.value).toUpperCase();
  if (!code) return;

  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const registry = ss.getSheetByName(SHEET_NAME);
  const commissionSheet = ss.getSheetByName(AFFILIATE_COMMISSIONS_SHEET);

  if (!registry || !commissionSheet) {
    throw new Error('Registry / Commission sheet tidak ditemukan.');
  }

  const affiliate = findAffiliateByCode_(registry, code);

  if (!affiliate) {
    sheet.getRange(e.range.getRow(), 31).setValue('INVALID_AFFILIATE');
    return;
  }

  try {
    const result = createCommissionForReviewedTransaction_(
      sheet,
      e.range.getRow(),
      registry,
      commissionSheet,
      affiliate
    );
    sheet.getRange(e.range.getRow(), 20).setValue(result);
  } catch (err) {
    sheet.getRange(e.range.getRow(), 31).setValue('ERROR');
    sheet.getRange(e.range.getRow(), 20).setValue(String(err && err.message ? err.message : err));
  }
}

  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const registry = ss.getSheetByName(SHEET_NAME);

  const now = new Date();

  sheets.tx.appendRow([
    now,
    transactionId || Utilities.getUuid(),
    eventType,
    paymentStatus,
    productId,
    productName,
    buyerName,
    buyerEmail,
    buyerPhone,
    amount,
    currency,
    affiliate ? affiliate.code : affiliateCode,
    affiliate ? affiliate.slug : affiliateSlug,
    attributionStatus,
    attributionSource,
    eligible ? 'YES' : 'NO',
    commissionAmount,
    commissionStatus,
    payload.raw,
    eligible ? '' : 'Jika transaksi ini milik affiliate, isi kolom Manual Affiliate Code.',
    now,
    createdAt,
    messageAction,
    messageCode,
    totalPrice,
    totalAddon,
    discount,
    shipping,
    JSON.stringify(items),
    '',
    affiliate ? 'AUTO_ATTRIBUTED' : 'NEEDS_REVIEW',
    '',
  ]);

      transactionId || '',

      affiliate.code,
      affiliate.slug,
      affiliate.name,
      productName || 'Paket Panglima UMKM',
      amount,
      PANGLIMA_COMMISSION_RATE,
      PANGLIMA_COMMISSION_AMOUNT,
      'PENDING',
      'UNPAID',
      '',
      '',
      '',
      '',
    ]);

    updateRegistryPurchaseSummary_(
      registry,
      affiliate.row,
      amount,
      PANGLIMA_COMMISSION_AMOUNT,
      now
    );
  }

  return json_({
    ok: true,
    transactionId,
    paymentStatus,
    attributionStatus,
    attributionSource,
    commissionEligible: eligible,
    commissionAmount,
  });
}

function workerSecretEquals_(left, right) {
  left = String(left || '');
  right = String(right || '');

  const leftDigest = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256,
    left,
    Utilities.Charset.UTF_8
  );

  const rightDigest = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256,
    right,
    Utilities.Charset.UTF_8
  );

  let diff = leftDigest.length ^ rightDigest.length;

  for (let i = 0; i < leftDigest.length; i++) {
    diff |= leftDigest[i] ^ rightDigest[i];
  }

  return diff === 0;
}


function validateWorkerToGasPaymentRequest_(e) {
  const configuredSecret = getWorkerToGasSecret_();

  if (!configuredSecret) {
    return {
      ok: false,
      error: 'WORKER_TO_GAS_SECRET_NOT_CONFIGURED',
    };
  }

  const suppliedSecret = clean_(
    e &&
    e.parameter &&
    e.parameter.worker_key
      ? e.parameter.worker_key
      : ''
  );

  if (
    !suppliedSecret ||
    !workerSecretEquals_(suppliedSecret, configuredSecret)
  ) {
    return {
      ok: false,
      error: 'UNAUTHORIZED_WORKER_TO_GAS_PAYMENT',
    };
  }

  return {
    ok: true,
  };
}

      itemsSummary: summarizeLynkItems_(items),
      parseResult: 'REJECTED_INVALID_LYNK_PAYLOAD',
      rawPayload: payload.raw,
    });

    return json_({
      ok: false,
      ignored: true,
      error: 'INVALID_LYNK_PAYMENT_PAYLOAD',
    });
  }

  return handlePaymentWebhook_(e, p);
}

function rebuildAffiliateFinancialSummary_() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const registry = ss.getSheetByName(SHEET_NAME);
  const commissionSheet = ss.getSheetByName(AFFILIATE_COMMISSIONS_SHEET);

  if (!registry || !commissionSheet) {
    throw new Error('Registry / Affiliate Commissions tidak ditemukan.');
  }

  ensureStep9RegistryColumns_(registry);

  const lastRegistryRow = registry.getLastRow();

  if (lastRegistryRow < 2) {
    return 'Tidak ada affiliate untuk dihitung.';
  }

  // Reset kolom AG:AM / 33:39 untuk seluruh affiliate.
  registry
    .getRange(2, 33, lastRegistryRow - 1, 7)
    .clearContent();

  const registryRows = registry
    .getRange(2, 1, lastRegistryRow - 1, Math.max(39, registry.getLastColumn()))
    .getDisplayValues();

  const registryByCode = {};

  registryRows.forEach((r, index) => {
    const code = clean_(r[10]).toUpperCase();

    if (code) {
      registryByCode[code] = {
        row: index + 2,
        purchases: 0,
        revenue: 0,
        earned: 0,
        pending: 0,
        approved: 0,
        paid: 0,
        lastPurchase: null,
      };
    }
  });

  if (commissionSheet.getLastRow() >= 2) {
    const rows = commissionSheet
      .getRange(
        2,
        1,
        commissionSheet.getLastRow() - 1,
        Math.max(16, commissionSheet.getLastColumn())
      )
      .getValues();

    rows.forEach(r => {
      const transactionId = clean_(r[1]);

      if (isStep9TestTransactionId_(transactionId)) return;

      const affiliateCode = clean_(r[3]).toUpperCase();
      const summary = registryByCode[affiliateCode];

      if (!summary) return;

      const paymentDate = r[2] instanceof Date ? r[2] : new Date(r[2]);
      const saleAmount = Number(r[7]) || 0;
      const commissionAmount = Number(r[9]) || 0;
      const commissionStatus = clean_(r[10]).toUpperCase();
      const payoutStatus = clean_(r[11]).toUpperCase();

      summary.purchases += 1;
      summary.revenue += saleAmount;
      summary.earned += commissionAmount;

      if (payoutStatus === 'PAID' || commissionStatus === 'PAID') {
        summary.paid += commissionAmount;
      } else if (commissionStatus === 'APPROVED') {
        summary.approved += commissionAmount;
      } else {
        summary.pending += commissionAmount;
      }

      if (
        paymentDate instanceof Date &&
        !isNaN(paymentDate.getTime()) &&
        (!summary.lastPurchase || paymentDate > summary.lastPurchase)
      ) {
        summary.lastPurchase = paymentDate;
      }
    });
  }

  Object.keys(registryByCode).forEach(code => {
    const s = registryByCode[code];

    registry.getRange(s.row, 33, 1, 7).setValues([[
      s.purchases,
      s.revenue,
      s.earned,
      s.pending,
      s.approved,
      s.paid,
      s.lastPurchase || '',
    ]]);
  });

  return 'Financial summary affiliate berhasil dihitung ulang.';
}

  const subject = 'Konfirmasi pendaftaran affiliate UMKM Naik Kelas';

function verifyAffiliateEmailData_(token) {
  token =
    clean_(
      token
    );

  if (
    !token ||
    token.length < 20
  ) {
    return {
      ok: false,
      verified: false,
      error:
        'INVALID_VERIFICATION_TOKEN',
      title:
        'Tautan verifikasi tidak valid.',
      message:
        'Silakan gunakan tautan terbaru dari email pendaftaran Anda.'
    };
  }

  const lock =
    LockService.getScriptLock();

  try {
    lock.waitLock(
      15000
    );

    const ss =
      SpreadsheetApp.openById(
        SPREADSHEET_ID
      );

    const sheet =
      ss.getSheetByName(
        SHEET_NAME
      );

    if (!sheet) {
      return {
        ok: false,
        verified: false,
        error:
          'REGISTRY_NOT_FOUND',
        title:
          'Data pendaftaran tidak tersedia.',
        message:
          'Silakan hubungi admin UMKM Naik Kelas.'
      };
    }

    const lastRow =
      sheet.getLastRow();

    if (lastRow < 2) {
      return {
        ok: false,
        verified: false,
        error:
          'REGISTRATION_NOT_FOUND',
        title:
          'Data pendaftaran tidak ditemukan.',
        message:
          'Silakan daftar kembali.'
      };
    }

    /*
     * Y = Email Verification Token
     */
    const tokens =
      sheet
        .getRange(
          2,
          25,
          lastRow - 1,
          1
        )
        .getDisplayValues();

    let foundRow = -1;

    for (
      let i = 0;
      i < tokens.length;
      i++
    ) {
      if (
        clean_(
          tokens[i][0]
        ) === token
      ) {
        foundRow =
          i + 2;
        break;
      }
    }

    if (foundRow === -1) {
      return {
        ok: false,
        verified: false,
        error:
          'VERIFICATION_TOKEN_NOT_FOUND',
        title:
          'Tautan verifikasi tidak ditemukan.',
        message:
          'Tautan mungkin salah atau data pendaftaran sudah berubah.'
      };
    }

    /*
     * Baca kondisi aktual langsung dari Registry.
     *
     * N = Status
     * X = Email Verified
     * Z = Email Verified At
     */
    const currentStatus =
      clean_(
        sheet
          .getRange(
            foundRow,
            14
          )
          .getDisplayValue()
      )
        .toUpperCase();

    const currentVerified =
      clean_(
        sheet
          .getRange(
            foundRow,
            24
          )
          .getDisplayValue()
      )
        .toUpperCase();

    const alreadyVerified =
      currentVerified ===
      'YES';

    /*
     * Verifikasi email harus idempotent.
     * Jika link dibuka ulang, data tetap konsisten.
     */
    if (!alreadyVerified) {
      sheet
        .getRange(
          foundRow,
          24
        )
        .setValue(
          'YES'
        );

      sheet
        .getRange(
          foundRow,
          26
        )
        .setValue(
          new Date()
        );
    }

    /*
     * Setelah email terverifikasi:
     * PENDING_EMAIL / EMAIL_ERROR -> PENDING
     *
     * Status lain seperti APPROVED / REJECTED / SUSPENDED
     * tidak disentuh.
     */
    if (
      currentStatus ===
        'PENDING_EMAIL' ||
      currentStatus ===
        'EMAIL_ERROR'
    ) {
      sheet
        .getRange(
          foundRow,
          14
        )
        .setValue(
          'PENDING'
        );
    }

    /*
     * Pastikan write benar-benar dikirim ke Spreadsheet
     * sebelum halaman sukses dikembalikan.
     */
    SpreadsheetApp.flush();

    /*
     * READ-BACK VALIDATION:
     * Jangan pernah mengembalikan sukses hanya berdasarkan
     * asumsi setValue berhasil.
     */
    const savedVerified =
      clean_(
        sheet
          .getRange(
            foundRow,
            24
          )
          .getDisplayValue()
      )
        .toUpperCase();

    const savedStatus =
      clean_(
        sheet
          .getRange(
            foundRow,
            14
          )
          .getDisplayValue()
      )
        .toUpperCase();

    const savedVerifiedAt =
      sheet
        .getRange(
          foundRow,
          26
        )
        .getValue();

    if (
      savedVerified !==
      'YES'
    ) {
      return {
        ok: false,
        verified: false,
        error:
          'VERIFICATION_WRITE_NOT_SAVED',
        title:
          'Verifikasi belum tersimpan.',
        message:
          'Sistem belum berhasil menyimpan konfirmasi email Anda. Silakan coba kembali beberapa saat lagi.'
      };
    }

    if (
      (
        currentStatus ===
          'PENDING_EMAIL' ||
        currentStatus ===
          'EMAIL_ERROR'
      ) &&
      savedStatus !==
        'PENDING'
    ) {
      return {
        ok: false,
        verified: false,
        error:
          'STATUS_WRITE_NOT_SAVED',
        title:
          'Verifikasi belum selesai.',
        message:
          'Email sudah terverifikasi, tetapi status pendaftaran belum berhasil diperbarui. Silakan coba kembali.'
      };
    }

    if (
      !savedVerifiedAt &&
      !alreadyVerified
    ) {
      return {
        ok: false,
        verified: false,
        error:
          'VERIFIED_AT_NOT_SAVED',
        title:
          'Verifikasi belum selesai.',
        message:
          'Konfirmasi email belum tersimpan lengkap. Silakan coba kembali.'
      };
    }

    const data =
      getAffiliateRowData_(
        sheet,
        foundRow
      );

    const displayName =
      clean_(
        data.firstName ||
        data.name ||
        ''
      );

    return {
      ok: true,
      verified: true,
      alreadyVerified:
        alreadyVerified,
      status:
        savedStatus,
      title:
        alreadyVerified
          ? 'Email sudah terverifikasi.'
          : 'Email berhasil diverifikasi.',
      message:
        displayName
          ? (
              'Terima kasih, ' +
              displayName +
              '. Pendaftaran Anda akan ditinjau admin maksimal 1x24 jam. ' +
              'Setelah akun disetujui, Anda akan menerima email aktivasi dan dapat login ke Portal Affiliate menggunakan OTP. ' +
              'Jika setelah 1x24 jam belum ada kabar, silakan hubungi CS UMKM Naik Kelas melalui halo@umkmnaikkelas.id.'
            )
          : (
              'Pendaftaran Anda akan ditinjau admin maksimal 1x24 jam. ' +
              'Setelah akun disetujui, Anda akan menerima email aktivasi dan dapat login ke Portal Affiliate menggunakan OTP. ' +
              'Jika setelah 1x24 jam belum ada kabar, silakan hubungi CS UMKM Naik Kelas melalui halo@umkmnaikkelas.id.'
            ),
      homeUrl:
        SITE_URL
    };

  } catch (err) {
    return {
      ok: false,
      verified: false,
      error:
        'VERIFICATION_EXCEPTION',
      title:
        'Verifikasi belum berhasil.',
      message:
        String(
          err &&
          err.message
            ? err.message
            : err
        )
    };

  } finally {
    try {
      lock.releaseLock();
    } catch (ignore) {}
  }
}

function handleAffiliateApproval(e) {
  try {
    if (!e || !e.range) return;

    const sheet = e.range.getSheet();
    if (sheet.getName() !== SHEET_NAME) return;
    if (e.range.getRow() < 2) return;
    if (e.range.getColumn() !== 14) return; // N Status only

    const row = e.range.getRow();
    const status = clean_(sheet.getRange(row, 14).getDisplayValue()).toUpperCase();
    if (status !== 'APPROVED') return;

    const data = getAffiliateRowData_(sheet, row);
    const validCheckout = /^https:\/\//i.test(data.checkoutUrl) && data.checkoutUrl.indexOf('lynk.id') !== -1;
    const duplicateOkay = data.duplicateFlag === 'OK';
    const verifiedOkay = data.emailVerified === 'YES';
    const validEmail = isValidEmail_(data.email);

    if (!verifiedOkay || !validCheckout || !duplicateOkay || !validEmail) {
      let nextStatus = 'PENDING';
      let reason = '';

      if (!verifiedOkay) {
        nextStatus = 'PENDING_EMAIL';
        reason = 'Email affiliate belum diverifikasi.';
      } else if (!validCheckout) {
        nextStatus = 'PENDING';
        reason = 'Link Lynk Affiliate belum valid.';
      } else if (!duplicateOkay) {
        nextStatus = 'REVIEW_DUPLICATE';
        reason = 'Duplicate Flag belum OK.';
      } else {
        nextStatus = 'PENDING';
        reason = 'Alamat email tidak valid.';
      }

      sheet.getRange(row, 14).setValue(nextStatus);
      sheet.getRange(row, 23).setValue(reason);
      return;
    }

    const approvalCell = sheet.getRange(row, 16); // P
    if (!approvalCell.getValue()) approvalCell.setValue(new Date());

    if (data.approvalEmailStatus === 'SENT') return;
    sendApprovalEmailForRow_(sheet, row);
  } catch (err) {
    console.error(err);
    try {
      if (e && e.range) {
        e.range.getSheet().getRange(e.range.getRow(), 23).setValue(String(err && err.message ? err.message : err));
      }
    } catch (_) {}
  }
}


function sendApprovalEmailForRow_(sheet, row, forceResend) {
  const data = getAffiliateRowData_(sheet, row);

  /*
   * Heal link lama di Registry agar seluruh sistem memakai URL canonical.
   * M  = Halaman Affiliate
   * AC = Marketing Kit URL
   */
  const canonicalPage =
    SITE_URL +
    '/?affiliate=' +
    encodeURIComponent(
      clean_(
        data.slug || ''
      )
    );

  const canonicalKit =
    canonicalPage +
    '&kit=1';

  sheet
    .getRange(
      row,
      13
    )
    .setValue(
      canonicalPage
    );

  sheet
    .getRange(
      row,
      29
    )
    .setValue(
      canonicalKit
    );

  data.affiliatePage =
    canonicalPage;

  data.marketingKitUrl =
    canonicalKit;

  const statusCell = sheet.getRange(row, 21); // U
  const sentAtCell = sheet.getRange(row, 22); // V
  const errorCell = sheet.getRange(row, 23);  // W

  if (!forceResend && data.approvalEmailStatus === 'SENT') return;

  try {
    statusCell.setValue('SENDING');
    errorCell.clearContent();

    if (MailApp.getRemainingDailyQuota() < 1) {
      throw new Error('Kuota pengiriman email Apps Script hari ini sudah habis.');
    }

    const subject = `Selamat! Affiliate Panglima UMKM Anda Aktif — ${data.referralCode}`;
    const htmlBody = buildApprovalEmailHtml_(data);
    const plainBody = buildApprovalEmailText_(data);
    const kitAttachment = Utilities.newBlob(
      buildMarketingKitText_(data),
      'text/plain',
      `Marketing_Kit_Panglima_${data.referralCode}.txt`
    );

    const aliases = GmailApp.getAliases();
    if (!aliases.includes(SENDER_ALIAS)) {
      throw new Error(
        `Sender ${SENDER_ALIAS} belum terdaftar di Gmail > Settings > Accounts and Import > Send mail as.`
      );
    }

    GmailApp.sendEmail(data.email, subject, plainBody, {
      htmlBody,
      name: EMAIL_SENDER_NAME,
      from: SENDER_ALIAS,
      replyTo: REPLY_TO_EMAIL || SENDER_ALIAS,
      attachments: [kitAttachment],
    });

    statusCell.setValue('SENT');
    sentAtCell.setValue(new Date());
    errorCell.clearContent();
  } catch (err) {
    statusCell.setValue('ERROR');
    errorCell.setValue(String(err && err.message ? err.message : err));
    throw err;
  }
}

function getApprovedAffiliateRegistry_() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) throw new Error('Sheet tidak ditemukan: ' + SHEET_NAME);

  ensureStep6AHeaders_(sheet);

  const lastRow = sheet.getLastRow();
  const rows = lastRow > 1
    ? sheet.getRange(2, 1, lastRow - 1, 29).getDisplayValues()
    : [];

  const partners = {};

  rows.forEach((r) => {
    const name = clean_(r[1]);
    const whatsapp = normalizePhone_(r[3]);
    const referralCode = clean_(r[10]).toUpperCase();
    const slug = slugify_(r[11]);
    const status = clean_(r[13]).toUpperCase();
    const checkoutUrl = clean_(r[14]);
    const duplicateFlag = clean_(r[19]).toUpperCase();
    const emailVerified = clean_(r[23]).toUpperCase();

    const validCheckout = /^https:\/\//i.test(checkoutUrl) && checkoutUrl.indexOf('lynk.id') !== -1;

    if (!name || !whatsapp || !referralCode || !slug) return;
    if (status !== 'APPROVED') return;
    if (duplicateFlag !== 'OK') return;
    if (emailVerified !== 'YES') return;
    if (!validCheckout) return;

    const affiliatePage = clean_(r[12]) || `${SITE_URL}/?affiliate=${encodeURIComponent(slug)}`;
    const kitUrl = clean_(r[28]) || `${SITE_URL}/?affiliate=${encodeURIComponent(slug)}&kit=1`;

    partners[slug] = {
      name,
      title: 'Partner Panglima UMKM',
      whatsapp,
      code: referralCode,
      affiliatePage,
      kitUrl,
      checkoutUrl,
      status: 'APPROVED',
      active: true,
    };
  });

  return {
    ok: true,
    generatedAt: new Date().toISOString(),
    count: Object.keys(partners).length,
    partners,
  };
}

function createVerificationToken_() {
  return (Utilities.getUuid() + Utilities.getUuid()).replace(/-/g, '');
}


function createReferralCode_(firstName, whatsapp) {
  const letters = removeDiacritics_(firstName)
    .replace(/[^a-zA-Z]/g, '')
    .toUpperCase()
    .slice(0, 3)
    .padEnd(3, 'X');
  const digits = normalizePhone_(whatsapp);
  const last3 = digits.slice(-3).padStart(3, '0');
  return letters + last3;
}


function getFirstName_(name) {
  return clean_(name).split(/\s+/)[0] || '';
}


function normalizePhone_(value) {
  let digits = String(value || '').replace(/\D/g, '');
  if (digits.startsWith('0')) digits = '62' + digits.slice(1);
  return digits;
}


function slugify_(value) {
  return removeDiacritics_(String(value || ''))
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-+|-+$/g, '') || 'partner';
}


function removeDiacritics_(value) {
  return String(value || '').normalize('NFD').replace(/[\u0300-\u036f]/g, '');
}


function isValidEmail_(email) {
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(clean_(email));
}


function escapeHtml_(value) {
  return String(value || '')
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#039;');
}


function jsonOrJsonp_(data, callback) {
  const safeCallback = String(callback || '').replace(/[^a-zA-Z0-9_.$]/g, '');
  if (safeCallback) {
    return ContentService
      .createTextOutput(`${safeCallback}(${JSON.stringify(data)});`)
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return json_(data);
}
