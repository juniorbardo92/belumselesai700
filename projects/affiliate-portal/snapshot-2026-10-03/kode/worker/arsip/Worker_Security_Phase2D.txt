/* =========================================================
   RESPONSE HELPERS
   ========================================================= */

function jsonResponse(data, status = 200, extraHeaders = {}) {
  return new Response(JSON.stringify(data), {
    status,
    headers: {
      "Content-Type": "application/json; charset=UTF-8",
      "Cache-Control": "no-store",
      ...extraHeaders,
    },
  });
}

function portalCorsHeaders(origin) {
  const allowed = PORTAL_ALLOWED_ORIGINS.has(origin);

  return {
    ...(allowed
      ? {
          "Access-Control-Allow-Origin": origin,
        }
      : {}),
    "Access-Control-Allow-Methods": "POST, OPTIONS",
    "Access-Control-Allow-Headers": "Content-Type",
    "Access-Control-Max-Age": "86400",
    "Vary": "Origin",
    "Cache-Control": "no-store",
  };
}

function isPortalOriginAllowed(request) {
  const origin = request.headers.get("origin") || "";

  // Server-to-server / manual HTTP tester boleh tanpa Origin.
  if (!origin) return true;

  // Portal Admin yang disajikan langsung oleh Worker
  // melakukan POST same-origin ke /portal.
  const workerOrigin = new URL(request.url).origin;

  if (origin === workerOrigin) {
    return true;
  }

  return PORTAL_ALLOWED_ORIGINS.has(origin);
}


/* =========================================================
   SECURITY PHASE 2D — OTP RATE LIMIT ONLY
   Binding: OTP_RATE_LIMITER (5 requests / 60 seconds)
   Hanya berlaku untuk request/verify OTP admin + affiliate.
   ========================================================= */

const OTP_RATE_LIMIT_ACTIONS = new Set([
  "portal_request_otp",
  "portal_verify_otp",
  "admin_request_otp",
  "admin_verify_otp",
]);

async function enforceOtpRateLimit(request, env, action) {
  if (!OTP_RATE_LIMIT_ACTIONS.has(action)) {
    return { allowed: true };
  }

  // Fail-open jika binding belum tersedia agar fungsi produksi lain
  // tidak ikut terganggu oleh salah konfigurasi binding.
  if (!env?.OTP_RATE_LIMITER?.limit) {
    return { allowed: true };
  }

  const clientIp =
    request.headers.get("CF-Connecting-IP") ||
    "unknown";

  const result =
    await env.OTP_RATE_LIMITER.limit({
      key: `${action}:${clientIp}`,
    });

  return {
    allowed: Boolean(result?.success),
  };
}



/* =========================================================
   PANG LIMA AFFILIATE LYNK LINK VALIDATION
   ========================================================= */

function parsePanglimaAffiliateLynkUrl(
  value
) {
  const raw =
    String(
      value || ""
    ).trim();

  if (!raw) {
    return {
      valid: false,
      retryable: false,
      error:
        "LYNK_AFFILIATE_URL_REQUIRED",
      message:
        "Link Affiliate Lynk.id Paket Panglima UMKM wajib diisi.",
    };
  }

  let parsed;

  try {
    parsed =
      new URL(
        raw
      );
  } catch (error) {
    return {
      valid: false,
      retryable: false,
      error:
        "LYNK_AFFILIATE_URL_FORMAT_INVALID",
      message:
        "Link Affiliate Lynk.id belum valid. Gunakan link lengkap seperti https://lynk.id/usernameanda/9rov84wv3grz.",
    };
  }

  const host =
    String(
      parsed.hostname || ""
    )
      .trim()
      .toLowerCase()
      .replace(
        /^www\./,
        ""
      );

  if (
    parsed.protocol !==
      "https:" ||
    host !==
      "lynk.id" ||
    parsed.username ||
    parsed.password ||
    parsed.port
  ) {
    return {
      valid: false,
      retryable: false,
      error:
        "LYNK_AFFILIATE_URL_HOST_INVALID",
      message:
        "Gunakan link HTTPS resmi dari lynk.id.",
    };
  }

  const segments =
    parsed.pathname
      .split("/")
      .filter(Boolean);

  if (
    segments.length !==
      2
  ) {
    return {
      valid: false,
      retryable: false,
      error:
        "LYNK_AFFILIATE_URL_PATH_INVALID",
      message:
        "Format link belum benar. Gunakan link affiliate pribadi untuk Paket Panglima UMKM.",
    };
  }

  const username =
    String(
      segments[0] || ""
    ).trim();

  const productId =
    String(
      segments[1] || ""
    ).trim();

  if (
    !/^[A-Za-z0-9._-]{2,80}$/.test(
      username
    )
  ) {
    return {
      valid: false,
      retryable: false,
      error:
        "LYNK_AFFILIATE_USERNAME_INVALID",
      message:
        "Username pada link Lynk.id belum valid.",
    };
  }

  if (
    productId !==
    PANGLIMA_LYNK_PRODUCT_ID
  ) {
    return {
      valid: false,
      retryable: false,
      error:
        "LYNK_PRODUCT_ID_INVALID",
      message:
        "Link tersebut bukan link affiliate untuk Paket Panglima UMKM.",
    };
  }

  if (
    username.toLowerCase() ===
    PANGLIMA_LYNK_MERCHANT_USERNAME
      .toLowerCase()
  ) {
    return {
      valid: false,
      retryable: false,
      error:
        "LYNK_MERCHANT_LINK_NOT_ALLOWED",
      message:
        "Link merchant utama tidak dapat digunakan. Masukkan link affiliate Lynk.id pribadi Anda untuk Paket Panglima UMKM.",
    };
  }

  return {
    valid: true,
    retryable: false,
    username,
    productId,
    canonicalUrl:
      `https://lynk.id/${username}/${PANGLIMA_LYNK_PRODUCT_ID}`,
  };
}


async function validatePanglimaAffiliateLynkUrl(
  value
) {
  const parsed =
    parsePanglimaAffiliateLynkUrl(
      value
    );

  if (!parsed.valid) {
    return parsed;
  }

  let response;

  try {
    response =
      await fetch(
        parsed.canonicalUrl,
        {
          method:
            "GET",
          redirect:
            "follow",
          headers: {
            "Accept":
              "text/html,application/xhtml+xml,application/json;q=0.9,*/*;q=0.8",
            "User-Agent":
              "Mozilla/5.0 (compatible; UMKMNaikKelas-LinkValidator/1.0)",
          },
        }
      );
  } catch (error) {
    return {
      ...parsed,
      valid: false,
      retryable: true,
      error:
        "LYNK_VALIDATION_UNAVAILABLE",
      message:
        "Link belum dapat diverifikasi otomatis. Silakan coba kembali beberapa saat lagi atau hubungi CS UMKM Naik Kelas.",
    };
  }

  const status =
    Number(
      response.status || 0
    );

  /*
   * Body tidak diperlukan.
   * Batalkan stream agar resource Worker tidak terpakai lebih lama.
   */
  try {
    if (
      response.body &&
      typeof response.body.cancel ===
        "function"
    ) {
      await response.body.cancel();
    }
  } catch (error) {
    // no-op
  }

  if (
    status >= 200 &&
    status < 400
  ) {
    return {
      ...parsed,
      valid: true,
      retryable: false,
      liveChecked: true,
      upstreamStatus:
        status,
      message:
        "Link affiliate Lynk.id Paket Panglima UMKM valid.",
    };
  }

  if (
    status === 404 ||
    status === 410
  ) {
    return {
      ...parsed,
      valid: false,
      retryable: false,
      liveChecked: true,
      upstreamStatus:
        status,
      error:
        "LYNK_AFFILIATE_URL_NOT_FOUND",
      message:
        "Link Affiliate Lynk.id tidak ditemukan. Pastikan link affiliate Paket Panglima UMKM yang dimasukkan benar.",
    };
  }

  if (
    status === 401 ||
    status === 403 ||
    status === 429 ||
    status >= 500
  ) {
    return {
      ...parsed,
      valid: false,
      retryable: true,
      liveChecked: true,
      upstreamStatus:
        status,
      error:
        "LYNK_VALIDATION_UNAVAILABLE",
      message:
        "Link belum dapat diverifikasi otomatis saat ini. Silakan coba kembali beberapa saat lagi atau hubungi CS UMKM Naik Kelas.",
    };
  }

  return {
    ...parsed,
    valid: false,
    retryable: false,
    liveChecked: true,
    upstreamStatus:
      status,
    error:
      "LYNK_AFFILIATE_URL_INVALID",
    message:
      "Link Affiliate Lynk.id belum valid. Pastikan link yang dimasukkan adalah link affiliate pribadi untuk Paket Panglima UMKM.",
  };
}


async function handleLynkAffiliateValidation(
  request
) {
  const origin =
    request.headers.get(
      "origin"
    ) || "";

  const cors =
    portalCorsHeaders(
      origin
    );

  if (
    request.method ===
    "OPTIONS"
  ) {
    if (
      !isPortalOriginAllowed(
        request
      )
    ) {
      return jsonResponse(
        {
          ok: false,
          error:
            "ORIGIN_NOT_ALLOWED",
        },
        403,
        cors
      );
    }

    return new Response(
      null,
      {
        status: 204,
        headers: cors,
      }
    );
  }

  if (
    !isPortalOriginAllowed(
      request
    )
  ) {
    return jsonResponse(
      {
        ok: false,
        error:
          "ORIGIN_NOT_ALLOWED",
      },
      403,
      cors
    );
  }

  if (
    request.method !==
    "POST"
  ) {
    return jsonResponse(
      {
        ok: false,
        error:
          "METHOD_NOT_ALLOWED",
      },
      405,
      cors
    );
  }

  const {
    parsed,
  } =
    await readIncomingBody(
      request
    );

  const validation =
    await validatePanglimaAffiliateLynkUrl(
      parsed
        .lynkAffiliateUrl ||
      parsed
        .checkoutUrl ||
      parsed
        .lynkUrl ||
      ""
    );

  const status =
    validation.valid
      ? 200
      : validation.retryable
      ? 503
      : 400;

  return jsonResponse(
    {
      ok:
        validation.valid,
      ...validation,
    },
    status,
    cors
  );
}


async function handleAffiliateRegistration(
  request,
  env
) {
  const origin =
    request.headers.get(
      "origin"
    ) || "";

  const cors =
    portalCorsHeaders(
      origin
    );

  if (
    request.method ===
    "OPTIONS"
  ) {
    if (
      !isPortalOriginAllowed(
        request
      )
    ) {
      return jsonResponse(
        {
          ok: false,
          error:
            "ORIGIN_NOT_ALLOWED",
        },
        403,
        cors
      );
    }

    return new Response(
      null,
      {
        status: 204,
        headers: cors,
      }
    );
  }

  if (
    !isPortalOriginAllowed(
      request
    )
  ) {
    return jsonResponse(
      {
        ok: false,
        error:
          "ORIGIN_NOT_ALLOWED",
      },
      403,
      cors
    );
  }

  if (
    request.method !==
    "POST"
  ) {
    return jsonResponse(
      {
        ok: false,
        error:
          "METHOD_NOT_ALLOWED",
      },
      405,
      cors
    );
  }

  if (
    !env
      .GOOGLE_AFFILIATE_WEBHOOK_URL
  ) {
    return jsonResponse(
      {
        ok: false,
        error:
          "GOOGLE_ENDPOINT_NOT_CONFIGURED",
        message:
          "Layanan pendaftaran belum siap. Silakan hubungi admin UMKM Naik Kelas.",
      },
      503,
      cors
    );
  }

  const {
    parsed,
  } =
    await readIncomingBody(
      request
    );

  const validation =
    await validatePanglimaAffiliateLynkUrl(
      parsed
        .lynkAffiliateUrl ||
      parsed
        .checkoutUrl ||
      parsed
        .lynkUrl ||
      ""
    );

  if (
    !validation.valid
  ) {
    return jsonResponse(
      {
        ok: false,
        ...validation,
      },
      validation.retryable
        ? 503
        : 400,
      cors
    );
  }

  const normalizedPayload = {
    ...parsed,

    action:
      "affiliate_register",

    lynkAffiliateUrl:
      validation.canonicalUrl,

    lynkValidationStatus:
      "VALID",

    lynkValidationCheckedAt:
      new Date()
        .toISOString(),

    lynkValidatedBy:
      "cloudflare-worker",
  };

  const googleFormBody =
    new URLSearchParams(
      Object.entries(
        normalizedPayload
      ).reduce(
        (
          acc,
          [
            key,
            value,
          ]
        ) => {
          if (
            value !== undefined &&
            value !== null
          ) {
            acc[key] =
              typeof value ===
                "string"
                ? value
                : String(
                    value
                  );
          }

          return acc;
        },
        {}
      )
    ).toString();

  const downstream =
    await forwardToGoogle(
      env,
      googleFormBody,
      "application/x-www-form-urlencoded;charset=UTF-8",
      "panglima-affiliate-registration"
    );

  if (
    !downstream.ok
  ) {
    return jsonResponse(
      {
        ok: false,
        error:
          downstream.error ||
          "REGISTRATION_DOWNSTREAM_ERROR",
        message:
          "Pendaftaran belum dapat diproses. Silakan coba kembali.",
        downstreamStatus:
          downstream.status,
      },
      502,
      cors
    );
  }

  if (
    downstream.data &&
    typeof downstream.data ===
      "object"
  ) {
    return jsonResponse(
      downstream.data,
      200,
      cors
    );
  }

  return jsonResponse(
    {
      ok: false,
      error:
        "INVALID_REGISTRATION_RESPONSE",
      message:
        "Pendaftaran belum dapat diproses. Silakan coba kembali.",
    },
    502,
    cors
  );
}

/* =========================================================
   BODY PARSER
   ========================================================= */

async function readIncomingBody(request) {
  const raw = await request.text();

  const contentType =
    request.headers.get("content-type") ||
    "application/json";

  let parsed = {};

  if (raw) {
    try {
      parsed = JSON.parse(raw);
    } catch (e) {
      if (
        contentType.includes(
          "application/x-www-form-urlencoded"
        )
      ) {
        parsed = Object.fromEntries(
          new URLSearchParams(raw)
        );
      }
    }
  }

  return {
    raw,
    parsed,
    contentType,
  };
}


/* =========================================================
   SIGNATURE DIAGNOSTIC
   ========================================================= */

function bytesToHex(buffer) {
  return [...new Uint8Array(buffer)]
    .map((b) =>
      b.toString(16).padStart(2, "0")
    )
    .join("");
}

function bytesToBase64(buffer) {
  const bytes =
    new Uint8Array(buffer);

  let binary = "";

  for (const b of bytes) {
    binary +=
      String.fromCharCode(b);
  }

  return btoa(binary);
}

async function hmacSha256(raw, secret) {
  const encoder =
    new TextEncoder();

  const key =
    await crypto.subtle.importKey(
      "raw",
      encoder.encode(secret),
      {
        name: "HMAC",
        hash: "SHA-256",
      },
      false,
      ["sign"]
    );

  const mac =
    await crypto.subtle.sign(
      "HMAC",
      key,
      encoder.encode(raw)
    );

  return {
    hex:
      bytesToHex(mac),

    base64:
      bytesToBase64(mac),
  };
}

async function sha256(text) {
  const digest =
    await crypto.subtle.digest(
      "SHA-256",
      new TextEncoder().encode(
        text
      )
    );

  return {
    hex:
      bytesToHex(digest),

    base64:
      bytesToBase64(digest),
  };
}

function normalizeSignature(value) {
  let s =
    String(
      value || ""
    ).trim();

  if (
    (
      s.startsWith('"') &&
      s.endsWith('"')
    ) ||
    (
      s.startsWith("'") &&
      s.endsWith("'")
    )
  ) {
    s =
      s.slice(
        1,
        -1
      );
  }

  s =
    s
      .replace(
        /^sha256=/i,
        ""
      )
      .trim();

  return s;
}

async function diagnoseLynkSignature(
  raw,
  signature,
  merchantKey
) {
  const normalized =
    normalizeSignature(
      signature
    );

  if (!normalized) {
    return {
      available: false,
      merchantKeyReady:
        Boolean(
          merchantKey
        ),
      match:
        "NO_SIGNATURE",
    };
  }

  if (!merchantKey) {
    return {
      available: true,
      merchantKeyReady:
        false,
      match:
        "MERCHANT_KEY_NOT_CONFIGURED",
    };
  }

  const [
    hmac,
    rawThenKey,
    keyThenRaw,
  ] =
    await Promise.all([
      hmacSha256(
        raw,
        merchantKey
      ),

      sha256(
        raw +
        merchantKey
      ),

      sha256(
        merchantKey +
        raw
      ),
    ]);

  const candidates = {
    HMAC_SHA256_HEX:
      hmac.hex,

    HMAC_SHA256_BASE64:
      hmac.base64,

    SHA256_BODY_PLUS_KEY_HEX:
      rawThenKey.hex,

    SHA256_BODY_PLUS_KEY_BASE64:
      rawThenKey.base64,

    SHA256_KEY_PLUS_BODY_HEX:
      keyThenRaw.hex,

    SHA256_KEY_PLUS_BODY_BASE64:
      keyThenRaw.base64,
  };

  for (
    const [
      name,
      candidate,
    ] of
    Object.entries(
      candidates
    )
  ) {
    if (
      normalized ===
      candidate
    ) {
      return {
        available: true,
        merchantKeyReady:
          true,
        match:
          name,
      };
    }
  }

  return {
    available: true,
    merchantKeyReady:
      true,
    match:
      "NO_MATCH",
  };
}


/* =========================================================
   GOOGLE APPS SCRIPT FORWARDER
   ========================================================= */

async function forwardToGoogle(
  env,
  raw,
  contentType,
  forwardedBy,
  requireWorkerSecret = false
) {
  if (
    !env
      .GOOGLE_AFFILIATE_WEBHOOK_URL
  ) {
    return {
      ok: false,
      error:
        "DOWNSTREAM_NOT_CONFIGURED",
      status: 500,
      text: "",
      data: null,
    };
  }

  let downstreamUrl =
    env
      .GOOGLE_AFFILIATE_WEBHOOK_URL;

  if (
    requireWorkerSecret
  ) {
    if (
      !env
        .WORKER_TO_GAS_SECRET
    ) {
      return {
        ok: false,
        error:
          "WORKER_TO_GAS_SECRET_NOT_CONFIGURED",
        status: 500,
        text: "",
        data: null,
      };
    }

    const securedUrl =
      new URL(
        downstreamUrl
      );

    securedUrl
      .searchParams
      .set(
        "worker_key",
        env
          .WORKER_TO_GAS_SECRET
      );

    downstreamUrl =
      securedUrl
        .toString();
  }

  let googleResponse;

  try {
    googleResponse =
      await fetch(
        downstreamUrl,
        {
          method:
            "POST",

          headers: {
            "Content-Type":
              contentType,

            "X-Forwarded-By":
              forwardedBy,
          },

          body:
            raw,

          redirect:
            "follow",
        }
      );

  } catch (error) {
    console.error(
      "Google Apps Script fetch error:",
      String(error)
    );

    return {
      ok: false,
      error:
        "DOWNSTREAM_CONNECTION_FAILED",
      status: 502,
      text: "",
      data: null,
    };
  }

  const text =
    await googleResponse.text();

  let data = null;

  try {
    data =
      text
        ? JSON.parse(text)
        : null;
  } catch (e) {
    data = null;
  }

  return {
    ok:
      googleResponse.ok,

    status:
      googleResponse.status,

    text,
    data,
  };
}

async function handlePortal(
  request,
  env
) {
  const origin =
    request.headers.get(

    ) || "";

  const cors =
    portalCorsHeaders(
      origin
    );

  if (
    request.method ===
    "OPTIONS"
  ) {
    if (
      origin &&
      !PORTAL_ALLOWED_ORIGINS.has(
        origin
      )
    ) {
      return jsonResponse(
        {
          ok: false,
          error:
            "ORIGIN_NOT_ALLOWED",
        },
        403,
        cors
      );
    }

    return new Response(
      null,
      {
        status: 204,
        headers: cors,
      }
    );
  }

  if (
    request.method !==
    "POST"
  ) {
    return jsonResponse(
      {
        ok: false,
        error:
          "METHOD_NOT_ALLOWED",
      },
      405,
      cors
    );
  }

  if (
    !isPortalOriginAllowed(
      request
    )
  ) {
    return jsonResponse(
      {
        ok: false,
        error:
          "ORIGIN_NOT_ALLOWED",
      },
      403,
      cors
    );
  }

  const {
    raw,
    parsed,
    contentType,
  } =
    await readIncomingBody(
      request
    );

  const action =
    String(
      parsed?.action || ""
    )
      .trim()
      .toLowerCase();

  if (
    !PORTAL_ALLOWED_ACTIONS.has(
      action
    )
  ) {
    return jsonResponse(
      {
        ok: false,
        error:
          "PORTAL_ACTION_NOT_ALLOWED",
      },
      400,
      cors
    );
  }

  const otpRateLimit =
    await enforceOtpRateLimit(
      request,
      env,
      action
    );

  if (!otpRateLimit.allowed) {
    return jsonResponse(
      {
        ok: false,
        error:
          "OTP_RATE_LIMITED",
        message:
          "Terlalu banyak percobaan OTP. Silakan tunggu sebentar lalu coba lagi.",
      },
      429,
      cors
    );
  }

  /*
   * Jangan log:
   * - email
   * - OTP
   * - sessionToken
   * - body mentah
   */
  console.log(
    JSON.stringify({
      timestamp:
        new Date()
          .toISOString(),

      channel:
        "affiliate_portal",

      action,

      origin:
        origin || null,
    })
  );

  /*
   * Apps Script doPost portal membaca e.parameter.
   * Karena website mengirim JSON ke Worker, ubah payload
   * menjadi application/x-www-form-urlencoded sebelum
   * diteruskan ke Google Apps Script.
   */
  const googleFormBody =
    new URLSearchParams(
      Object.entries(parsed || {})
        .reduce((acc, [key, value]) => {
          if (
            value !== undefined &&
            value !== null
          ) {
            acc[key] =
              typeof value === "string"
                ? value
                : String(value);
          }

          return acc;
        }, {})
    ).toString();

  const downstream =
    await forwardToGoogle(
      env,
      googleFormBody,
      "application/x-www-form-urlencoded;charset=UTF-8",
      "panglima-affiliate-portal"
    );

  console.log(
    JSON.stringify({
      timestamp:
        new Date()
          .toISOString(),

      channel:
        "affiliate_portal",

      action,

      downstreamStatus:
        downstream.status,
    })
  );

  if (!downstream.ok) {
    return jsonResponse(
      {
        ok: false,
        error:
          downstream.error ||
          "PORTAL_DOWNSTREAM_ERROR",
        downstreamStatus:
          downstream.status,
      },
      502,
      cors
    );
  }

  if (
    downstream.data &&
    typeof downstream.data ===
      "object"
  ) {
    return jsonResponse(
      downstream.data,
      200,
      cors
    );
  }

  /*
   * DEBUG AMAN:
   * Jika Google Apps Script merespons non-JSON,
   * tampilkan preview singkat agar penyebabnya terlihat.
   * Tidak menggunakan variabel debugPreview,
   * sehingga tidak ada risiko duplicate declaration.
   */
  return jsonResponse(
    {
      ok: false,
      error:
        "INVALID_PORTAL_RESPONSE",

      message:
        "Google Apps Script response: " +
        (
          String(
            downstream.text || ""
          )
            .replace(
              /<[^>]*>/g,
              " "
            )
            .replace(
              /\s+/g,
              " "
            )
            .trim()
            .slice(
              0,
              500
            ) ||
          "[EMPTY RESPONSE]"
        ),

      downstreamStatus:
        downstream.status,
    },
    502,
    cors
  );
}



/* =========================================================
   AFFILIATE EMAIL VERIFICATION PROXY
   GET /verify?token=...
   ========================================================= */

async function handleAffiliateVerification(
  request,
  env
) {
  if (
    request.method !== "GET" &&
    request.method !== "HEAD"
  ) {
    return jsonResponse(
      {
        ok: false,
        error:
          "METHOD_NOT_ALLOWED",
      },
      405
    );
  }

  function renderVerificationPage(
    success,
    title,
    message,
    statusCode = 200
  ) {
    const accent =
      success
        ? "#15803d"
        : "#b91c1c";

    const soft =
      success
        ? "#f0fdf4"
        : "#fef2f2";

    const label =
      success
        ? "VERIFIKASI BERHASIL"
        : "VERIFIKASI BELUM BERHASIL";

    const safe = (value) =>
      String(
        value ?? ""
      )
        .replace(
          /&/g,
          "&amp;"
        )
        .replace(
          /</g,
          "&lt;"
        )
        .replace(
          />/g,
          "&gt;"
        )
        .replace(
          /"/g,
          "&quot;"
        )
        .replace(
          /'/g,
          "&#039;"
        );

    const html =
      `<!doctype html>
<html lang="id">
<head>
  <meta charset="utf-8">
  <meta name="viewport" content="width=device-width,initial-scale=1">
  <meta name="robots" content="noindex,nofollow">
  <title>${safe(title)}</title>
</head>
<body style="margin:0;background:#f6f3ed;font-family:Arial,Helvetica,sans-serif;color:#171717">
  <main style="min-height:100vh;display:flex;align-items:center;justify-content:center;padding:24px;box-sizing:border-box">
    <section style="width:100%;max-width:560px;background:#fff;border:1px solid #e7e5e4;border-radius:24px;overflow:hidden;box-shadow:0 20px 55px rgba(0,0,0,.08)">
      <div style="height:8px;background:linear-gradient(90deg,#8c0d10,#b8892e)"></div>

      <div style="padding:38px 30px;text-align:center">
        <div style="width:64px;height:64px;margin:0 auto 20px;border-radius:50%;display:flex;align-items:center;justify-content:center;background:${soft};color:${accent};font-size:32px;font-weight:900">
          ${success ? "✓" : "!"}
        </div>

        <div style="font-size:11px;font-weight:900;letter-spacing:.16em;color:${accent}">
          ${label}
        </div>

        <h1 style="margin:14px 0 10px;font-size:28px;line-height:1.2">
          ${safe(title)}
        </h1>

        <p style="margin:0 auto;max-width:450px;color:#57534e;font-size:15px;line-height:1.7">
          ${safe(message)}
        </p>

        <a
          href="https://umkmnaikkelas.id/"
          style="display:inline-flex;align-items:center;justify-content:center;min-height:48px;margin-top:28px;padding:0 22px;border-radius:999px;background:#8c0d10;color:#fff;text-decoration:none;font-weight:800"
        >
          Kembali ke UMKM Naik Kelas
        </a>

        <div style="margin-top:24px;color:#a8a29e;font-size:11px">
          UMKM Naik Kelas
        </div>
      </div>
    </section>
  </main>
</body>
</html>`;

    return new Response(
      request.method === "HEAD"
        ? null
        : html,
      {
        status:
          statusCode,
        headers: {
          "Content-Type":
            "text/html; charset=UTF-8",
          "Cache-Control":
            "no-store, no-cache, must-revalidate",
          "Pragma":
            "no-cache",
          "X-Robots-Tag":
            "noindex, nofollow",
          "Referrer-Policy":
            "no-referrer",
          "X-Content-Type-Options":
            "nosniff",
          "X-Frame-Options":
            "DENY",
        },
      }
    );
  }

  if (
    !env
      .GOOGLE_AFFILIATE_WEBHOOK_URL
  ) {
    return renderVerificationPage(
      false,
      "Layanan verifikasi belum siap.",
      "Silakan coba kembali beberapa saat lagi.",
      503
    );
  }

  const url =
    new URL(
      request.url
    );

  const token =
    String(
      url.searchParams.get(
        "token"
      ) || ""
    ).trim();

  if (
    !token ||
    token.length < 20
  ) {
    return renderVerificationPage(
      false,
      "Tautan verifikasi tidak valid.",
      "Silakan gunakan tautan terbaru dari email pendaftaran Anda.",
      400
    );
  }

  let appsScriptUrl;

  try {
    appsScriptUrl =
      new URL(
        env
          .GOOGLE_AFFILIATE_WEBHOOK_URL
      );

    appsScriptUrl.search = "";

    appsScriptUrl.searchParams.set(
      "action",
      "verify_json"
    );

    appsScriptUrl.searchParams.set(
      "token",
      token
    );

  } catch (error) {
    return renderVerificationPage(
      false,
      "Konfigurasi verifikasi bermasalah.",
      "Silakan hubungi admin UMKM Naik Kelas.",
      503
    );
  }

  let response;

  try {
    response =
      await fetch(
        appsScriptUrl.toString(),
        {
          method:
            "GET",
          redirect:
            "follow",
          headers: {
            "Accept":
              "application/json",
          },
        }
      );

  } catch (error) {
    console.error(
      "Affiliate verification downstream connection failed."
    );

    return renderVerificationPage(
      false,
      "Layanan verifikasi sementara tidak dapat diakses.",
      "Silakan coba kembali beberapa saat lagi.",
      502
    );
  }

  const text =
    await response.text();

  let result = null;

  try {
    result =
      text
        ? JSON.parse(text)
        : null;
  } catch (error) {
    result = null;
  }

  if (
    !response.ok ||
    !result ||
    typeof result !==
      "object"
  ) {
    console.error(
      JSON.stringify({
        channel:
          "affiliate_verification",
        downstreamStatus:
          response.status,
        jsonValid:
          Boolean(result),
      })
    );

    return renderVerificationPage(
      false,
      "Verifikasi belum dapat diproses.",
      "Silakan coba kembali menggunakan tautan terbaru dari email Anda.",
      502
    );
  }

  if (
    result.ok &&
    result.verified
  ) {
    return renderVerificationPage(
      true,
      result.title ||
        "Email berhasil diverifikasi.",
      result.message ||
        "Pendaftaran Anda akan ditinjau admin maksimal 1x24 jam. Setelah disetujui, Anda akan menerima email aktivasi dan dapat login menggunakan OTP. Jika setelah 1x24 jam belum ada kabar, silakan hubungi CS UMKM Naik Kelas melalui halo@umkmnaikkelas.id.",
      200
    );
  }

  return renderVerificationPage(
    false,
    result.title ||
      "Verifikasi belum berhasil.",
    result.message ||
      "Silakan gunakan tautan terbaru dari email pendaftaran Anda.",
    400
  );
}

<script>
(function(){
  const SESSION_KEY = 'umkmnaikkelas_affiliate_admin_session_v1';

  const state = {
    email: '',
    token: '',
    dashboard: null,
    affiliates: [],
    commissions: [],

    /*
     * Mencegah request restore session lama
     * menimpa layar OTP login baru.
     */
    authFlowId: 0,
  };

  const $ = (id) => document.getElementById(id);

  function esc(value){
    return String(value ?? '')
      .replace(/&/g,'&amp;')
      .replace(/</g,'&lt;')
      .replace(/>/g,'&gt;')
      .replace(/"/g,'&quot;')
      .replace(/'/g,'&#039;');
  }

  function money(value){
    const n = Number(value || 0);
    return new Intl.NumberFormat('id-ID',{
      style:'currency',
      currency:'IDR',
      maximumFractionDigits:0
    }).format(Number.isFinite(n) ? n : 0);
  }

  function number(value){
    const n = Number(value || 0);
    return new Intl.NumberFormat('id-ID').format(Number.isFinite(n) ? n : 0);
  }

  function dateText(value){
    if(!value) return '-';
    const d = new Date(value);
    if(Number.isNaN(d.getTime())) return '-';

    return new Intl.DateTimeFormat('id-ID',{
      day:'2-digit',
      month:'short',
      year:'numeric',
      hour:'2-digit',
      minute:'2-digit'
    }).format(d);
  }

  function message(target,type,text){
    const el = $(target);

    if(!text){
      el.innerHTML = '';
      return;
    }

    el.innerHTML =
      '<div class="msg ' +
      (type === 'ok' ? 'ok' : 'err') +
      '">' +
      esc(text) +
      '</div>';
  }

  async function api(payload){
    const res = await fetch('/portal',{
      method:'POST',
      headers:{
        'Content-Type':'application/json'
      },
      body:JSON.stringify({
        ...payload,
        userAgent:navigator.userAgent || ''
      })
    });

    let data = {};

    try{
      data = await res.json();
    }catch(e){}

    if(!res.ok || data.ok === false){
      const error = new Error(
        data.message ||
        data.error ||
        'Portal admin sedang tidak dapat diakses.'
      );

      error.data = data;
      throw error;
    }

    return data;
  }

  function readSession(){
    try{
      const raw = localStorage.getItem(SESSION_KEY);
      if(!raw) return null;

      const data = JSON.parse(raw);

      if(!data.token || !data.expiresAt){
        localStorage.removeItem(SESSION_KEY);
        return null;
      }

      const expires = new Date(data.expiresAt).getTime();

      if(!expires || Date.now() >= expires){
        localStorage.removeItem(SESSION_KEY);
        return null;
      }

      return data;
    }catch(e){
      localStorage.removeItem(SESSION_KEY);
      return null;
    }
  }

  function saveSession(token,expiresAt,email){
    localStorage.setItem(
      SESSION_KEY,
      JSON.stringify({
        token,
        expiresAt,
        email
      })
    );
  }

  function clearSession(){
    localStorage.removeItem(SESSION_KEY);
  }

  function showLogin(stage){
    $('dashboardView').classList.add('hidden');
    $('loginView').classList.remove('hidden');

    if(stage === 'otp'){
      $('emailForm').classList.add('hidden');
      $('otpForm').classList.remove('hidden');
      $('otpEmailLabel').textContent = state.email;
      $('adminOtp').focus();
    }else{
      $('otpForm').classList.add('hidden');
      $('emailForm').classList.remove('hidden');
      $('adminEmail').value = state.email || '';
    }
  }

  function showDashboard(){
    $('loginView').classList.add('hidden');
    $('dashboardView').classList.remove('hidden');
  }

  function setBusy(button,busy,busyText,normalText){
    button.disabled = busy;
    button.textContent = busy ? busyText : normalText;
  }

  async function requestOtp(){
    const email =
      String($('adminEmail').value || '')
        .trim()
        .toLowerCase();

    if(!/^[^\\s@]+@[^\\s@]+\\.[^\\s@]+$/.test(email)){
      message('loginMessage','err','Masukkan alamat email admin yang valid.');
      return;
    }

    /*
     * Login baru selalu membatalkan restore
     * session lama yang mungkin masih berjalan.
     */
    state.authFlowId += 1;
    const thisFlowId =
      state.authFlowId;

    clearSession();

    state.token = '';
    state.dashboard = null;
    state.affiliates = [];
    state.commissions = [];
    state.email = email;

    const btn = $('sendOtpBtn');
    setBusy(btn,true,'Mengirim OTP…','Kirim Kode OTP Admin');
    message('loginMessage','', '');

    try{
      const data = await api({
        action:'admin_request_otp',
        email
      });

      /*
       * Abaikan response request lama
       * jika flow login sudah berubah.
       */
      if(
        thisFlowId !==
        state.authFlowId
      ){
        return;
      }

      showLogin('otp');

      message(
        'loginMessage',
        'ok',
        data.message ||
        'Kode OTP sudah dikirim ke email administrator.'
      );

    }catch(err){
      if(
        thisFlowId !==
        state.authFlowId
      ){
        return;
      }

      showLogin('email');

      message(
        'loginMessage',
        'err',
        err.message ||
        'Akses ditolak.'
      );

    }finally{
      setBusy(btn,false,'Mengirim OTP…','Kirim Kode OTP Admin');
    }
  }

  async function verifyOtp(){
    const otp =
      String($('adminOtp').value || '')
        .replace(/\\D/g,'')
        .slice(0,6);

    if(!/^\\d{6}$/.test(otp)){
      message('loginMessage','err','Masukkan kode OTP 6 digit.');
      return;
    }

    const btn = $('verifyOtpBtn');
    setBusy(btn,true,'Memverifikasi…','Masuk ke Portal Admin');
    message('loginMessage','', '');

    try{
      const data = await api({
        action:'admin_verify_otp',
        email:state.email,
        otp
      });

      if(!data.authenticated || !data.sessionToken){
        throw new Error(
          data.message ||
          'Kode OTP salah atau sudah kedaluwarsa.'
        );
      }

      state.authFlowId += 1;
      state.token = data.sessionToken;

      saveSession(
        data.sessionToken,
        data.expiresAt,
        state.email
      );

      $('adminOtp').value = '';

      await loadDashboard();
    }catch(err){
      message('loginMessage','err',err.message);
    }finally{
      setBusy(btn,false,'Memverifikasi…','Masuk ke Portal Admin');
    }
  }

  async function loadDashboard(){
    const thisFlowId =
      state.authFlowId;

    const tokenAtStart =
      state.token;

    if(!tokenAtStart){
      /*
       * Jangan paksa balik ke email jika
       * pengguna sudah berada di flow OTP.
       */
      return;
    }

    const btn = $('refreshBtn');

    if(btn){
      setBusy(btn,true,'Memuat…','Refresh');
    }

    message('dashMessage','', '');

    try{
      const data = await api({
        action:'admin_dashboard',
        sessionToken:tokenAtStart
      });

      /*
       * Jika selama request ini pengguna
       * memulai login baru, response lama
       * tidak boleh mengubah UI.
       */
      if(
        thisFlowId !==
        state.authFlowId ||
        tokenAtStart !==
        state.token
      ){
        return;
      }

      if(!data.authenticated){
        throw new Error('Sesi admin berakhir. Silakan masuk kembali.');
      }

      state.dashboard = data;
      state.affiliates = Array.isArray(data.affiliates)
        ? data.affiliates
        : [];
      state.commissions = Array.isArray(data.commissions)
        ? data.commissions
        : [];

      renderDashboard();
      showDashboard();

    }catch(err){
      /*
       * Abaikan error dari request restore
       * session lama yang sudah dibatalkan
       * oleh login OTP baru.
       */
      if(
        thisFlowId !==
        state.authFlowId ||
        tokenAtStart !==
        state.token
      ){
        return;
      }

      clearSession();
      state.token = '';
      state.dashboard = null;
      state.affiliates = [];
      state.commissions = [];

      showLogin('email');

      message(
        'loginMessage',
        'err',
        err.message
      );

    }finally{
      if(btn){
        setBusy(btn,false,'Memuat…','Refresh');
      }
    }
  }

  function renderDashboard(){
    const d = state.dashboard || {};
    const s = d.summary || {};

    $('heroTotal').textContent = number(s.totalAffiliates);
    $('generatedAt').textContent = 'Update: ' + dateText(d.generatedAt);

    $('sTotal').textContent = number(s.totalAffiliates);
    $('sApproved').textContent = number(s.approved);
    $('sPending').textContent = number(s.pending);
    $('sReview').textContent = number(s.review);
    $('sVerified').textContent = number(s.emailVerified);

    $('sPV').textContent = number(s.totalPageViews);
    $('sCheckout').textContent = number(s.totalCheckoutClicks);
    $('sPurchases').textContent = number(s.totalPurchases);
    $('sRevenue').textContent = money(s.grossRevenue);

    $('sEarned').textContent = money(s.commissionEarned);
    $('sCommissionPending').textContent = money(s.commissionPending);
    $('sCommissionApproved').textContent = money(s.commissionApproved);
    $('sCommissionPaid').textContent = money(s.commissionPaid);

    renderRows();
    renderCommissionRows();
  }

  function renderCommissionRows(){
    const rows =
      state.commissions || [];

    $('commissionDbCount').textContent =
      number(rows.length) +
      ' transaksi komisi';

    $('commissionEmptyState').classList.toggle(
      'hidden',
      rows.length !== 0
    );

    $('commissionRows').innerHTML =
      rows.map((c) => {
        const commissionStatus =
          String(
            c.commissionStatus || ''
          ).toUpperCase();

        const payoutStatus =
          String(
            c.payoutStatus || ''
          ).toUpperCase();

        let action = '';

        if(
          commissionStatus === 'PENDING'
        ){
          action =
            '<button type="button" class="action-btn action-approve" data-commission-action="1" data-id="' +
            esc(c.commissionId || '') +
            '" data-code="' +
            esc(c.affiliateCode || '') +
            '" data-name="' +
            esc(c.affiliateName || 'Affiliate') +
            '" data-status="APPROVED">Approve Komisi</button>';
        }else if(
          commissionStatus === 'APPROVED' &&
          payoutStatus !== 'PAID'
        ){
          action =
            '<button type="button" class="action-btn action-suspend" data-commission-action="1" data-id="' +
            esc(c.commissionId || '') +
            '" data-code="' +
            esc(c.affiliateCode || '') +
            '" data-name="' +
            esc(c.affiliateName || 'Affiliate') +
            '" data-status="PAID">Tandai Dibayar</button>';
        }else{
          action =
            '<span class="status">' +
            esc(
              payoutStatus === 'PAID'
                ? 'SELESAI'
                : commissionStatus || '-'
            ) +
            '</span>';
        }

        return '<tr>' +
          '<td class="nowrap muted">' + esc(dateText(c.paymentDate)) + '</td>' +
          '<td><div class="strong">' + esc(c.affiliateName || '-') + '</div><div class="code" style="margin-top:4px">' + esc(c.affiliateCode || '-') + '</div></td>' +
          '<td><div class="break">' + esc(c.transactionId || '-') + '</div></td>' +
          '<td class="strong">' + esc(c.product || '-') + '</td>' +
          '<td class="right strong nowrap">' + esc(money(c.saleAmount)) + '</td>' +
          '<td class="right strong nowrap">' + esc(money(c.baseCommission)) + '</td>' +
          '<td><span class="status">' + esc(commissionStatus || '-') + '</span></td>' +
          '<td><span class="status">' + esc(payoutStatus || '-') + '</span></td>' +
          '<td class="nowrap muted">' + esc(dateText(c.approvedAt)) + '</td>' +
          '<td class="nowrap muted">' + esc(dateText(c.paidAt)) + '</td>' +
          '<td><div class="break">' + esc(c.payoutId || '-') + '</div></td>' +
          '<td><div class="action-stack">' + action + '</div></td>' +
        '</tr>';
      }).join('');
  }

  function renderRows(){
    const q =
      String($('searchBox').value || '')
        .trim()
        .toLowerCase();

      String($('statusFilter').value || 'ALL')

        a.name,
        a.email,
        a.whatsapp,
        a.city,
        a.profession,
        a.referralCode,
        a.slug,
        a.social

      number(rows.length) +

 async function updateAffiliateStatus(
    referralCode,
    affiliateName,
    newStatus,
    clickedButton
  ){
    const isApprove =
      newStatus === 'APPROVED';

    if(!state.token){
      const sessionMessage =
        'Sesi admin tidak tersedia.';

      message(
        'dashMessage',
        'err',
        sessionMessage
      );

      /*
       * Feedback tambahan HANYA untuk Approve.
       */
      if(isApprove){
        window.alert(
          sessionMessage
        );
      }

      return;
    }

    const actionLabel =
      newStatus === 'APPROVED'
        ? 'APPROVE'
        : newStatus === 'REJECTED'
          ? 'REJECT'
          : 'SUSPEND';

    const confirmed =
      window.confirm(
        actionLabel +
        ' affiliate ' +
        affiliateName +
        ' (' +
        referralCode +
        ')?'
      );

    if(!confirmed){
      return;
    }

    const notes =
      window.prompt(
        'Catatan admin (opsional):',
        ''
      );

    if(notes === null){
      return;
    }

    message(
      'dashMessage',
      '',
      ''
    );

    /*
     * Visual proses HANYA untuk tombol Approve yang diklik.
     * Reject / Suspend tetap memakai perilaku sebelumnya.
     */
    const originalText =
      clickedButton
        ? clickedButton.textContent
        : '';

    if(
      isApprove &&
      clickedButton
    ){
      clickedButton.textContent =
        'Memproses…';
    }

    /*
     * Pertahankan perilaku lama:
     * semua tombol action dinonaktifkan selama request berlangsung.
     */
    document
      .querySelectorAll(
        '[data-affiliate-action]'
      )
      .forEach(
        (btn) => {
          btn.disabled = true;
        }
      );

    try{
      /*
       * Payload API TIDAK DIUBAH.
       */
      const data =
        await api({
          action:
            'admin_update_affiliate_status',
          sessionToken:
            state.token,
          referralCode:
            referralCode,
          newStatus:
            newStatus,
          adminNotes:
            notes || ''
        });

      const successMessage =
        data.message ||
        'Status affiliate berhasil diperbarui.';

      /*
       * PENTING:
       * Refresh dashboard lebih dahulu.
       * loadDashboard() mengosongkan dashMessage di awal,
       * jadi pesan sukses harus dipasang SETELAH refresh selesai.
       */
      await loadDashboard();

      message(
        'dashMessage',
        'ok',
        successMessage
      );

      /*
       * Feedback popup HANYA untuk Approve,
       * supaya admin selalu melihat hasil klik.
       */
      if(isApprove){
        window.alert(
          successMessage +
          (
            data.approvalEmailError
              ? '\\n\\nStatus sudah APPROVED, tetapi email aktivasi belum terkirim: ' +
  data.approvalEmailError
              : ''
          )
        );
      }

    }catch(err){
      const errorMessage =
        err &&
        err.message
          ? err.message
          : 'Status affiliate belum dapat diperbarui.';

      message(
        'dashMessage',
        'err',
        errorMessage
      );

      /*
       * Error popup HANYA untuk Approve.
       */
      if(isApprove){
        window.alert(
          errorMessage
        );
      }

    }finally{
      /*
       * Pertahankan perilaku lama.
       */
      document
        .querySelectorAll(
          '[data-affiliate-action]'
        )
        .forEach(
          (btn) => {
            btn.disabled = false;
          }
        );

      /*
       * Kembalikan teks tombol HANYA untuk Approve.
       */
      if(
        isApprove &&
        clickedButton
      ){
        clickedButton.textContent =
          originalText ||
          'Approve';
      }
    }
  }

function adminPortalResponse(request) {
  if (
    request.method === "HEAD"
  ) {
    return new Response(
      null,
      {
        status: 200,
        headers: {
          "Content-Type":
            "text/html; charset=UTF-8",
          "Cache-Control":
            "no-store",
          "X-Robots-Tag":
            "noindex, nofollow",
        },
      }
    );
  }

  if (
    request.method !== "GET"
  ) {
    return jsonResponse(
      {
        ok: false,
        error:
          "METHOD_NOT_ALLOWED",
      },
      405
    );
  }

  return new Response(
    adminPortalHtml(),
    {
      status: 200,
      headers: {
        "Content-Type":
          "text/html; charset=UTF-8",
        "Cache-Control":
          "no-store, no-cache, must-revalidate",
        "Pragma":
          "no-cache",
        "X-Robots-Tag":
          "noindex, nofollow",
        "Referrer-Policy":
          "no-referrer",
        "X-Content-Type-Options":
          "nosniff",
        "X-Frame-Options":
          "DENY",
        "Content-Security-Policy":
          "default-src 'self'; connect-src 'self'; img-src 'self' data:; style-src 'self' 'unsafe-inline'; script-src 'self' 'unsafe-inline'; frame-ancestors 'none'; base-uri 'none'; form-action 'self'",
      },
    }
  );
}



/* =========================================================
   MAIN WORKER
   ========================================================= */

export default {
  async fetch(
    request,
    env
  ) {
    const url =
      new URL(
        request.url
      );

    /*
     * Universal safe request log.
     * Tidak pernah log nilai token.
     */
    console.log(
      JSON.stringify({
        timestamp:
          new Date()
            .toISOString(),

        stage:
          "REQUEST_RECEIVED",

        method:
          request.method,

        path:
          url.pathname,

        hasTokenQuery:
          url.searchParams.has(
            "token"
          ),
      })
    );

    /*
     * Validasi link affiliate Lynk.id Paket Panglima UMKM.
     */
    if (
      url.pathname ===
        "/validate-lynk-affiliate" ||
      url.pathname ===
        "/validate-lynk-affiliate/"
    ) {
      return handleLynkAffiliateValidation(
        request
      );
    }

    /*
     * Pendaftaran affiliate hanya diteruskan ke Apps Script
     * setelah link Lynk.id lolos validasi live.
     */
    if (
      url.pathname ===
        "/affiliate-register" ||
      url.pathname ===
        "/affiliate-register/"
    ) {
      return handleAffiliateRegistration(
        request,
        env
      );
    }

    /*
     * Stable public email verification URL.
     * Token tidak pernah dicatat pada log.
     */
    if (
      url.pathname === "/verify" ||
      url.pathname === "/verify/"
    ) {
      return handleAffiliateVerification(
        request,
        env
      );
    }

    /*
     * STEP 10C:
     * Admin Affiliate Portal UI.
     * Tidak membutuhkan Hostinger/Horizons.
     */
    if (
      url.pathname === "/admin" ||
      url.pathname === "/admin/"
    ) {
      return adminPortalResponse(
        request
      );
    }

    /*
     * STEP 10B:
     * Affiliate Portal endpoint.
     */
    if (
      url.pathname ===
      "/portal"
    ) {
      return handlePortal(
        request,
        env
      );
    }

    /*
     * Root health check /
     */
    if (
      request.method ===
        "GET" ||
      request.method ===
        "HEAD"
    ) {
      return jsonResponse({
        ok: true,
        service:
          "Panglima Lynk Webhook + Affiliate Portal",
        status:
          "READY",
        version:
          WORKER_VERSION,

        merchantKeyConfigured:
          Boolean(
            env
              .LYNK_MERCHANT_KEY
          ),

        portalReady:
          Boolean(
            env
              .GOOGLE_AFFILIATE_WEBHOOK_URL
          ),

        portalEndpoint:
          "/portal",

        adminPortal:
          "/admin",

        lynkAffiliateValidation:
          "/validate-lynk-affiliate",

        affiliateRegistration:
          "/affiliate-register",
      });
    }

    if (
      request.method !==
      "POST"
    ) {
      return jsonResponse(
        {
          ok: false,
          error:
            "METHOD_NOT_ALLOWED",
        },
        405
      );
    }

    /*
     * LYNK WEBHOOK
     */
    const {
      raw,
      parsed,
      contentType,
    } =
      await readIncomingBody(
        request
      );

    const eventType =
      getEventType(
        parsed
      );

    const refId =
      getRefId(
        parsed
      );

    const items =
      getItems(
        parsed
      );

    const grandTotal =
      getGrandTotal(
        parsed
      );

    const suppliedToken =
      url.searchParams.get(
        "token"
      ) || "";

    const hasValidToken =
      Boolean(
        env
          .LYNK_WEBHOOK_TOKEN
      ) &&
      suppliedToken ===
        env
          .LYNK_WEBHOOK_TOKEN;

    const signature =
      request.headers.get(
        "x-signature"
      ) ||
      request.headers.get(
        "x-lynk-signature"
      ) ||
      "";

    const signatureDiagnostic =
      await diagnoseLynkSignature(
        raw,
        signature,
        env
          .LYNK_MERCHANT_KEY ||
          ""
      );

    console.log(
      JSON.stringify({
        timestamp:
          new Date()
            .toISOString(),

        method:
          request.method,

        event:
          eventType ||
          "UNKNOWN",

        refId:
          refId ||
          null,

        itemCount:
          items.length,

        grandTotal,

        hasToken:
          Boolean(
            suppliedToken
          ),

        validToken:
          hasValidToken,

        hasSignature:
          Boolean(
            signature
          ),

        signatureLength:
          signature
            ? signature.length
            : 0,

        merchantKeyReady:
          signatureDiagnostic
            .merchantKeyReady,

        signatureMatch:
          signatureDiagnostic
            .match,
      })
    );

    const looksLikeRealPayment =
      (
        eventType ===
          "payment.received" ||
        eventType ===
          "payment_received"
      ) &&
      Boolean(
        refId
      ) &&
      items.length > 0 &&
      grandTotal > 0;

    /*
     * Lynk Test URL / dummy payload:
     * tetap 200.
     */
    if (
      !looksLikeRealPayment
    ) {
      return jsonResponse({
        ok: true,
        received: true,
        test: true,
        ignored: true,

        event:
          eventType ||
          null,

        signatureDetected:
          Boolean(
            signature
          ),

        signatureMatch:
          signatureDiagnostic
            .match,
      });
    }

    /*
     * Real payment masih wajib token.
     */
    if (
      !hasValidToken
    ) {
      return jsonResponse(
        {
          ok: false,
          error:
            "UNAUTHORIZED_PAYMENT_WEBHOOK",
        },
        401
      );
    }

    const downstream =
      await forwardToGoogle(
        env,
        raw,
        contentType,
        "panglima-lynk-webhook",
        true
      );

    console.log(
      JSON.stringify({
        timestamp:
          new Date()
            .toISOString(),

        event:
          eventType,

        refId:
          refId ||
          null,

        downstreamStatus:
          downstream.status,

        signatureMatch:
          signatureDiagnostic
            .match,
      })
    );

    if (
      !downstream.ok
    ) {
      return jsonResponse(
        {
          ok: false,
          received: true,
          forwarded: false,

          error:
            downstream.error ||
            "DOWNSTREAM_ERROR",

          downstreamStatus:
            downstream.status,
        },
        502
      );
    }

    return jsonResponse({
      ok: true,
      received: true,
      forwarded: true,

      event:
        eventType,

      refId:
        refId ||
        null,

      signatureDetected:
        Boolean(
          signature
        ),

      signatureMatch:
        signatureDiagnostic
          .match,
    });
  },
};
